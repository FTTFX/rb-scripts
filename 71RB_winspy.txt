-- formatter
local function fmt(v)
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then local p = v.Position return string.format("CF(%.0f,%.0f,%.0f)", p.X, p.Y, p.Z)
    elseif t == "string" then return #v > 40 and '"'..v:sub(1,40)..'..."' or '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    elseif t == "boolean" then return tostring(v)
    else return t end
end
local function fmtArgs(...)
    local n = select("#", ...)
    if n == 0 then return "()" end
    local p = {}
    for i = 1, math.min(n, 8) do p[i] = fmt((select(i, ...))) end
    return table.concat(p, " | ")
end
local nameCache = setmetatable({}, {__mode = "k"})
local function getName(r)
    local n = nameCache[r]
    if n == nil then local ok, f = pcall(function() return r:GetFullName() end); n = ok and f or "?"; nameCache[r] = n end
    return n
end

-- hook __namecall → ring buffer (FireServer/InvokeServer)
if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local m = getnc()
        if m == "FireServer" or m == "InvokeServer" then
            local nm = getName(self):gsub("^.-Remotes%.", "")
            local r = S.ring
            r[#r+1] = {t = os.clock(), s = m.." @"..nm.." "..fmtArgs(...)}
            if #r > RINGMAX then table.remove(r, 1) end
        end
        return oldNc(self, ...)
    end))
    S.restoreNc = oldNc
end

-- ==================== SET PAD: ดูดข้อมูล + ดักการแตะ ====================
local function dumpPad(p)
    add("📦 PAD = "..getName(p))
    add(string.format("  Class:%s CanTouch:%s CanCollide:%s", p.ClassName, tostring(p.CanTouch), tostring(p.CanCollide)))
    local attrs = p:GetAttributes()
    local na = 0
    for k, v in pairs(attrs) do add("  attr "..k.." = "..tostring(v)); na = na + 1 end
    if na == 0 then add("  (ไม่มี attribute)") end
    for _, c in ipairs(p:GetChildren()) do add("  child "..c.ClassName..":"..c.Name) end
    -- หา value ชื่อ debounce/cooldown/time แถวๆแผ่น
    local scope = p.Parent or p
    for _, v in ipairs(scope:GetDescendants()) do
        local ln = v.Name:lower()
        if v:IsA("ValueBase") and (ln:find("debounce") or ln:find("cooldown") or ln:find("cd") or ln:find("time")) then
            add("  🔎 "..v.ClassName..":"..v.Name.." = "..tostring(v.Value))
        end
    end
end

setPadB.MouseButton1Click:Connect(function()
    local root = hrp(); if not root then add("❌ ไม่มีตัวละคร"); return end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = {LP.Character}
    local res = workspace:Raycast(root.Position, Vector3.new(0,-15,0), rp)
    if not res then add("❌ ไม่เจอแผ่นใต้เท้า (ยืนบนแผ่นเหลืองก่อน)"); return end
    local p = res.Instance
    S.pad = p
    if S.padConn then pcall(function() S.padConn:Disconnect() end) end
    S.padConn = p.Touched:Connect(function(hit)
        if hit and hit:IsDescendantOf(LP.Character) then add(string.format("👆 TOUCH @%.2f", os.clock())) end
    end)
    CONNS[#CONNS+1] = S.padConn
    add("===== SET PAD =====")
    dumpPad(p)
    add("ตอนนี้เหยียบเข้า-ออกแผ่นได้เลย ดู TOUCH vs WIN")
end)

-- ==================== ประวัติ part ใต้เท้า (จับแผ่น win ก่อนวาป) ====================
local posHist = {}
local rpFoot = RaycastParams.new()
rpFoot.FilterType = Enum.RaycastFilterType.Exclude
bind(RunService.Heartbeat, function()
    local root = hrp(); if not root then return end
    rpFoot.FilterDescendantsInstances = {LP.Character}
    local res = workspace:Raycast(root.Position, Vector3.new(0,-8,0), rpFoot)
    posHist[#posHist+1] = {t = os.clock(), part = res and res.Instance, p = root.Position}
    if #posHist > 40 then table.remove(posHist, 1) end
end)

-- ==================== Win watch ====================
local lastWinT = 0
local function report(name, old, new)
    local now = os.clock()
    local gap = lastWinT > 0 and string.format(" (ห่างครั้งก่อน %.2fs)", now - lastWinT) or ""
    lastWinT = now
    add("🏆 "..name..": "..old.." → "..new..gap)
    -- จับแผ่น finish = part ใต้เท้า "เฟรมก่อนวาป" (ตรวจระยะกระโดดไกล)
    local pad, jumpFrom
    for i = #posHist, 2, -1 do
        local a, b = posHist[i-1], posHist[i]
        if a.p and b.p and (a.p - b.p).Magnitude > 40 then pad, jumpFrom = a.part, a.p; break end
    end
    if pad then
        S.pad, S.lastWin = pad, CFrame.new(jumpFrom)
        add("🎯 แผ่น finish (ก่อนวาป) = "..getName(pad))
        add("   → _G.WINSPY.pad พร้อม firetouchinterest")
    else
        local last = posHist[#posHist]
        if last and last.part then S.pad = last.part; add("🎯 (ไม่เจอวาป) ใต้เท้า = "..getName(last.part)) end
    end
    -- remote ใน 3s ก่อนหน้า (ถ้ามี)
    for i = #S.ring, 1, -1 do
        local e = S.ring[i]
        if now - e.t <= 3 then add(string.format("   remote -%.2fs %s", now - e.t, e.s)) end
    end
end

local function candidates()
    local out, ls = {}, LP:FindFirstChild("leaderstats")
    if ls then for _, v in ipairs(ls:GetChildren()) do if v:IsA("ValueBase") then out[#out+1] = v end end end
    for _, v in ipairs(LP:GetDescendants()) do
        if v:IsA("ValueBase") and v.Name:lower():find("win") then return {v} end
    end
    return out
end

local cands = candidates()
if #cands == 0 then
    add("❌ หา Win/leaderstats ไม่เจอ — บอกชื่อ stat มา")
else
    local names = {}
    for _, v in ipairs(cands) do
        names[#names+1] = v.Name
        S.prev[v] = v.Value
        bind(v.Changed, function(nv)
            local ov = S.prev[v]; S.prev[v] = nv
            if type(nv) == "number" and type(ov) == "number" and nv > ov then report(v.Name, ov, nv) end
        end)
    end
    add("👀 เฝ้า: "..table.concat(names, ", "))
    add("ยืนบนแผ่นเหลือง → SET PAD")
end
