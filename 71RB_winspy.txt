-- วิธีใช้: รันก่อน → วิ่งถึงเส้น finish ให้ชนะ 1 ครั้ง → มันจำ CFrame จุดที่ชนะ
--   ถ้าไม่มี remote = win เป็น touch/region ที่ตำแหน่งนั้น → เอาพิกัดไปทำ teleport-farm
--   พิกัดเก็บใน _G.WINSPY.lastWin (CFrame) + copy clipboard อัตโนมัติ
local VERSION = "1.1"
local Players, RunService = game:GetService("Players"), game:GetService("RunService")
local LP = Players.LocalPlayer

local hookmeta = hookmetamethod
local getnc    = getnamecallmethod
local newcc    = newcclosure or function(f) return f end
local clip     = setclipboard or toclipboard
local okHooks  = hookmeta and getnc

-- state ถาวร (hook ครั้งเดียว, ring คงค่าข้าม re-run)
_G.WINSPY = _G.WINSPY or {}
local S = _G.WINSPY
S.ring = S.ring or {}        -- {t=os.clock(), s="..."}
S.prev = S.prev or {}        -- ค่าเดิมของแต่ละ value
local RINGMAX = 50

-- guard (GUI/conns เท่านั้น — hook ถาวร)
if _G.WINSPY_CONNS then for _, c in pairs(_G.WINSPY_CONNS) do pcall(function() c:Disconnect() end) end end
_G.WINSPY_CONNS = {}
local CONNS = _G.WINSPY_CONNS
local function bind(s, f) local c = s:Connect(f); CONNS[#CONNS+1] = c; return c end
pcall(function() ((gethui and gethui()) or LP.PlayerGui):FindFirstChild("WINSPYGUI"):Destroy() end)

-- formatter
local function fmt(v)
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then local p = v.Position return string.format("CF(%.0f,%.0f,%.0f)", p.X, p.Y, p.Z)
    elseif t == "string" then return #v > 40 and '"'..v:sub(1,40)..'..."' or '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    elseif t == "boolean" then return tostring(v)
    else return t end
end
local function fmtArgs(...)
    local n = select("#", ...)
    if n == 0 then return "()" end
    local p = {}
    for i = 1, math.min(n, 8) do p[i] = fmt((select(i, ...))) end
    return table.concat(p, " | ")
end

local nameCache = setmetatable({}, {__mode = "k"})
local function getName(r)
    local n = nameCache[r]
    if n == nil then local ok, f = pcall(function() return r:GetFullName() end); n = ok and f or "?"; nameCache[r] = n end
    return n
end

-- hook __namecall → เก็บ FireServer/InvokeServer ใน ring buffer
if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local m = getnc()
        if m == "FireServer" or m == "InvokeServer" then
            local nm = getName(self):gsub("^.-Remotes%.", "")
            local r = S.ring
            r[#r+1] = {t = os.clock(), s = m.." @"..nm.." "..fmtArgs(...)}
            if #r > RINGMAX then table.remove(r, 1) end
        end
        return oldNc(self, ...)
    end))
    S.restoreNc = oldNc
end

-- ==================== Win detect + report ====================
local function hrp() local c = LP.Character; return c and c:FindFirstChild("HumanoidRootPart") end

local function report(name, old, new)
    local now = os.clock()
    local lines = {"🏆 "..name..": "..tostring(old).." → "..tostring(new)}
    local root = hrp()
    if root then
        S.lastWin = root.CFrame
        local p = root.Position
        lines[#lines+1] = string.format("📍 จุดที่ชนะ: %.1f, %.1f, %.1f", p.X, p.Y, p.Z)
        pcall(function() if clip then clip(string.format("Vector3.new(%.2f,%.2f,%.2f)", p.X, p.Y, p.Z)) end end)
    end
    lines[#lines+1] = "FireServer ใน 3s ก่อนหน้า:"
    local n0 = #lines
    for i = #S.ring, 1, -1 do
        local e = S.ring[i]
        if now - e.t <= 3 then lines[#lines+1] = string.format("  -%.2fs  %s", now - e.t, e.s) end
    end
    if #lines == n0 then lines[#lines+1] = "  (ไม่มี remote — win=touch/region ที่จุดนี้)" end
    setText(table.concat(lines, "\n"))
end

-- หา value ที่จะเฝ้า: ชื่อมี "win" ก่อน, ไม่งั้นเอา leaderstats ทั้งหมด
local function candidates()
    local out, ls = {}, LP:FindFirstChild("leaderstats")
    if ls then for _, v in ipairs(ls:GetChildren()) do if v:IsA("ValueBase") then out[#out+1] = v end end end
    for _, v in ipairs(LP:GetDescendants()) do
        if v:IsA("ValueBase") and v.Name:lower():find("win") then return {v} end  -- win-named ชนะเสมอ
    end
    return out
end

local cands = candidates()
if #cands == 0 then
    setText("❌ หา Win/leaderstats ไม่เจอ\nไปทำให้ชนะ 1 ครั้งแล้วดู console / บอกชื่อ stat มา")
else
    local names = {}
    for _, v in ipairs(cands) do
        names[#names+1] = v.Name
        S.prev[v] = v.Value
        bind(v.Changed, function(nv)
            local ov = S.prev[v]; S.prev[v] = nv
            if type(nv) == "number" and type(ov) == "number" and nv > ov then report(v.Name, ov, nv) end
        end)
    end
    setText("👀 เฝ้าดู: "..table.concat(names, ", ").."\nไปทำให้ชนะ 1 ครั้ง...")
end
