-- ── Log ────────────────────────────────────────────────────────────
local logLines = {}
local logOrder = 0
local lastState = ""
local logColors = {
    G = Color3.fromRGB(80,255,120), Y = Color3.fromRGB(255,210,0),
    R = Color3.fromRGB(255,80,80), B = Color3.fromRGB(140,160,255),
    S = Color3.fromRGB(140,140,140),
}
local SCRIPT_T0 = os.clock()
local lastLogT = SCRIPT_T0
local MAX_LOG = 150
local logLabels = {}

local function log(txt, col)
    local c = logColors[col] or logColors.S
    local now = os.clock()
    txt = string.format("[%s +%.2fs] %s", os.date("%H:%M:%S"), now - lastLogT, txt)
    lastLogT = now
    logOrder += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.BackgroundTransparency = 1; l.Font = Enum.Font.Code
    l.TextSize = 11; l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = c; l.Text = txt; l.LayoutOrder = logOrder; l.Parent = scroll
    table.insert(logLines, txt)
    table.insert(logLabels, l)
    while #logLabels > MAX_LOG do
        local oldL = table.remove(logLabels, 1)
        if oldL then oldL:Destroy() end
    end
    while #logLines > MAX_LOG do table.remove(logLines, 1) end
    scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
end

local lastStateT = 0
local function setState(txt, col)
    if txt == lastState then return end
    local now = os.clock()
    if now - lastStateT < 0.3 then
        lastState = txt
        return
    end
    lastState = txt
    lastStateT = now
    log(txt, col)
end

function getCommF()
    local remotes = RS:FindFirstChild("Remotes")

local noclipParts, noclipChar, noclipRefreshed = {}, nil, 0
local function setNoclip(on)
    local char = LP.Character
    if not char then return end
    local now = os.clock()
    if char ~= noclipChar or now - noclipRefreshed > 2 then
        noclipChar = char
        noclipRefreshed = now
        table.clear(noclipParts)
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then table.insert(noclipParts, p) end
        end
    end
    for i = #noclipParts, 1, -1 do
        local p = noclipParts[i]
        if p.Parent then
            if p.CanCollide == on then p.CanCollide = not on end
        else
            table.remove(noclipParts, i)
        end
    end
end

task.spawn(function()
    while true do
        task.wait(0.2)
        if _G.BF04V3_LOOP then pcall(setNoclip, true) end
    end
end)

local BLOCKING_CLASSES = {
    BodyVelocity = true, BodyGyro = true, BodyPosition = true,
    BodyThrust = true, BodyAngularVelocity = true,
    AlignPosition = true, AlignOrientation = true,
    LinearVelocity = true, AngularVelocity = true, VectorForce = true,

local function releaseControl()
    pcall(function() if currentTween then currentTween:Cancel() currentTween = nil end end)
    pcall(function()
        if _G.BF04V3_BV then _G.BF04V3_BV:Destroy() _G.BF04V3_BV = nil end
    end)
    local char = LP.Character
    if not char then return end
    pcall(function()
        for _, obj in ipairs(char:GetDescendants()) do
            if BLOCKING_CLASSES[obj.ClassName] then obj:Destroy() end
        end
    end)
    pcall(function()
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Anchored = false
                if part.Name ~= "HumanoidRootPart" then part.CanCollide = true end
            end
        end
    end)
    pcall(function()
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.Sit = false
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            if hum.WalkSpeed <= 0 then hum.WalkSpeed = 16 end
            if hum.JumpPower <= 0 then hum.JumpPower = 50 end
            hum.AutoRotate = true
        end
    end)
    pcall(function()
        local pm = LP:FindFirstChild("PlayerScripts") and LP.PlayerScripts:FindFirstChild("PlayerModule")
        if pm then
            local controls = require(pm):GetControls()
            if controls then controls:Enable() end
        end
    end)
    setNoclip(false)
end

-- ── Smart Travel ──────────────────────────────────────────────────
local SEARCH_R = 3000
local SPAWN_DB = {}
local function rememberSpawn(name, pos)
    local list = SPAWN_DB[name]
    if not list then
        list = {}
        SPAWN_DB[name] = list
    end
    for _, p in ipairs(list) do
        if (p - pos).Magnitude < 200 then return end
    end
    if #list < 8 then table.insert(list, pos) end
end

local function flySmooth(destCF, speed)
    stopTween()
    local char, hrp = getChar()
    if not char or not hrp then return false end

    local dist = (destCF.Position - hrp.Position).Magnitude
    if dist < 15 then return true end

    local spd = speed or 380
    if dist >= 1000 then spd = math.min(spd, 250)
    elseif dist >= 500 then spd = math.min(spd, 300)
    elseif dist >= 250 then spd = math.min(spd, 600) end

    local tween = TweenService:Create(hrp,
        TweenInfo.new(dist / spd, Enum.EasingStyle.Linear),
        { CFrame = destCF })
    currentTween = tween
    tween:Play()
    tween.Completed:Wait()
    currentTween = nil

    local _, h2 = getChar()
    if not h2 then return false end
    local left = (destCF.Position - h2.Position).Magnitude
    return left <= 30
end

    repeat
        pcall(function()
            hum:ChangeState(15)
            hrp.CFrame = targetCF
            task.wait()
            hrp.CFrame = targetCF
        end)
        task.wait(0.5)
        local _, h = getChar()
        if not h then return false end
        hrp = h
    until (targetCF.Position - hrp.Position).Magnitude <= 1500 or os.clock() > deadline

    if (targetCF.Position - hrp.Position).Magnitude > 1500 then return false end

    pcall(function() CommF:InvokeServer("SetSpawnPoint") end)
    task.wait(0.2)
    pcall(function() CommF:InvokeServer("SetSpawnPoint") end)
    task.wait(0.2)
    pcall(function() CommF:InvokeServer("TeleportToSpawn") end)
    task.wait(1.5)
    return true
end

local function toTarget(targetCF)
    local char, hrp = getChar()
    if not char or not hrp then return false end
    local dist = (targetCF.Position - hrp.Position).Magnitude
    if dist < 25 then
        stopTween()
        hrp.CFrame = targetCF
        return true
    end
    if BTP_ON and dist > BTP_MIN_DIST then
        stopTween()
        if bypassTP(targetCF) then return true end
    end
    return flySmooth(targetCF, 420)
end

function tp(cframe) return toTarget(cframe) end

local QUEST_ITEMS = {
    Relic = true, Cup = true, Torch = true, Key = true,
}
function equipWeapon()
    local char, _, hum = getChar()
    if not char or not hum then return false end
    local held = char:FindFirstChildOfClass("Tool")
    if held and not QUEST_ITEMS[held.Name] then return true end
    if held then pcall(function() hum:UnequipTools() end); task.wait(0.1) end
    local bp = LP:FindFirstChild("Backpack")
    if not bp then return false end
    local tool = bp:FindFirstChild("Combat")
    if not tool then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and not QUEST_ITEMS[t.Name] then tool = t break end
        end
    end
    if tool then pcall(function() hum:EquipTool(tool) end); return true end
    return false
end

local function aimCam(mob)
    local mhrp = mob and (mob:FindFirstChild("HumanoidRootPart") or mob:FindFirstChild("Head"))
    if not mhrp then return end
    local cam = Workspace.CurrentCamera
    if cam then cam.CFrame = CFrame.lookAt(cam.CFrame.Position, mhrp.Position) end
end

local function closeDialogue()
    pcall(function()
        local main = pg:FindFirstChild("Main")
        if not main then return end
        for _, name in ipairs({"Dialogue", "Conversations", "Talk", "NPCDialog"}) do
            local d = main:FindFirstChild(name)
            if d and d:IsA("GuiObject") and d.Visible then d.Visible = false end
        end
    end)
end

local SIM_RADIUS_T = 0
local function boostSimulationRadius()
    local now = os.clock()
    if now - SIM_RADIUS_T < 5 then return end
    SIM_RADIUS_T = now
    pcall(function()
        if setsimulationradius then setsimulationradius(math.huge, math.huge) end
        if sethiddenproperty then
            sethiddenproperty(LP, "MaximumSimulationRadius", math.huge)
            sethiddenproperty(LP, "SimulationRadius", math.huge)
            sethiddenproperty(Workspace, "PhysicsSteppedMethod", "Transient")
        end
    end)
end

local function bringMobsToTarget(primaryTarget, mobName)
    if not primaryTarget or not primaryTarget.Parent then return end
    local _, myHrp = getChar()
    local pHrp = myHrp or primaryTarget:FindFirstChild("HumanoidRootPart")
    if not pHrp then return end
    boostSimulationRadius()
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return end
    for _, otherMob in ipairs(enemies:GetChildren()) do
        if otherMob ~= primaryTarget and otherMob ~= LP.Character then
            local isBoss = string.find(otherMob.Name, "Boss") or string.find(otherMob.Name, "King")
                or string.find(otherMob.Name, "Leader") or string.find(otherMob.Name, "Admiral")
            local oHum = otherMob:FindFirstChild("Humanoid")
            local oHrp = otherMob:FindFirstChild("HumanoidRootPart")
            local isQuestMob = mobName and string.find(otherMob.Name, mobName)
            if isQuestMob and not isBoss and oHum and oHrp and oHum.Health > 0 then
                pcall(function()
                    oHum.PlatformStand = true
                    oHum.Sit = true
                    oHrp.CanCollide = false
                    oHrp.Velocity = Vector3.new(0, 0, 0)
                    oHrp.Anchored = false
                    oHrp.CFrame = pHrp.CFrame * CFrame.new(math.random(-6, 6), 0, math.random(-6, 6))
                end)
            end
        end
    end
end

local function directRemoteAttack(mobTarget, mobName)
    if not mobTarget or not mobTarget.Parent then return end
    local rAttack, rHit = getCombatRemotes()
    pcall(function()
        local char = LP.Character
        if char then
            if char:FindFirstChild("Stun") then char.Stun.Value = 0 end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum.Sit = false end
        end
    end)
    local equipped = LP.Character and LP.Character:FindFirstChildOfClass("Tool")
    if not equipped then return end
    local mainPart = mobTarget:FindFirstChild("Head") or mobTarget:FindFirstChild("HumanoidRootPart")
    if not mainPart then return end
    local othersEnemies = { { mobTarget, mainPart } }
    if MAGNET_ON then
        local enemies = Workspace:FindFirstChild("Enemies")
        if enemies then
            for _, e in ipairs(enemies:GetChildren()) do
                if #othersEnemies >= 8 then break end
                if e ~= mobTarget then
                    local isBoss = string.find(e.Name, "Boss") or string.find(e.Name, "King")
                    local eHum = e:FindFirstChild("Humanoid")
                    local ePart = e:FindFirstChild("Head") or e:FindFirstChild("HumanoidRootPart")
                    if not isBoss and eHum and eHum.Health > 0 and ePart
                       and (not mobName or string.find(e.Name, mobName))
                       and (ePart.Position - mainPart.Position).Magnitude < 100 then
                        othersEnemies[#othersEnemies + 1] = { e, ePart }
                    end
                end
            end
        end
    end
    pcall(function()
        if rAttack then
            rAttack:FireServer(0); rAttack:FireServer(1); rAttack:FireServer(2); rAttack:FireServer(3)
        end
    end)
    pcall(function()
        if rHit then rHit:FireServer(mainPart, othersEnemies) end
    end)
end

-- ── Main Loop (ตรงกับ BF04 ดั้งเดิม) ──────────────────────────────
local LAST_MOB_NAME = ""
local LAST_MOB_POS = nil
local PATROL_I = 0

local function mainLoop()
    log("=== START AUTO V3 Sea 3 ===", "Y")
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ!", "R") return end
    getCombatRemotes()

    while AUTO_ON and _G.BF04V3_LOOP and not getChar() do
        setState("รอตัวละคร spawn...", "S")
        task.wait(1)
    end

    while AUTO_ON and _G.BF04V3_LOOP do
        task.wait(0.25)
        pcall(function()
            local char, hrp = getChar()
            if not char then setState("รอตัวละคร...", "S") return end

            local currentLevel = getMyLevel()
            local qdata = getQuestData(currentLevel)

            local UIQ = pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Quest")
            if not UIQ then setState("หา Quest UI ไม่เจอ", "R") return end

            local titleTxt = ""
            pcall(function() titleTxt = UIQ.Container.QuestTitle.Title.Text end)

            -- 1. เควสไม่ตรงเลเวล → AbandonQuest เพื่อรับเควสใหม่
            if UIQ.Visible and titleTxt ~= "" and not string.find(titleTxt, qdata.mob) then
                setState("เควสไม่ตรง (" .. titleTxt .. ") → AbandonQuest", "Y")
                CommF:InvokeServer("AbandonQuest")
                task.wait(0.5)
                return
            end

            -- 2. ยังไม่มีเควส → บินไปหา NPC ตามเลเวล → รับเควส
            if not UIQ.Visible then
                local dist = (qdata.questCF.Position - hrp.Position).Magnitude
                if dist > 25 then
                    setState("บินไปรับเควส: " .. qdata.mob .. " (" .. math.floor(dist) .. ")", "B")
                    tp(qdata.questCF)
                else
                    setState("รับเควส: " .. qdata.questName .. " [Lv." .. currentLevel .. "]", "Y")
                    CommF:InvokeServer("StartQuest", qdata.questName, qdata.questNum)
                    local waited = 0
                    while not UIQ.Visible and waited < 20 and AUTO_ON and _G.BF04V3_LOOP do
                        task.wait(0.1)
                        waited += 1
                        closeDialogue()
                    end
                    if UIQ.Visible then
                        setState("รับเควสสำเร็จ ✓ → บินไปจุดเกิดมอน", "G")
                        tp(qdata.mobCF)
                        task.wait(0.3)
                    else
                        log("เซิร์ฟเวอร์ยังไม่เปิดรับเควส " .. qdata.questName, "R")
                        task.wait(1)
                    end
                end

            -- 3. มีเควสแล้ว → ล็อกมอนตามเลเวลและตี
            else
                closeDialogue()
                local enemies = Workspace:FindFirstChild("Enemies")
                local target = nil
                local bestDist = math.huge
                if enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        if v:FindFirstChild("HumanoidRootPart") then
                            rememberSpawn(v.Name, v.HumanoidRootPart.Position)
                        end
                        local isMatch = string.find(v.Name, qdata.mob)
                        local isBoss = string.find(v.Name, "Boss") or string.find(v.Name, "King")
                        if not string.find(qdata.mob, "King") and not string.find(qdata.mob, "Boss") and isBoss then
                            isMatch = false
                        end
                        if isMatch and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            local d = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
                            if d < bestDist and d < SEARCH_R then
                                bestDist = d
                                target = v
                            end
                        end
                    end
                end

                if not target and enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        local isBoss = string.find(v.Name, "Boss") or string.find(v.Name, "King")
                        local isMatch = string.find(v.Name, qdata.mob) and not
                            (isBoss and not string.find(qdata.mob, "King") and not string.find(qdata.mob, "Boss"))
                        if isMatch and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            local d = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
                            if d < bestDist then bestDist = d; target = v end
                        end
                    end
                    if target then
                        setState("เจอ " .. qdata.mob .. " ไกล " .. math.floor(bestDist) .. " → บินไปหา", "B")
                        tp(CFrame.new(target.HumanoidRootPart.Position + Vector3.new(0, 20, 0)))
                    end
                end

                if not target then
                    local sameMob = (LAST_MOB_NAME == qdata.mob)
                    local anchor = (sameMob and LAST_MOB_POS) or qdata.mobCF.Position
                    local known = SPAWN_DB[qdata.mob]
                    if not (sameMob and LAST_MOB_POS) and known and #known > 0 then
                        PATROL_I = (PATROL_I or 0) + 1
                        local p = known[(PATROL_I % #known) + 1]
                        setState("ไปยืนจุดเกิดที่จำไว้ " .. qdata.mob, "B")
                        tp(CFrame.new(p + Vector3.new(0, 15, 0)))
                        task.wait(0.5)
                        return
                    end
                    if sameMob and LAST_MOB_POS then
                        setState("มอนตายหมด → รอ respawn", "S")
                        tp(CFrame.new(anchor + Vector3.new(0, 20, 0)))
                    else
                        PATROL_I = (PATROL_I or 0) + 1
                        local ang = PATROL_I * 1.2
                        local r = 60 + (PATROL_I % 6) * 50
                        local off = Vector3.new(math.cos(ang) * r, 15, math.sin(ang) * r)
                        setState("ตระเวนหา " .. qdata.mob, "Y")
                        tp(CFrame.new(anchor + off))
                    end
                    task.wait(0.5)
                    return
                end

                local v = target
                if LAST_MOB_NAME ~= qdata.mob then
                    LAST_MOB_NAME = qdata.mob
                    PATROL_I = 0
                end
                LAST_MOB_POS = v.HumanoidRootPart.Position
                setState("สู้ " .. v.Name .. " (HP " .. math.floor(v.Humanoid.Health) .. ")", "Y")

                equipWeapon()
                local lastCamT, lastMagnetT = 0, 0
                local lockDist = bestDist

                while v.Parent and v.Humanoid.Health > 0 and UIQ.Visible and AUTO_ON and _G.BF04V3_LOOP do
                    local myChar, _, myHum = getChar()
                    if not myChar or (myHum and myHum.Health <= 0) then
                        setState("ตาย → รอเกิดใหม่", "R")
                        LP.CharacterAdded:Wait()
                        task.wait(3)
                        break
                    end

                    local mhrp = v:FindFirstChild("HumanoidRootPart")
                    if mhrp then
                        local hrpNow = myChar:FindFirstChild("HumanoidRootPart")
                        if hrpNow then
                            applyBodyVelocity(hrpNow)
                            local dist = (mhrp.Position - hrpNow.Position).Magnitude
                            if dist > lockDist + 150 then break end
                            if dist > 25 then
                                if not tp(CFrame.new(mhrp.Position + Vector3.new(0, 12, 0))) then break end
                            end
                        end
                        if os.clock() - lastCamT >= 0.2 then
                            lastCamT = os.clock()
                            aimCam(v)
                        end
                        if MAGNET_ON and os.clock() - lastMagnetT >= 0.25 then
                            lastMagnetT = os.clock()
                            bringMobsToTarget(v, qdata.mob)
                        end
                    end

                    directRemoteAttack(v, qdata.mob)
                    task.wait(0.05)
                end
            end
        end)
    end
    stopTween()
    releaseControl()
    log("=== STOP AUTO ===", "R")
end

autoBtn.MouseButton1Click:Connect(function()
    if AUTO_ON then
        AUTO_ON = false
        _G.BF04V3_LOOP = nil
        stopTween()
        autoBtn.Text = "AUTO: OFF"
        autoBtn.BackgroundColor3 = Color3.fromRGB(30, 120, 50)
        releaseControl()

        AUTO_ON = true
        autoBtn.Text = "AUTO: ON"
        autoBtn.BackgroundColor3 = Color3.fromRGB(180, 40, 40)
        _G.BF04V3_LOOP = true
        task.spawn(mainLoop)
    end
end)
