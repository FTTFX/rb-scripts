local function isSea3()

    local map = Workspace:FindFirstChild("Map")
    if not map then return false end
    for _, n in ipairs({ "Port Town", "Hydra Island", "Great Tree", "Castle on the Sea",
        "PortTown", "HydraIsland", "GreatTree", "CakeLoaf", "Tiki Outpost" }) do
        if map:FindFirstChild(n) then return true end
    end
    return false
end

local QUEST_ITEMS = { Relic = true, Cup = true, Torch = true, Key = true }
local function equipWeapon()
    local char, _, hum = getChar()
    if not char or not hum then return false end
    local cur = char:FindFirstChildOfClass("Tool")
    if cur and not QUEST_ITEMS[cur.Name] and cur.ToolTip ~= "Wear" then return true end
    local bp = LP:FindFirstChild("Backpack")
    if not bp then return false end
    for _, tip in ipairs({ "Melee", "Sword", "Blox Fruit", "Gun" }) do
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and t.ToolTip == tip and not QUEST_ITEMS[t.Name] then
                pcall(function() hum:EquipTool(t) end)
                return true
            end
        end
    end
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") and not QUEST_ITEMS[t.Name] then
            pcall(function() hum:EquipTool(t) end)
            return true
        end
    end
    return false
end

local function applyBV(hrp)
    if _G.BF04V3_BV and _G.BF04V3_BV.Parent then return end
    local att = hrp:FindFirstChild("BF04V3_Att")
    if not att then
        att = Instance.new("Attachment"); att.Name = "BF04V3_Att"; att.Parent = hrp
    end
    local lv = Instance.new("LinearVelocity")
    lv.Name = "BF04V3_Noclip"; lv.Attachment0 = att; lv.MaxForce = 1e6
    lv.VectorVelocity = Vector3.new(0, 0, 0)
    lv.RelativeTo = Enum.ActuatorRelativeTo.World; lv.Parent = hrp
    _G.BF04V3_BV = lv
end

local noclipParts = {}
local function setNoclip(on)
    local char = LP.Character
    if not char then return end
    if on then
        noclipParts = {}
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then noclipParts[p] = p.CanCollide; p.CanCollide = false end
        end
    else
        for p, v in pairs(noclipParts) do if p.Parent then p.CanCollide = v end end
        noclipParts = {}
    end
end

local BLOCKING = {
    BodyVelocity = true, BodyGyro = true, BodyPosition = true,
    LinearVelocity = true, AlignPosition = true, AlignOrientation = true,
}
local function releaseControl()
    stopTween()
    pcall(function() if _G.BF04V3_BV then _G.BF04V3_BV:Destroy() _G.BF04V3_BV = nil end end)
    local char = LP.Character
    if not char then return end
    pcall(function()
        for _, obj in ipairs(char:GetDescendants()) do
            if BLOCKING[obj.ClassName] then obj:Destroy() end
        end
    end)
    pcall(function()
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then p.Anchored = false end
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false; hum.Sit = false end
    end)
    setNoclip(false)
    pcall(function()
        require(LP.PlayerScripts:WaitForChild("PlayerModule")):GetControls():Enable()
    end)
end

local function flyTo(destCF, speed)
    stopTween()
    local char, hrp = getChar()
    if not char or not hrp then return false end
    applyBV(hrp); setNoclip(true)
    local dist = (destCF.Position - hrp.Position).Magnitude
    if dist < 12 then return true end
    local spd = speed or FLY_SPEED
    if dist >= 1000 then spd = math.min(spd, 240)
    elseif dist >= 500 then spd = math.min(spd, 280) end
    local tween = TweenService:Create(hrp, TweenInfo.new(dist / spd, Enum.EasingStyle.Linear), { CFrame = destCF })
    currentTween = tween; tween:Play(); tween.Completed:Wait(); currentTween = nil
    local _, h2 = getChar()
    if not h2 then return false end
    return (destCF.Position - h2.Position).Magnitude <= 30
end

local function boostSim()
    pcall(function()
        if setsimulationradius then setsimulationradius(math.huge, math.huge) end
        if sethiddenproperty then
            sethiddenproperty(LP, "MaximumSimulationRadius", math.huge)
            sethiddenproperty(LP, "SimulationRadius", math.huge)
        end
    end)
end

    if not myHrp then return end
    boostSim()
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return end
    for _, e in ipairs(enemies:GetChildren()) do
        if e ~= primary then
            local hum = e:FindFirstChild("Humanoid")
            local ehrp = e:FindFirstChild("HumanoidRootPart")
            if hum and ehrp and hum.Health > 0 and string.find(e.Name, mobName)
               and not string.find(e.Name, "Boss") and not string.find(e.Name, "Leader") then
                pcall(function()
                    hum.PlatformStand = true; ehrp.CanCollide = false
                    ehrp.CFrame = myHrp.CFrame * CFrame.new(math.random(-6, 6), 0, math.random(-6, 6))
                end)
            end
        end
    end
end

local function attack(mobTarget, mobName, allowAoE)
    if not mobTarget or not mobTarget.Parent then return end
    local rA, rH = getCombatRemotes()
    if not (LP.Character and LP.Character:FindFirstChildOfClass("Tool")) then return end
    local mainPart = mobTarget:FindFirstChild("Head") or mobTarget:FindFirstChild("HumanoidRootPart")
    if not mainPart then return end
    local others = { { mobTarget, mainPart } }
    if allowAoE and MAGNET_ON then
        local enemies = Workspace:FindFirstChild("Enemies")
        if enemies then
            for _, e in ipairs(enemies:GetChildren()) do
                if #others >= 8 then break end
                if e ~= mobTarget then
                    local hum = e:FindFirstChild("Humanoid")
                    local part = e:FindFirstChild("Head") or e:FindFirstChild("HumanoidRootPart")
                    if hum and part and hum.Health > 0 and string.find(e.Name, mobName)
                       and (part.Position - mainPart.Position).Magnitude < 100 then
                        others[#others + 1] = { e, part }
                    end
                end
            end
        end
    end
    pcall(function()
        if rA then rA:FireServer(0); rA:FireServer(1); rA:FireServer(2); rA:FireServer(3) end
    end)
    pcall(function() if rH then rH:FireServer(mainPart, others) end end)
end

local function findAlive(name, folder)
    folder = folder or Workspace:FindFirstChild("Enemies")
    if not folder then return nil end
    for _, e in ipairs(folder:GetChildren()) do
        if string.find(e.Name, name) then
            local hum = e:FindFirstChild("Humanoid")
            local ehrp = e:FindFirstChild("HumanoidRootPart")
            if hum and ehrp and hum.Health > 0 then return e end
        end
    end
end

local function findMobNear(mobName, nearPos)
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return nil end
    local best, bestD = nil, math.huge
    local origin = nearPos
    if not origin then
        local _, hrp = getChar(); origin = hrp and hrp.Position
    end
    if not origin then return nil end
    for _, e in ipairs(enemies:GetChildren()) do
        if string.find(e.Name, mobName) then
            local hum = e:FindFirstChild("Humanoid")
            local ehrp = e:FindFirstChild("HumanoidRootPart")
            if hum and ehrp and hum.Health > 0 then
                local d = (ehrp.Position - origin).Magnitude
                if d < bestD then bestD = d; best = e end
            end
        end
    end
    return best
end

local function getQuestUI()
    local main = pg:FindFirstChild("Main")
    return main and main:FindFirstChild("Quest")
end

local function questTitle()
    local UIQ = getQuestUI()
    local t = ""
    pcall(function() t = UIQ.Container.QuestTitle.Title.Text end)
    return t, UIQ
end

local function closeDialogue()
    pcall(function()
        local main = pg:FindFirstChild("Main")
        if not main then return end
        for _, name in ipairs({ "Dialogue", "NPC", "Choices" }) do
            local d = main:FindFirstChild(name)
            if d and d:IsA("GuiObject") and d.Visible then d.Visible = false end
        end
    end)
end

-- หาบอสเกาะที่เกิดอยู่ (Enemies หรือ ReplicatedStorage)
local function findSpawnedBoss()
    if not BOSS_ON then return nil end
    for _, b in ipairs(SEA3_BOSSES) do
        local live = findAlive(b.name, Workspace:FindFirstChild("Enemies"))
        if live then return b, live end
        local rs = findAlive(b.name, RS)
        if rs then return b, rs end
    end
end

local function fightTarget(mob, label, useMagnet)
    local mhrp = mob:FindFirstChild("HumanoidRootPart")
    local mhum = mob:FindFirstChild("Humanoid")
    if not mhrp or not mhum or mhum.Health <= 0 then return false end
    local _, hrp = getChar()
    if not hrp then return false end
    equipWeapon(); ensureBuso()
    local hover = mhrp.CFrame * CFrame.new(0, HOVER_Y, 0)
    local dist = (hrp.Position - mhrp.Position).Magnitude
    if dist > 40 then
        flyTo(hover, FLY_SPEED)
    else
        pcall(function() applyBV(hrp); hrp.CFrame = hover end)
    end
    if useMagnet then bringMobs(mob, label) end
    attack(mob, label, useMagnet)
    return true
end

-- ── Main loop ─────────────────────────────────────────────────────
local function mainLoop()
    log("=== V3 Sea3 Quest+Boss START ===")
    if not isSea3() then
        log("ไม่ใช่ทะเล 3 — หยุด"); setState("ต้องอยู่ทะเล 3")
        AUTO_ON = false; autoBtn.Text = "AUTO: OFF"
        autoBtn.BackgroundColor3 = Color3.fromRGB(30, 120, 50)
        return
    end
    local lvl = getMyLevel()
    if lvl < 1500 then
        log("Lv." .. lvl .. " < 1500"); setState("Lv ต่ำไป")
        AUTO_ON = false; autoBtn.Text = "AUTO: OFF"; return
    end

    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ"); setState("ไม่มี CommF"); return end
    getCombatRemotes()
    log("Lv." .. lvl .. " | เควสมอน + บอสเกาะ (ไม่ทำเควสนอก)")

    while AUTO_ON and _G.BF04V3_LOOP do
        local char, hrp, hum = getChar()
        if not char or (hum and hum.Health <= 0) then
            setState("รอเกิดใหม่..."); log("ตาย → รอเกิด")
            LP.CharacterAdded:Wait(); task.wait(2)
        else
            -- 1) บอสเกาะเกิด → รับเควสบอสแล้วตี (priority)
            local bossData, bossMob = findSpawnedBoss()
            if bossData then
                local title, UIQ = questTitle()
                if UIQ and UIQ.Visible and not string.find(title, bossData.name) then
                    setState("ทิ้งเควสเก่า → รับบอส " .. bossData.name)
                    pcall(function() CommF:InvokeServer("AbandonQuest") end)
                    task.wait(0.4)
                end
                title, UIQ = questTitle()
                if not (UIQ and UIQ.Visible and string.find(title, bossData.name)) then
                    setState("บินรับเควสบอส " .. bossData.name)
                    local d = (bossData.questCF.Position - hrp.Position).Magnitude
                    if d > 15 then
                        flyTo(bossData.questCF, FLY_SPEED)
                    else
                        log("StartQuest " .. bossData.questName .. ",3 (" .. bossData.name .. ")")
                        pcall(function()
                            CommF:InvokeServer("StartQuest", bossData.questName, bossData.questNum)
                        end)
                        task.wait(0.5); closeDialogue()
                    end
                else
                    -- บอสอาจอยู่ใน RS ยังไม่ stream → บินไปจุดบอสก่อน
                    if not bossMob.Parent or bossMob.Parent == RS then
                        setState("รอสปอว์นบอส " .. bossData.name)
                        flyTo(bossData.bossCF * CFrame.new(0, HOVER_Y, 0), FLY_SPEED)
                        task.wait(0.3)
                        bossMob = findAlive(bossData.name, Workspace:FindFirstChild("Enemies"))
                            or findAlive(bossData.name, RS)
                    end
                    if bossMob and bossMob.Parent == Workspace:FindFirstChild("Enemies") then
                        setState("ตีบอส " .. bossData.name)
                        fightTarget(bossMob, bossData.name, false)
                    end
                end

            else
                -- 2) ฟาร์มเควสมอนตามเลเวล
                local q = getQuestData(getMyLevel())
                local title, UIQ = questTitle()
                if not UIQ then setState("หา Quest UI ไม่เจอ"); task.wait(1)
                else
                    -- เควสค้างที่ไม่ใช่มอน/บอสของเรา → ทิ้ง (กันเควสนอก)
                    if UIQ.Visible and title ~= ""
                       and not string.find(title, q.mob)
                       and not string.find(string.lower(title), "boss") then
                        local isOurBoss = false
                        for _, b in ipairs(SEA3_BOSSES) do
                            if string.find(title, b.name) then isOurBoss = true; break end
                        end
                        if not isOurBoss then
                            setState("ทิ้งเควสนอก: " .. title)
                            pcall(function() CommF:InvokeServer("AbandonQuest") end)
                            task.wait(0.4)
                        end
                    end

                    title, UIQ = questTitle()
                    if not UIQ.Visible then
                        local d = (q.questCF.Position - hrp.Position).Magnitude
                        if d > 15 then
                            setState("บินรับเควส " .. q.mob)
                            flyTo(q.questCF, FLY_SPEED)
                        else
                            setState("StartQuest " .. q.questName .. "," .. q.questNum)
                            pcall(function()
                                CommF:InvokeServer("StartQuest", q.questName, q.questNum)
                            end)
                            local waited = 0
                            while not UIQ.Visible and waited < 15 and AUTO_ON do
                                task.wait(0.1); waited += 1; closeDialogue()
                            end
                            if UIQ.Visible then log("รับเควส ✓ " .. q.mob)
                            else log("รับเควสไม่ติด " .. q.questName); task.wait(0.8) end
                        end
                    else
                        closeDialogue()
                        setState("หา " .. q.mob)
                        local mob = findMobNear(q.mob)
                        if not mob then
                            flyTo(q.mobCF * CFrame.new(0, HOVER_Y, 0), FLY_SPEED)
                            task.wait(0.35)
                            mob = findMobNear(q.mob, q.mobCF.Position)
                        end
                        if mob then
                            local mhum = mob:FindFirstChild("Humanoid")
                            setState("ตี " .. q.mob .. " HP " .. (mhum and math.floor(mhum.Health) or "?"))
                            fightTarget(mob, q.mob, true)
                        else
                            setState("รอสปอว์น " .. q.mob)
                            task.wait(0.4)
                        end
                    end
                end
            end
        end
        RunService.Heartbeat:Wait()
    end

    releaseControl()
    log("=== STOP ==="); setState("หยุดแล้ว")
end

        AUTO_ON = false; _G.BF04V3_LOOP = nil; stopTween()
        autoBtn.Text = "AUTO: OFF"; autoBtn.BackgroundColor3 = Color3.fromRGB(30, 120, 50)
        releaseControl(); log("ปิด AUTO")

        _G.BF04V3_LOOP = true
        task.spawn(mainLoop)
    end
end)
