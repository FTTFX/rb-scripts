local function findLiveMobCF(mobName)
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return nil end
    for _, e in ipairs(enemies:GetChildren()) do
        if e.Name:find(mobName, 1, true) then
            local hrp = e:FindFirstChild("HumanoidRootPart")
            local hum = e:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                return hrp.CFrame
            end
        end
    end
end

local function findLiveNpcCF(mobName, qName)
    local folder = Workspace:FindFirstChild("NPCs")
    if not folder then return nil end
    local keys = { "Quest Giver" }
    if mobName then
        local w = {}
        for part in mobName:gmatch("%S+") do w[#w+1] = part end
        if #w >= 2 then keys[#keys+1] = w[1] .. " " .. w[2] end
        keys[#keys+1] = mobName
    end
    if qName then keys[#keys+1] = qName end
    for _, npc in ipairs(folder:GetChildren()) do
        local nl = npc.Name:lower()
        for _, k in ipairs(keys) do
            if k ~= "" and nl:find(k:lower(), 1, true) then
                local p = npc:FindFirstChild("HumanoidRootPart")
                    or npc:FindFirstChildWhichIsA("BasePart")
                if p then return p.CFrame end
            end
        end
    end
end

-- เลือกเควส LevelReq สูงสุดที่ <= level (ข้ามบอส Task=1 ถ้ามีตัวเลือกฟาร์ม)
local function getQuestData(level)
    level = level or 1500
    local data = loadQuests()
    local bestFarm, bestBoss = nil, nil
    if data then
        for qName, entry in pairs(data) do
            if type(entry) == "table" then
                for num, sub in pairs(entry) do
                    if type(sub) == "table" and type(sub.LevelReq) == "number"
                       and sub.LevelReq <= level and type(sub.Task) == "table" then
                        local mob, need = next(sub.Task)
                        if mob and type(need) == "number" then
                            local cand = {
                                mob = mob,
                                qName = qName,
                                qNum = tonumber(num) or num,
                                need = need,
                                levelReq = sub.LevelReq,
                            }
                            if need <= 1 then
                                if not bestBoss or cand.levelReq > bestBoss.levelReq then
                                    bestBoss = cand
                                end
                            else
                                if not bestFarm or cand.levelReq > bestFarm.levelReq
                                   or (cand.levelReq == bestFarm.levelReq and cand.qNum > bestFarm.qNum) then
                                    bestFarm = cand
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    local pick = bestFarm or bestBoss
    if not pick then
        -- fallback ถ้า require ไม่ได้
        return { mob = "Dragon Crew Warrior", qName = "DragonCrewQuest", qNum = 1,
            npc = KNOWN["Dragon Crew Warrior"].npc, mobCF = KNOWN["Dragon Crew Warrior"].mobCF }
    end

    local known = KNOWN[pick.mob]
    local npcCF = findLiveNpcCF(pick.mob, pick.qName)
        or (known and known.npc)
        or CFrame.new()
    local mobCF = findLiveMobCF(pick.mob)
        or (known and known.mobCF)
        or npcCF

    return {
        mob = pick.mob,
        qName = pick.qName,
        qNum = pick.qNum,
        npc = npcCF,
        mobCF = mobCF,
        need = pick.need,
        levelReq = pick.levelReq,
    }
end

end

-- Noclip: ปิด CanCollide เท่านั้น (ไม่ ChangeState ทุกเฟรม — กันเด้งตำแหน่ง)
local noclipParts, noclipChar, noclipRefreshed = {}, nil, 0
local function setNoclip(on)
    local char = LP.Character
    if not char then return end
    local now = os.clock()
    if char ~= noclipChar or now - noclipRefreshed > 2 then
        noclipChar = char
        noclipRefreshed = now
        table.clear(noclipParts)
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then table.insert(noclipParts, p) end
        end
    end
    for i = #noclipParts, 1, -1 do
        local p = noclipParts[i]
        if p.Parent then
            if on then p.CanCollide = false end
        else
            table.remove(noclipParts, i)
        end
    end
end

    while true do
        task.wait(0.2)
        if _G.BF04V3_ON then pcall(setNoclip, true) end
    end
end)

-- ── บินแบบ BF04: Tween จบรอบเดียว + Bypass TP ระยะไกล ───────────
local currentTween = nil
local function stopTween()
    if currentTween then
        pcall(function() currentTween:Cancel() end)
        currentTween = nil
    end
end

local function flySmooth(destCF, speed)
    stopTween()
    local _, hrp = getChar()
    if not hrp then return false end
    setNoclip(true)
    local dist = (destCF.Position - hrp.Position).Magnitude
    if dist < 12 then
        hrp.CFrame = destCF
        return true
    end
    local spd = speed or 200
    if dist >= 1000 then spd = math.min(spd, 180)
    elseif dist >= 500 then spd = math.min(spd, 220)
    end
    local tween = TweenService:Create(hrp,
        TweenInfo.new(dist / spd, Enum.EasingStyle.Linear),
        { CFrame = destCF })
    currentTween = tween
    tween:Play()
    tween.Completed:Wait()
    currentTween = nil
    local _, h2 = getChar()
    if not h2 then return false end
    local left = (destCF.Position - h2.Position).Magnitude
    if left > 30 then
        log(string.format("บินไม่ถึง (เหลือ %.0f) → รอบใหม่", left), "Y")
        return false
    end
    return true
end

local function bypassTP(targetCF)
    local char, hrp, hum = getChar()
    if not char or not hrp or not hum then return false end
    local cf = getCommF()
    if not cf then return false end
    local startDist = (targetCF.Position - hrp.Position).Magnitude
    log(string.format("Bypass TP %.0f studs...", startDist), "B")
    local deadline = os.clock() + 12
    repeat
        pcall(function()
            hum:ChangeState(15)
            hrp.CFrame = targetCF
            task.wait()
            hrp.CFrame = targetCF
        end)
        task.wait(0.4)
        local c2, h2, hum2 = getChar()
        if not h2 then return false end
        hrp, hum = h2, hum2
    until (targetCF.Position - hrp.Position).Magnitude <= 1500 or os.clock() > deadline
    if (targetCF.Position - hrp.Position).Magnitude > 1500 then return false end
    pcall(function() cf:InvokeServer("SetSpawnPoint") end)
    task.wait(0.25)
    pcall(function() cf:InvokeServer("SetSpawnPoint") end)
    task.wait(0.25)
    pcall(function() cf:InvokeServer("TeleportToSpawn") end)
    task.wait(1.2)
    return true
end

local function tp(destCF)
    local _, hrp = getChar()
    if not hrp then return false end
    local dist = (destCF.Position - hrp.Position).Magnitude
    if dist < 25 then
        stopTween()
        hrp.CFrame = destCF
        return true
    end
    if dist > 2500 then
        stopTween()
        if bypassTP(destCF) then
            _, hrp = getChar()
            if hrp and (destCF.Position - hrp.Position).Magnitude <= 80 then
                return true
            end
        end
    end
    return flySmooth(destCF, 200)
end

local function gotoAndWait(cf)
    local t0 = os.clock()
    while _G.BF04V3_ON and os.clock() - t0 < 180 do
        local _, hrp = getChar()
        if not hrp then return false end
        local d = (hrp.Position - cf.Position).Magnitude
        if d <= 25 then
            hrp.CFrame = cf
            return true
        end
        if d > 80 then
            log(string.format("บิน %.0f studs...", d), "B")
        end
        tp(cf)
        task.wait(0.2)
        _, hrp = getChar()
        if hrp and (hrp.Position - cf.Position).Magnitude <= 25 then
            return true
        end
    end
    return false
end

-- ── Weapon ─────────────────────────────────────────────────────────
local QUEST_ITEMS = {
    Relic = true, Cup = true, Torch = true, Key = true,
    ["Special Key"] = true, ["Prison Key"] = true, ["Water Key"] = true,
    ["Library Key"] = true, ["God's Chalice"] = true, ["Fist of Darkness"] = true,
    ["Holy Torch"] = true, ["Hallow Essence"] = true,
}
local function equipWeapon()
    local char, _, hum = getChar()
    if not char then return false end
    local held = char:FindFirstChildOfClass("Tool")
    if held and not QUEST_ITEMS[held.Name] then return true end
    local bp = LP:FindFirstChild("Backpack")
    if not bp then return false end
    -- ลำดับ Melee > Sword > อื่นๆ (ตัดผลปีศาจออก กันมั่ว)
    local best
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") and not QUEST_ITEMS[t.Name] then
            local tip = tostring(t.ToolTip or "")
            if tip == "Melee" then best = t break
            elseif tip == "Sword" then best = best or t
            elseif not best and tip ~= "Blox Fruit" then best = t end
        end
    end
    if best then
        pcall(function() hum:EquipTool(best) end)
        return true
    end
    return false
end

-- ── Combat remotes (fallback หลักเมื่อ CombatFramework require ไม่ได้) ─
local RegisterAttack, RegisterHit
local function getCombatRemotes()
    if RegisterAttack and RegisterHit then return RegisterAttack, RegisterHit end
    pcall(function()
        local net = RS:FindFirstChild("Modules") and RS.Modules:FindFirstChild("Net")
        if net then
            RegisterAttack = net:FindFirstChild("RE/RegisterAttack")
            RegisterHit = net:FindFirstChild("RE/RegisterHit")
        end
    end)
    return RegisterAttack, RegisterHit
end

local function fireRemoteHit(mobTarget, mobName)
    if not mobTarget or not mobTarget.Parent then return end
    local rAttack, rHit = getCombatRemotes()
    local char = LP.Character
    if not char or not char:FindFirstChildOfClass("Tool") then return end
    local mainPart = mobTarget:FindFirstChild("Head")
        or mobTarget:FindFirstChild("HumanoidRootPart")
    if not mainPart then return end
    local pack = { { mobTarget, mainPart } }
    if _G.BF04V3_MAGNET then
        local enemies = Workspace:FindFirstChild("Enemies")
        if enemies then
            for _, e in ipairs(enemies:GetChildren()) do
                if #pack >= 8 then break end
                if e ~= mobTarget and e.Name:find(mobName or "", 1, true)
                   and not e.Name:find("Boss") then
                    local eh = e:FindFirstChildOfClass("Humanoid")
                    local ep = e:FindFirstChild("Head") or e:FindFirstChild("HumanoidRootPart")
                    if eh and eh.Health > 0 and ep
                       and (ep.Position - mainPart.Position).Magnitude < 100 then
                        pack[#pack + 1] = { e, ep }
                    end
                end
            end
        end
    end
    pcall(function()
        if rAttack then
            rAttack:FireServer(0); rAttack:FireServer(1)
            rAttack:FireServer(2); rAttack:FireServer(3)
        end
        if rHit then rHit:FireServer(mainPart, pack) end
    end)
end

-- หา activeController: require → getupvalues → getgc (ไม่บังคับ)
local activeCtrl = nil
local function resolveController()
    if activeCtrl then return activeCtrl end
    -- 1) require ตรง
    pcall(function()
        local mod = LP.PlayerScripts:FindFirstChild("CombatFramework")
        if not mod then return end
        local CF = require(mod)
        if CF and CF.activeController then
            activeCtrl = CF.activeController
            return
        end
        if debug and debug.getupvalues then
            local ups = debug.getupvalues(CF)
            if ups and ups[2] and ups[2].activeController then
                activeCtrl = ups[2].activeController
            end
        end
    end)
    if activeCtrl then return activeCtrl end
    -- 2) ไล่ getgc หา table ที่มี hitboxMagnitude
    pcall(function()
        if not getgc then return end
        for _, v in ipairs(getgc(true)) do
            if type(v) == "table" and rawget(v, "hitboxMagnitude")
               and rawget(v, "timeToNextAttack") and rawget(v, "attacking") ~= nil then
                activeCtrl = v
                break
            end
        end
    end)
    return activeCtrl
end

-- ── Magnet (สไตล์ Xenon: isnetworkowner + BodyVelocity) ───────────
local function bringMobs(primary, mobName, myHrp)
    if not _G.BF04V3_MAGNET then return end
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return end
    pcall(function()
        if setsimulationradius then setsimulationradius(1e9, 1e9) end
    end)
    for _, v in ipairs(enemies:GetChildren()) do
        if v ~= primary and v.Name:find(mobName) then
            local hum = v:FindFirstChildOfClass("Humanoid")
            local hrp = v:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and hrp and not v.Name:find("Boss") then
                local owner = true
                pcall(function() owner = isnetworkowner(hrp) end)
                if owner and (hrp.Position - myHrp.Position).Magnitude < 370 then
                    hrp.CanCollide = false
                    hrp.Velocity = Vector3.new(0, 0, 0)
                    local bv = hrp:FindFirstChild("BF04_BV")
                    if not bv then
                        bv = Instance.new("BodyVelocity")
                        bv.Name = "BF04_BV"
                        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                        bv.P = 1e4
                        bv.Parent = hrp
                    end
                    bv.Velocity = Vector3.new(0, 0, 0)
                    hum.JumpPower = 0
                    hum.WalkSpeed = 0
                    hum:ChangeState(14)
                    hrp.CFrame = myHrp.CFrame * CFrame.new(math.random(-5, 5), 0, math.random(-5, 5))
                end
            end
        end
    end
end

local function mainLoop()
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ!", "R") return end

    while _G.BF04V3_ON do
        task.wait(0.2)
        local ok, err = pcall(function()
            local char, hrp = getChar()
            if not char then return end

            local qdata = getQuestData(getMyLevel())
            local UIQ = pg.Main:FindFirstChild("Quest")
            if not UIQ then return end

            -- ═══ ยังไม่มีเควส → บินไป NPC (ใกล้แล้วค่อย StartQuest) ═══
            if not UIQ.Visible then
                FASTATTACK_ON = false
                local _, hrpNow = getChar()

                local distNpc = (hrpNow.Position - qdata.npc.Position).Magnitude

                if distNpc > 20 then
                    setState(string.format("บินไป NPC %s (%.0f studs) Lv.%d", qdata.mob, distNpc, lv), "B")
                    tp(qdata.npc)
                    return -- รอบหน้าค่อยเช็คระยะอีกครั้ง (แบบ BF04)
                end

                -- ใกล้ NPC จริงแล้ว → ยืนนิ่ง + รับเควส
                stopTween()
                hrpNow.CFrame = qdata.npc
                task.wait(0.35)
                -- เช็คอีกครั้งว่ายังอยู่ใกล้ (กันเด้ง anti-cheat)
                local _, hCheck = getChar()
                if not hCheck or (hCheck.Position - qdata.npc.Position).Magnitude > 35 then
                    log("ตำแหน่งเด้งกลับ → บินใหม่", "Y")
                    task.wait(0.5)
                    return
                end

                setState(string.format("StartQuest %s #%d (ระยะ %.0f)", qdata.qName, qdata.qNum,
                    (hCheck.Position - qdata.npc.Position).Magnitude), "Y")
                local invokeOk, invokeResult = pcall(function()
                    return CommF:InvokeServer("StartQuest", qdata.qName, qdata.qNum)
                end)
                if not invokeOk then
                    log("StartQuest error: " .. tostring(invokeResult), "R")
                end
                pcall(function() CommF:InvokeServer("SetSpawnPoint") end)

                local waited = 0
                while not UIQ.Visible and waited < 25 and _G.BF04V3_ON do
                    task.wait(0.1)
                    waited += 1
                    closeDialogue()
                end

                if UIQ.Visible then
                    log("รับเควสสำเร็จ ✓ → ไปจุดมอน", "G")
                    tp(qdata.mobCF)
                else
                    local _, hFail = getChar()
                    local dFail = hFail and (hFail.Position - qdata.npc.Position).Magnitude or -1
                    log(string.format("ไม่เปิดเควส | Lv.%d | ระยะ %.0f | ตอบ: %s",
                        lv, dFail, tostring(invokeResult)), "R")
                    task.wait(1.5)
                end
                return
            end

            -- ═══ มีเควส → เช็คชื่อเควสตรงไหม (กัน Abandon วนลูป) ═══
            local titleTxt = ""
            pcall(function() titleTxt = UIQ.Container.QuestTitle.Title.Text end)
            if titleTxt ~= "" and not titleTxt:find(qdata.mob, 1, true) then
                setState("เควสไม่ตรง (" .. titleTxt .. ") → เปลี่ยน", "Y")
                CommF:InvokeServer("AbandonQuest")
                task.wait(0.5)
                return
            end

            -- ═══ หามอนตัวใกล้สุด (ไม่เอาตัวที่แบนไว้) ═══
            local enemies = Workspace:FindFirstChild("Enemies")
            if not enemies then return end
            local now = os.clock()
            for k in pairs(IGNORE_MOBS) do
                if not k.Parent or now >= IGNORE_MOBS[k] then IGNORE_MOBS[k] = nil end
            end

            local target, bestD = nil, math.huge
            for _, v in ipairs(enemies:GetChildren()) do
                if v.Name:find(qdata.mob, 1, true) and not IGNORE_MOBS[v] then
                    local mhum = v:FindFirstChildOfClass("Humanoid")
                    local mhrp = v:FindFirstChild("HumanoidRootPart")
                    if mhum and mhrp and mhum.Health > 0 then
                        local d = (mhrp.Position - hrp.Position).Magnitude
                        if d < bestD then bestD = d; target = v end
                    end
                end
            end

            -- ═══ ไม่เจอมอน → บินไปจุดเกิดมอน ═══
            if not target then
                setState("ไม่เจอ " .. qdata.mob .. " → บินไปจุดเกิด", "B")
                gotoAndWait(qdata.mobCF)
                return
            end

            -- ═══ สู้: ล็อกเหนือหัว + ขยาย hitbox + FastAttack ═══
            setState("สู้ " .. target.Name .. " (HP " .. math.floor(target.Humanoid.Health) .. ")", "Y")
            equipWeapon()
            -- Buso กัน elemental immunity
            pcall(function()
                if not LP.Character:FindFirstChild("HasBuso") then
                    CommF:InvokeServer("Buso")
                end
            end)
            CURRENT_TARGET, CURRENT_MOBNAME = target, qdata.mob
            FASTATTACK_ON = true
            local lastHp = target.Humanoid.Health
            local lastCheck = os.clock()
            local failCount = 0

            while target.Parent and target:FindFirstChild("Humanoid")
                  and target.Humanoid.Health > 0 and _G.BF04V3_ON do
                local myChar, myHrp = getChar()
                if not myChar then break end
                equipWeapon()

                local mhrp = target:FindFirstChild("HumanoidRootPart")
                if not mhrp then break end

                -- ลอยเหนือหัว 12 studs (hitbox 50 ครอบถึง + remote ระยะพอ)
                myHrp.CFrame = mhrp.CFrame * CFrame.new(0, 12, 0)

                pcall(function()
                    target.Humanoid.PlatformStand = true
                    target.Humanoid.Sit = true
                    mhrp.CanCollide = false
                    mhrp.Size = Vector3.new(50, 50, 50)
                end)

                bringMobs(target, qdata.mob, myHrp)
                fireRemoteHit(target, qdata.mob)

                if os.clock() - lastCheck >= 1.5 then
                    lastCheck = os.clock()
                    local hp = target.Humanoid.Health
                    if hp >= lastHp then
                        failCount += 1
                        log(string.format("[ตีไม่เข้า #%d] %s HP นิ่ง %.0f", failCount, target.Name, hp), "R")
                        if failCount >= 3 then
                            log("ข้ามตัวนี้ → หาตัวอื่น (แบน 45 วิ)", "Y")
                            IGNORE_MOBS[target] = os.clock() + 45
                            break
                        end
                    else
                        failCount = 0
                        lastHp = hp
                    end
                end
                task.wait()
            end

            FASTATTACK_ON = false
            CURRENT_TARGET, CURRENT_MOBNAME = nil, nil
            if target.Parent == nil or (target:FindFirstChildOfClass("Humanoid")
               and target.Humanoid.Health <= 0) then
                log(target.Name .. " ตายแล้ว ✓", "G")
            end
            task.wait(0.3)
        end)
        if not ok then
            log("Loop error: " .. tostring(err), "R")
            task.wait(1)
        end
    end
end
