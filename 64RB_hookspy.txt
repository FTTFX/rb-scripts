-- ==================== State (hook ครั้งเดียวตลอด session) ====================

if S.filters == nil then S.filters = {Loader=true, Input=true, Remote=true, Raycast=false} end
local MAXQ = 150

-- push + dedup ใน hook (sig ซ้ำ → count++ ไม่เพิ่ม entry)
local function push(cat, sig, text)
    if not S.enabled then return end
    if not S.filters[cat] then return end
    local q = S.queue
    local last = q[#q]
    if last and last.sig == sig then last.count = (last.count or 1) + 1 return end
    q[#q+1] = {cat=cat, sig=sig, text=text, count=1}
    if #q > MAXQ then table.remove(q, 1) end
end
S.push = push

end

-- ==================== Value formatter (เบา) ====================
local function fmt(v, full)
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then local p=v.Position return string.format("CF(%.0f,%.0f,%.0f)", p.X,p.Y,p.Z)
    elseif t == "string" then
        local mx = full and 160 or 24
        if #v > mx then return '"'..v:sub(1,mx)..'..."' end
        return '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    elseif t == "boolean" then return tostring(v)
    else return t end
end
local function fmtArgs(...)
    local n = select("#", ...)
    if n == 0 then return "(ไม่มี args)" end
    local p = {}
    for i = 1, math.min(n, 8) do p[i] = fmt(select(i, ...), true) end
    return table.concat(p, " | ")
end

-- ==================== Name cache (กัน GetFullName ทุก call) ====================
local nameCache = setmetatable({}, {__mode = "k"})
local function getName(r)
    local n = nameCache[r]
    if n == nil then
        local ok, full = pcall(function() return r:GetFullName() end)
        n = ok and full or "?"
        nameCache[r] = n
    end
    return n
end

-- ==================== Method → category (lookup เร็ว) ====================
local CAT = {
    FireServer="Remote", InvokeServer="Remote",
    HttpGet="Loader", HttpGetAsync="Loader",
    SendTouchEvent="Input", SendMouseButtonEvent="Input",
    SendMouseMoveEvent="Input", SendKeyEvent="Input",
    Raycast="Raycast", FindPartOnRay="Raycast",
    FindPartOnRayWithWhitelist="Raycast", FindPartOnRayWithIgnoreList="Raycast", Blockcast="Raycast",
}

-- ==================== dump loadstring (code หลัง key) ====================
local function dumpSource(src)
    S.dumpN = (S.dumpN or 0) + 1
    local n = S.dumpN
    push("Loader", "ls"..n, "★ loadstring #"..n.." ("..#src.." chars) = CODE หลัก!")
    pcall(function() if writefile then writefile("HK_dump_"..n..".lua", src)
        push("Loader", "lf"..n, "   → เซฟ: HK_dump_"..n..".lua") end end)
    pcall(function() if setclipboard then setclipboard(src)
        push("Loader", "lc"..n, "   → copy clipboard") end end)
end

-- ==================== HOOKS (ครั้งเดียว) ====================
if okHooks and not S.hooked then
    S.hooked = true

    -- loadstring (Loader) — global function
    if loadstring then
        local oldLoad
        oldLoad = hookfunc(loadstring, newcc(function(src, ...)
            if type(src) == "string" and #src > 50 then pcall(dumpSource, src) end
            return oldLoad(src, ...)
        end))
        S.restoreLoad = oldLoad
    end

    -- __namecall (LEAN) — getnc ไม่ pcall + early-return
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local m = getnc()
        local cat = CAT[m]
        if cat and S.filters[cat] and S.enabled then
            if cat == "Raycast" then
                push("Raycast", "Ry"..m, m.." (ยิงหาเป้า)")        -- เบาสุด ไม่ format
            elseif cat == "Loader" then
                local url = tostring((select(1, ...)))
                push("Loader", "hg"..url, m.." → "..url)
            elseif cat == "Input" then
                push("Input", "In"..m, m.."("..fmtArgs(...)..")")
            else  -- Remote
                local nm = getName(self)
                if not (nm:find("UpdateState") or nm:find("CameraRotation") or nm:find("Spectating")) then
                    local short = nm:gsub("^.-Remotes%.", "")
                    push("Remote", "Rm"..m..nm, m.." @"..short.." | "..fmtArgs(...))
                end
            end
        end
        return oldNc(self, ...)
    end))
    S.restoreNc = oldNc
end

local renderOrder, renderCount, MAXSHOW = 0, 0, 120
local copyLines = {}
local lastSig, lastLabel, lastCount = nil, nil, 0
local function addLine(cat, text, count, sig)
    count = count or 1
    sig = sig or (cat..text)
    if cat ~= "Loader" and sig == lastSig and lastLabel and lastLabel.Parent then
        lastCount += count
        lastLabel.Text = "["..cat:sub(1,3).."] "..text.."   ×"..lastCount
        copyLines[#copyLines] = lastLabel.Text
        return
    end
    renderOrder += 1; renderCount += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.AutomaticSize = Enum.AutomaticSize.Y
    l.BackgroundTransparency = 1; l.Font = Enum.Font.Code; l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = CAT_COL[cat] or Color3.fromRGB(200,200,200)
    l.Text = "["..cat:sub(1,3).."] "..text..(count > 1 and ("   ×"..count) or "")
    l.LayoutOrder = renderOrder; l.Parent = scroll
    lastSig, lastLabel, lastCount = sig, l, count
    copyLines[#copyLines+1] = l.Text
    if renderCount > MAXSHOW then
        for _, c in ipairs(scroll:GetChildren()) do
            if c:IsA("TextLabel") then c:Destroy() renderCount -= 1 break end
        end
    end
end
