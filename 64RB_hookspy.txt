-- ==================== Persistent state (hook ครั้งเดียวตลอด session) ====================
_G.HK_SPY = _G.HK_SPY or {}
local S = _G.HK_SPY
S.enabled = true
S.queue   = S.queue or {}          -- hook push เข้านี่, GUI poll ออก
S.filters = S.filters or {Input=true, Remote=true, Physics=true, Raycast=true}
local MAXQ = 200

local function push(cat, text)
    if not S.enabled then return end
    if not S.filters[cat] then return end
    local q = S.queue
    q[#q+1] = {cat = cat, text = text}
    if #q > MAXQ then table.remove(q, 1) end
end
S.push = push

end

-- ==================== Value formatter (เบา — ใช้ใน hook) ====================
local function fmt(v, depth)
    depth = depth or 0
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CF(%.0f,%.0f,%.0f)", p.X, p.Y, p.Z)
    elseif t == "EnumItem" then return tostring(v)
    elseif t == "table" then
        if depth > 1 then return "{...}" end
        local parts = {}
        local n = 0
        for k, val in pairs(v) do
            n += 1; if n > 4 then parts[#parts+1] = "..."; break end
            parts[#parts+1] = fmt(val, depth+1)
        end
        return "{"..table.concat(parts, ",").."}"
    elseif t == "string" then
        if #v > 24 then return '"'..v:sub(1,24)..'..."' end
        return '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    else return tostring(v) end
end

local function fmtArgs(...)
    local n = select("#", ...)
    local parts = {}
    for i = 1, math.min(n, 6) do parts[i] = fmt(select(i, ...)) end
    if n > 6 then parts[#parts+1] = "(+"..(n-6)..")" end
    return table.concat(parts, ", ")
end

-- ==================== HOOKS (ครั้งเดียวตลอด session) ====================
local INPUT_METHODS = {
    SendTouchEvent=true, SendMouseButtonEvent=true,
    SendMouseMoveEvent=true, SendKeyEvent=true, SendMouseWheelEvent=true,
}
local REMOTE_METHODS = { FireServer=true, InvokeServer=true, fireServer=true, invokeServer=true }
local RAYCAST_METHODS = {
    Raycast=true, FindPartOnRay=true,
    FindPartOnRayWithWhitelist=true, FindPartOnRayWithIgnoreList=true, Blockcast=true,
}
local PHYSICS_CLASSES = {
    BodyGyro=true, BodyVelocity=true, BodyPosition=true, BodyAngularVelocity=true,
    AlignOrientation=true, AlignPosition=true, LinearVelocity=true, AngularVelocity=true,
    VectorForce=true, Highlight=true,
}

if okHooks and not S.hooked then
    S.hooked = true

    -- __namecall: ดัก method call (input/remote/raycast)
    local oldNamecall
    oldNamecall = hookmeta(game, "__namecall", newcc(function(self, ...)
        local ok, method = pcall(getnc)
        if ok and method then
            if INPUT_METHODS[method] then
                push("Input", method.."("..fmtArgs(...)..")")
            elseif REMOTE_METHODS[method] then
                local name = "?"
                pcall(function() name = self:GetFullName() end)
                push("Remote", method.." @"..name.." | "..fmtArgs(...))
            elseif RAYCAST_METHODS[method] then
                push("Raycast", method.."("..fmtArgs(...)..")")
            end
        end
        return oldNamecall(self, ...)
    end))
    S.restoreNamecall = oldNamecall

    -- Instance.new: ดักสร้าง physics/highlight
    local oldNew
    oldNew = hookfunc(Instance.new, newcc(function(className, ...)
        if type(className) == "string" and PHYSICS_CLASSES[className] then
            push("Physics", "Instance.new('"..className.."')")
        end
        return oldNew(className, ...)
    end))
    S.restoreNew = oldNew
end

local renderOrder = 0
local renderCount = 0
local MAXSHOW = 150
local function addLine(cat, text)
    renderOrder += 1; renderCount += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.AutomaticSize = Enum.AutomaticSize.Y
    l.BackgroundTransparency = 1; l.Font = Enum.Font.Code; l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = CAT_COL[cat] or Color3.fromRGB(200,200,200)
    l.Text = "["..cat:sub(1,3).."] "..text
    l.LayoutOrder = renderOrder; l.Parent = scroll
    -- ลบบรรทัดเก่าเกิน
    if renderCount > MAXSHOW then
        for _, c in ipairs(scroll:GetChildren()) do
            if c:IsA("TextLabel") then c:Destroy() renderCount -= 1 break end
        end
    end
end
local function clearLog()
    for _, c in pairs(scroll:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end
    renderCount = 0
end
clrB.MouseButton1Click:Connect(clearLog)

-- ==================== Poll queue → render (drain ทุกเฟรม) ====================

    local q = S.queue
    if #q == 0 then return end
    for i = 1, #q do addLine(q[i].cat, q[i].text) end
    table.clear(q)
    task.defer(function()
        scroll.CanvasSize = UDim2.new(0,0,0,layout.AbsoluteContentSize.Y + 6)
        if autoScroll then scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y) end
    end)
end)
