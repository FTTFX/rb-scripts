-- push: เก็บทั้ง queue (LOG view) + stats (STATS view = นับครั้ง)
local function push(cat, sig, text)
    if not S.enabled then return end
    if not S.filters[cat] then return end
    -- STATS: นับครั้ง + เก็บ args ล่าสุด
    local st = S.stats[sig]
    if st then st.count = st.count + 1; st.text = text
    else S.stats[sig] = {cat = cat, text = text, count = 1} end
    -- LOG queue (dedup ซ้ำ → count++)
    local q = S.queue
    local last = q[#q]
    if last and last.sig == sig then
        last.count = (last.count or 1) + 1
        return
    end
    q[#q+1] = {cat = cat, text = text, sig = sig, count = 1}
    if #q > MAXQ then table.remove(q, 1) end
end
S.push = push

-- ==================== Value formatter (เบา — ใช้ใน hook) ====================
local function fmt(v, depth, full)
    depth = depth or 0
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CF(%.0f,%.0f,%.0f)", p.X, p.Y, p.Z)
    elseif t == "EnumItem" then return tostring(v)
    elseif t == "table" then
        if depth > 2 then return "{...}" end
        local parts = {}
        local n = 0
        for k, val in pairs(v) do
            n += 1; if n > 8 then parts[#parts+1] = "..."; break end
            parts[#parts+1] = tostring(k).."="..fmt(val, depth+1, full)
        end
        return "{"..table.concat(parts, ",").."}"
    elseif t == "string" then
        local max = full and 180 or 24
        if #v > max then return '"'..v:sub(1,max)..'..."' end
        return '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    else return tostring(v) end
end

local function fmtArgs(...)
    local n = select("#", ...)
    local parts = {}
    for i = 1, math.min(n, 6) do parts[i] = fmt(select(i, ...)) end
    if n > 6 then parts[#parts+1] = "(+"..(n-6)..")" end
    return table.concat(parts, ", ")
end

-- เวอร์ชันเต็ม (ไม่ truncate) สำหรับ remote — เห็น args ที่ส่ง position/rotation
local function fmtArgsFull(...)
    local n = select("#", ...)
    local parts = {}
    for i = 1, math.min(n, 8) do parts[i] = fmt(select(i, ...), 0, true) end
    if n > 8 then parts[#parts+1] = "(+"..(n-8)..")" end
    return table.concat(parts, ", ")
end

-- dump code หลัก (ที่ loadstring กำลังจะรัน = decrypt แล้ว) ลงไฟล์ + clipboard
local function dumpSource(src)
    S.dumpN = (S.dumpN or 0) + 1
    local n = S.dumpN
    push("Loader", "ls"..n, "★ loadstring #"..n.." ("..#src.." chars) = CODE หลัก!")
    local fname = "HK_dump_"..n..".lua"
    local saved = false
    pcall(function()
        if writefile then writefile(fname, src) saved = true
            push("Loader", "lsf"..n, "   → เซฟไฟล์: "..fname) end
    end)
    pcall(function()
        if setclipboard then setclipboard(src)
            push("Loader", "lsc"..n, "   → copy clipboard แล้ว (paste ได้)") end
    end)
    if not saved and not setclipboard then
        push("Loader", "lsx"..n, "   ⚠ executor ไม่มี writefile/setclipboard")
    end
end

    local oldNamecall
    oldNamecall = hookmeta(game, "__namecall", newcc(function(self, ...)
        local ok, method = pcall(getnc)
        if ok and method then
            if RAYCAST_METHODS[method] then
                -- ⭐ เบาสุด: ไม่ format RaycastParams (ตัวการ crash) — แค่นับ

                push("Loader", "hg"..url, method.." → "..url)

                push("Input", "In"..method, method.."("..fmtArgs(...)..")")

                local name = "?"
                pcall(function() name = self:GetFullName() end)
                -- ตัด ReplicatedStorage. ออกให้สั้น + args เต็ม (เห็น position/rotation)

                push("Remote", "Rm"..method..name, method.." @"..short.." | "..fmtArgsFull(...))
            end
        end
        return oldNamecall(self, ...)
    end))

    end))

local renderOrder = 0
local renderCount = 0
local MAXSHOW = 150
local copyLines = {}
-- dedup: ถ้า signature (cat+ชื่อ method) เหมือนบรรทัดล่าสุด → รวม ×N ไม่เพิ่มบรรทัด
local lastSig, lastLabel, lastCount = nil, nil, 0
local function addLine(cat, text, count, sig)
    count = count or 1
    sig = sig or (cat..":"..(text:match("^[%w_]+") or text))
    -- Loader = event สำคัญ ไม่รวม (เห็นทุกครั้ง) | หมวดอื่น spam → รวม ×N
    if cat ~= "Loader" and sig == lastSig and lastLabel and lastLabel.Parent then
        lastCount += count
        lastLabel.Text = "["..cat:sub(1,3).."] "..text.."   ×"..lastCount
        copyLines[#copyLines] = lastLabel.Text
        return
    end
    renderOrder += 1; renderCount += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.AutomaticSize = Enum.AutomaticSize.Y
    l.BackgroundTransparency = 1; l.Font = Enum.Font.Code; l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = CAT_COL[cat] or Color3.fromRGB(200,200,200)
    l.Text = "["..cat:sub(1,3).."] "..text..(count > 1 and ("   ×"..count) or "")
    l.LayoutOrder = renderOrder; l.Parent = scroll
    lastSig = sig; lastLabel = l; lastCount = count
    copyLines[#copyLines+1] = l.Text
    if renderCount > MAXSHOW then
        for _, c in ipairs(scroll:GetChildren()) do
            if c:IsA("TextLabel") then c:Destroy() renderCount -= 1 break end
        end
    end
end
local function clearLog()
    for _, c in pairs(scroll:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end
    renderCount = 0; lastSig = nil; lastLabel = nil; table.clear(copyLines)
end

    for _, st in pairs(S.stats) do arr[#arr+1] = st end
