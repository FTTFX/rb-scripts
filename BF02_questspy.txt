-- ── Helper ────────────────────────────────────────────────────────
local function fmt(v, full)
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then local p=v.Position return string.format("CF(%.0f,%.0f,%.0f)", p.X,p.Y,p.Z)
    elseif t == "string" then
        local mx = full and 200 or 30
        if #v > mx then return '"'..v:sub(1,mx)..'..."' end
        return '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    elseif t == "boolean" then return tostring(v)
    else return t end
end
local function fmtArgs(...)
    local n = select("#", ...)
    if n == 0 then return "(ไม่มี args)" end
    local p = {}
    for i = 1, math.min(n, 8) do p[i] = fmt(select(i, ...), true) end
    return table.concat(p, " | ")
end

-- ── Hook remotes (ครั้งเดียว — ดัก FireServer/InvokeServer ที่เกี่ยวกับ quest/combat) ──
if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local m = getnc()
        if (m == "FireServer" or m == "InvokeServer") and S.enabled ~= false then
            local ok, nm = pcall(function() return self:GetFullName() end)
            nm = ok and nm or "?"
            local lower = nm:lower()
            if lower:find("quest") or lower:find("interact") or lower:find("start")
               or lower:find("hit") or lower:find("attack") or lower:find("bandit")
               or lower:find("subclass") or lower:find("bestquest") or lower:find("getbest") then
                local short = nm:gsub("^.-Remotes%.", "")
                S.log[#S.log+1] = m .. " @" .. short .. " | " .. fmtArgs(...)
            end
        end
        return oldNc(self, ...)
    end))
end

-- ── Scan (task.spawn) ─────────────────────────────────────────────
task.spawn(function()
    task.wait(0.1)
    pcall(function()

    addLine("=== BF02 QUEST SPY v1.0 (Blox Fruits) ===", C.Y)
    addLine("hook: " .. (okHooks and "ON ✅" or "OFF ❌ (executor ไม่มี hookmetamethod)"), okHooks and C.G or C.R)
    addLine("", C.S)

    -- ══ 1) Quest UI text (PlayerGui.Main.Quest) ══
    addLine("── 1) Quest UI text (PlayerGui.Main.Quest) ──", C.Y)
    local qframe = pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Quest")
    if qframe then
        local function dumpUI(node, depth)
            depth = depth or 0
            if depth > 8 then return end
            for _, c in ipairs(node:GetChildren()) do
                local pad = string.rep("  ", depth)
                if c:IsA("TextLabel") or c:IsA("TextButton") then
                    local txt = tostring(c.Text)
                    if txt ~= "" then
                        addLine(("%s[%s] %s = %s"):format(pad, c.ClassName, c.Name, '"'..txt..'"'), c:IsA("TextButton") and C.O or C.G)
                    end
                end
                dumpUI(c, depth + 1)
            end
        end
        dumpUI(qframe)
    else
        addLine("  ⚠️ ไม่เจอ PlayerGui.Main.Quest (อาจต้องรับเควสก่อน)", C.O)
    end
    addLine("", C.S)

    -- ══ 2) LP.Data (ข้อมูลผู้เล่น/เควส) ══
    addLine("── 2) LP.Data ──", C.Y)
    local data = LP:FindFirstChild("Data")
    if data then
        for _, c in ipairs(data:GetChildren()) do
            local extra = c:IsA("ValueBase") and (" = " .. tostring(c.Value)) or ""
            addLine(("  %s [%s]%s"):format(c.Name, c.ClassName, extra), C.G)
            -- dump 1 level ลึก
            if c:IsA("Folder") or c:IsA("Configuration") then
                for _, cc in ipairs(c:GetChildren()) do
                    local e2 = cc:IsA("ValueBase") and (" = " .. tostring(cc.Value)) or ""
                    addLine(("     %s [%s]%s"):format(cc.Name, cc.ClassName, e2), C.S)
                end
            end
        end
    else
        addLine("  ⚠️ ไม่เจอ LP.Data (DataLoaded=false ใน BF01 — เกมยังไม่โหลดให้)", C.O)
    end
    addLine("", C.S)

    -- ══ 3) โครงสร้าง Bandit (Enemies folder) ══
    addLine("── 3) โครงสร้าง Enemies (model แรก ๆ) ──", C.Y)
    local enemies = workspace:FindFirstChild("Enemies")
    if enemies then
        for _, e in ipairs(enemies:GetChildren()) do
            if e:IsA("Model") then
                addLine(("  ▶ %s (%s)"):format(e.Name, e.ClassName), C.M)
                local hum = e:FindFirstChildOfClass("Humanoid")
                if hum then
                    addLine(("     Humanoid: Health=%.0f MaxHealth=%.0f"):format(hum.Health, hum.MaxHealth), C.G)
                end
                local hrp = e:FindFirstChild("HumanoidRootPart")
                if hrp then
                    addLine(("     HRP pos: %.0f, %.0f, %.0f"):format(hrp.Position.X, hrp.Position.Y, hrp.Position.Z), C.S)
                end
                for _, c in ipairs(e:GetChildren()) do
                    local extra = c:IsA("ValueBase") and (" = " .. tostring(c.Value)) or ""
                    addLine(("     %s [%s]%s"):format(c.Name, c.ClassName, extra), C.S)
                end
            end
        end
    else
        addLine("  ⚠️ ไม่เจอ workspace.Enemies", C.O)
    end
    addLine("", C.S)

    -- ══ 4) NPC quest giver + prompt ══
    addLine("── 4) NPC ที่ให้เควส (มี QUEST part หรือชื่อมี Quest) ──", C.Y)
    local npcs = workspace:FindFirstChild("NPCs")
    if npcs then
        for _, n in ipairs(npcs:GetChildren()) do
            local hasQuest = n:FindFirstChild("QUEST", true) or n.Name:lower():find("quest")
            if hasQuest then
                addLine(("  ▶ %s"):format(n.Name), C.M)
                for _, d in ipairs(n:GetDescendants()) do
                    if d:IsA("ProximityPrompt") then
                        addLine(("     Prompt: action=%s objText=%s hold=%s"):format(
                            tostring(d.ActionText), tostring(d.ObjectText), tostring(d.HoldDuration)), C.G)
                    end
                end
            end
        end
    else
        addLine("  ⚠️ ไม่เจอ workspace.NPCs", C.O)
    end
    addLine("", C.S)

    -- ══ 5) Modules ระบบเควส (ชื่อไฟล์จริง) ══
    addLine("── 5) Module ระบบเควสหลัก (ReplicatedStorage) ──", C.Y)
    for _, path in ipairs({
        "QuestServiceShared", "QuestServiceShared.Classes.Quest", "QuestServiceShared.Classes.Task",
        "Quests", "Controllers.QuestController",
    }) do
        local obj = RS:FindFirstChild(path)
        if obj then
            addLine(("  ✓ %s [%s]"):format(path, obj.ClassName), C.G)
        else
            -- path มีจุด → หาทีละส่วน
            local cur = RS
            local okAll = true
            for part in path:gmatch("[^.]+") do
                cur = cur and cur:FindFirstChild(part)
                if not cur then okAll = false break end
            end
            if okAll then
                addLine(("  ✓ %s [%s]"):format(path, cur.ClassName), C.G)
            else
                addLine(("  ✗ %s (ไม่เจอ)"):format(path), C.S)
            end
        end
    end
    addLine("", C.S)

    -- ══ สรุป + วิธีใช้ hook ══
    addLine("=== วิธีใช้ (สำคัญ) ===", C.Y)
    addLine("1. ไปรับเควสจาก NPC 'Bandit Quest Giver' (กด E ที่ตัว NPC)", C.W)
    addLine("2. ตี Bandit 1 ตัว (โจมตีปกติ)", C.W)
    addLine("3. กลับมาดู log ด้านล่าง (hook จะโชว์ remote + args จริง)", C.W)
    addLine("", C.S)
    addLine("=== HOOK LOG (FireServer/InvokeServer ที่เกี่ยวกับ quest/combat) ===", C.R)

    end)
    titleLbl.Text = "BF02 Quest Spy — done (ไปรับเควส+ตีมอน แล้วกด Copy)"
end)

-- ── Drain hook log → GUI (Heartbeat) ──────────────────────────────
RunService.Heartbeat:Connect(function()
    if S.enabled == false then return end
    local n = #S.log
    if n == 0 then return end
    local newLines = {}
    for i = 1, n do newLines[i] = S.log[i] end
    S.log = {}
    for _, txt in ipairs(newLines) do
        addLine("  " .. txt, C.R)
    end
end)
