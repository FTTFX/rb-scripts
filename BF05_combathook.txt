-- ── Stringify Helper ───────────────────────────────────────────────
local function valToStr(v, depth)
    depth = depth or 0
    if depth > 2 then return "..." end
    local t = typeof(v)
    if t == "string" then return '"' .. v .. '"'
    elseif t == "number" or t == "boolean" then return tostring(v)
    elseif t == "Instance" then return v:GetFullName()
    elseif t == "Vector3" then return string.format("V3(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then return string.format("CF(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "table" then
        local parts = {}
        local count = 0
        for k, val in pairs(v) do
            count += 1
            if count > 6 then table.insert(parts, "..."); break end
            table.insert(parts, tostring(k) .. "=" .. valToStr(val, depth + 1))
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    return "<" .. t .. ">"
end

local function scanCharacterCombat()
    local char = LP.Character
    if not char then log("ไม่เจอ Character", C.R) return end
    
    log("[Character Tools/Remotes]:", C.B)
    for _, v in ipairs(char:GetDescendants()) do
        if v:IsA("RemoteEvent") or v:IsA("RemoteFunction") or v:IsA("BindableEvent") then
            log("  -> " .. v.ClassName .. ": " .. v:GetFullName(), C.G)
        end
    end
    
    local bp = LP:FindFirstChild("Backpack")
    if bp then
        log("[Backpack Tools/Remotes]:", C.B)
        for _, v in ipairs(bp:GetDescendants()) do
            if v:IsA("RemoteEvent") or v:IsA("RemoteFunction") then
                log("  -> " .. v.ClassName .. ": " .. v:GetFullName(), C.G)
            end
        end
    end
end

scanCharacterCombat()

-- ── Hook All Remote Calls (__namecall) ─────────────────────────────
if hookmeta and getnc then
    local oldNc = nil
    local callCount = 0
    
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local method = getnc()
        if (method == "FireServer" or method == "InvokeServer") and typeof(self) == "Instance" then
            local name = self.Name
            local fullName = self:GetFullName()
            local args = {...}
            
            -- กรองข้อมูลที่ไม่เกี่ยวกับการโจมตีออก เพื่อไม่ให้ log รก (เช่น Movement, Clock, Heartbeat)
            local ignore = name == "DelayedRequestFunction" or name == "Clock" or name == "SetSpawnPoint"
            
            if not ignore then
                callCount += 1
                local argStrs = {}
                for i, a in ipairs(args) do
                    table.insert(argStrs, "[" .. i .. "]=" .. valToStr(a))
                end
                
                local col = C.W
                if string.find(string.lower(name .. fullName), "combat") or string.find(string.lower(name .. fullName), "attack") or string.find(string.lower(name .. fullName), "hit") or string.find(string.lower(name .. fullName), "damage") then
                    col = C.G -- สีเขียวเด่นถ้าเกี่ยวกับ Combat/Attack
                elseif string.find(string.lower(name), "commf") then
                    col = C.P -- สีชมพูถ้าเป็น CommF_
                end
                
                log(string.format("#%d [%s] %s:%s(%s)", callCount, self.ClassName, name, method, table.concat(argStrs, ", ")), col)
                log("    Path: " .. fullName, C.S)
            end
        end
        return oldNc(self, ...)
    end))
    log("Hook __namecall: สำเร็จ ✓ (พร้อมดักการคลิกโจมตี)", C.G)
else
    log("⚠️ Executor ไม่รองรับ hookmetamethod", C.R)
end
