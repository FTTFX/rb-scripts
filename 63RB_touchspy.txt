local function getTeamID(p)
    local id = p:GetAttribute("TeamID")
    if id and id ~= "" then return id end
    if p.Team then return "T_"..p.Team.Name end
    return nil
end

-- หา head ศัตรูใกล้สุดที่มี LOS (ไม่ล็อคเพื่อนทีมเดียวกัน)
local function findTarget()
    if not LPRoot then return nil end
    local fromPos = LPRoot.Position
    local myID = getTeamID(LP)
    local ignore = {LPChar}
    rayParams.FilterDescendantsInstances = ignore
    local bestHead, bestSq = nil, math_huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hum  = p.Character:FindFirstChildOfClass("Humanoid")
            local head = p.Character:FindFirstChild("Head")
            if hum and head and hum.Health > 0 and not (myID and getTeamID(p) == myID) then
                local d = (head.Position - fromPos)
                local sq = d.X*d.X + d.Y*d.Y + d.Z*d.Z
                if sq < bestSq and sq <= SHOOT_MAX_SQ then
                    bestSq, bestHead = sq, head
                end
            end
        end
    end
    return bestHead
end

    local head = findTarget()
    watchTarget = head

        Camera.CFrame = cf_lookAt(Camera.CFrame.Position, head.Position)
    end
end)

-- จาก log: executor ต้องการ "int" (Enum.UserInputState cast ไม่ได้)
-- → state ใช้ integer: Begin=0 Change=1 End=2
-- ลองหลาย signature เพื่อหาอันที่ executor รับ แล้วบอกให้รู้
local SIG = {
    {name="(id,int,x,y)",        fn=function(id,s,x,y) VIM:SendTouchEvent(id,s,x,y) end},
    {name="(id,int,x,y,game,0)", fn=function(id,s,x,y) VIM:SendTouchEvent(id,s,x,y,game,0) end},
    {name="(id,int,x,y,0)",      fn=function(id,s,x,y) VIM:SendTouchEvent(id,s,x,y,0) end},
}
_G.TS_SPY_TOUCHSIG = _G.TS_SPY_TOUCHSIG  -- จำ signature ที่ work ไว้

local function tryTouchSwipe(variant, x, y)
    -- swipe ขวา 120px: Begin(0) → Change(1) → End(2)
    local ok, err = pcall(function()
        variant.fn(99, 0, x,     y)
        task.wait(0.02)
        variant.fn(99, 1, x+120, y)
        task.wait(0.02)
        variant.fn(99, 2, x+120, y)
    end)
    return ok, err
end

local function testTouch()
    log("── TEST TOUCH ──", CInfo)
    log("TouchEnabled="..tostring(UIS.TouchEnabled).." | VIM="..tostring(VIM ~= nil), CWarn)
    if not VIM then log("VIM nil → executor ไม่ support SendTouchEvent", CErr); return end
    local vs = Camera.ViewportSize
    local x, y = vs.X * 0.7, vs.Y * 0.5

    for idx, variant in ipairs(SIG) do
        local before = touchHits
        local ok, err = tryTouchSwipe(variant, x, y)
        if not ok then
            log("✗ "..variant.name.." → "..tostring(err):sub(1,40), CErr)
        else
            task.wait(0.1)
            if touchHits > before then
                log("✅ ใช้ได้! "..variant.name.." ("..(touchHits-before).." events)", COk)
                log("→ จำ signature นี้ไว้ใช้ใน BackFace", COk)
                _G.TS_SPY_TOUCHSIG = variant.name
                return
            else
                log("△ "..variant.name.." ไม่ throw แต่ไม่มี event", CWarn)
            end
        end
        task.wait(0.05)
    end
    log("❌ ลองครบทุก signature แล้ว — executor นี้ส่ง touch ไม่ได้", CErr)
end

-- ==================== sendTouch (signature ที่พิสูจน์แล้วว่าใช้ได้) ====================
-- VIM:SendTouchEvent(id, stateInt, x, y)  state: 0=Begin 1=Change 2=End
local function sendTouch(id, state, x, y)
    pcall(function() VIM:SendTouchEvent(id, state, x, y) end)
end

-- ==================== SWIPE CAL (swipe จริง → วัด yaw ที่หมุน) ====================
-- พิสูจน์ว่า swipe หมุน "อะไร" + calibrate กี่ px = กี่องศา
local SWIPE_PX = 250  -- ระยะ swipe ขวา (px)
local function swipeCal()
    if not VIM then log("VIM nil", CErr); return end
    if not LPRoot then log("no HRP", CErr); return end
    log("── SWIPE CAL (+"..SWIPE_PX.."px ขวา) ──", CInfo)
    local cx, cy = getTouchPoint()
    local yaw0 = yawDeg(LPRoot.CFrame)
    local camY0 = math.deg(select(2, Camera.CFrame:ToOrientation()))

    task.spawn(function()
        sendTouch(99, 0, cx, cy)         -- Begin
        task.wait(0.02)
        for s = 1, 10 do                  -- 10 step drag ขวา (smooth)
            sendTouch(99, 1, cx + (SWIPE_PX * s / 10), cy)
            task.wait(0.012)
        end
        sendTouch(99, 2, cx + SWIPE_PX, cy)  -- End
        task.wait(0.15)

        local yaw1  = yawDeg(LPRoot.CFrame)
        local camY1 = math.deg(select(2, Camera.CFrame:ToOrientation()))
        local dChar = yaw1 - yaw0
        local dCam  = camY1 - camY0
        log(string.format("ตัวละคร yaw: %.0f° → %.0f°  (Δ%+.0f°)", yaw0, yaw1, dChar),
            math.abs(dChar) > 3 and COk or CWarn)
        log(string.format("กล้อง   yaw: Δ%+.0f°", dCam), CInfo)
        if math.abs(dChar) > 3 then
            log(string.format("✅ swipe หมุนตัวละคร! %dpx = %.0f° → %.1f px/°",
                SWIPE_PX, dChar, SWIPE_PX / math.abs(dChar)), COk)
            log("→ เอาค่า px/° นี้ไปใส่ BackFace ใน 06RB ได้", COk)
        elseif math.abs(dCam) > 3 then
            log("△ swipe หมุนแค่กล้อง ไม่หมุนตัวละคร", CWarn)
        else
            log("❌ swipe ไม่หมุนอะไรเลย — touch area อาจผิด (ลองเปลี่ยนตำแหน่ง)", CErr)
        end
    end)
end

-- ==================== BackFace closed-loop (touch swipe) ====================
-- หันหลังให้ศัตรูเสมอ: คำนวณ error องศา → swipe ตามทิศ → iterate จนหันถูก
-- ทิศจาก CAL: swipe ขวา(+px) → yaw ลด → ต้องเพิ่ม yaw ต้อง swipe ซ้าย(-px)
local DEAD_DEG = 6   -- ใกล้พอแล้ว ไม่ต้อง swipe
task.spawn(function()
    while task.wait(0.07) do
        if not _G.TS_SPY_CONNS or _G.TS_SPY_CONNS ~= CONNS then return end
        if not BACKFACE_ON or not VIM or not LPRoot then continue end
        local head = watchTarget or findTarget()
        if not head then continue end

        local myPos = LPRoot.Position
        local toE   = head.Position - myPos
        local flat  = v3_new(toE.X, 0, toE.Z)
        if flat.Magnitude < 1 then continue end
        local desired = (flat * -1).Unit          -- หน้าชี้ออกจากศัตรู = หลังเข้าหา

        local lv  = LPRoot.CFrame.LookVector
        local cur = v3_new(lv.X, 0, lv.Z)
        if cur.Magnitude < 0.01 then continue end
        cur = cur.Unit

        local cross = cur:Cross(desired)
        local dot   = math.clamp(cur:Dot(desired), -1, 1)
        local errDeg = math.deg(math.atan2(cross.Y, dot))
        if math.abs(errDeg) < DEAD_DEG then continue end   -- หันถูกแล้ว

        local deltaPx = math.clamp(-errDeg * PX_PER_DEG, -120, 120)
        local cx, cy = getTouchPoint()

        sendTouch(88, 0, cx, cy)               -- Begin
        task.wait(0.012)
        sendTouch(88, 1, cx + deltaPx*0.5, cy) -- Change (กลางทาง)
        task.wait(0.012)
        sendTouch(88, 1, cx + deltaPx, cy)     -- Change (สุด)
        task.wait(0.012)
        sendTouch(88, 2, cx + deltaPx, cy)     -- End
    end
end)

-- ==================== SCAN constraints ====================
local function scanConstraints()
    log("── SCAN constraints/motors ──", CInfo)
    if not LPChar then log("no char", CErr); return end
    local found = 0
    local CLASSES = {
        BodyGyro=true, BodyAngularVelocity=true, AlignOrientation=true,
        BodyPosition=true, AlignPosition=true, LinearVelocity=true,
        AngularVelocity=true, VectorForce=true,
    }
    for _, d in ipairs(LPChar:GetDescendants()) do
        if CLASSES[d.ClassName] then
            found += 1
            log(string.format("• %s : %s", d.ClassName, d:GetFullName():gsub(LPChar.Name, "Char")), CWarn)
        end
    end
    -- network owner hint
    if LPRoot then
        log("HRP.ReceiveAge="..string.format("%.3f", LPRoot.ReceiveAge).." (สูง=server คุม)", CInfo)
        log("HRP.Anchored="..tostring(LPRoot.Anchored), CInfo)
    end
    if found == 0 then log("ไม่เจอ constraint คุมการหมุน (rotation อาจคุมจาก server-side script)", CWarn) end
    log("รวม "..found.." constraint", COk)
end
