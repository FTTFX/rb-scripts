local CErr  = Color3.fromRGB(255,90,90)

local CInfo = Color3.fromRGB(150,180,255)

-- ==================== Live rotation readout ====================
local function yawDeg(cf)
    local _, y, _ = cf:ToOrientation()
    return math.deg(y)
end
local lastYaw = nil
bind(RunService.RenderStepped, function()
    if not LPRoot then readout.Text = "no HRP"; return end
    local cf = LPRoot.CFrame
    local yaw = yawDeg(cf)
    local dy = lastYaw and (yaw - lastYaw) or 0
    lastYaw = yaw
    local av = LPRoot.AssemblyAngularVelocity
    readout.Text = string.format(
        "yaw: %6.1f°  Δ%+.1f/f\nAngVel Y: %+.2f\nAnchored:%s PlatformStand:%s",
        yaw, dy, av.Y,
        tostring(LPRoot.Anchored),
        tostring(LPHum and LPHum.PlatformStand))
end)

-- ==================== TEST TOUCH ====================
-- ยิง touch swipe ขวา + listen ว่าเกมรับไหม
local touchHits = 0
bind(UIS.TouchStarted, function(t, gp) touchHits += 1; log("  ↳ TouchStarted hit#"..touchHits.." gp="..tostring(gp), COk) end)
bind(UIS.InputBegan, function(i, gp)
    if i.UserInputType == Enum.UserInputType.Touch then
        log("  ↳ InputBegan(Touch) pos="..math.floor(i.Position.X)..","..math.floor(i.Position.Y), COk)
    end
end)

local function testTouch()
    log("── TEST TOUCH ──", CInfo)
    log("TouchEnabled="..tostring(UIS.TouchEnabled).." | VIM="..tostring(VIM ~= nil), CWarn)
    if not VIM then log("VIM nil → executor ไม่ support SendTouchEvent", CErr); return end
    local vs = Camera.ViewportSize
    local x, y = vs.X * 0.7, vs.Y * 0.5
    local before = touchHits
    local ok, err = pcall(function()
        VIM:SendTouchEvent(99, Enum.UserInputState.Begin,  x,     y, game, 0)
        task.wait(0.02)
        VIM:SendTouchEvent(99, Enum.UserInputState.Change, x+120, y, game, 0)
        task.wait(0.02)
        VIM:SendTouchEvent(99, Enum.UserInputState.End,    x+120, y, game, 0)
    end)
    if not ok then log("SendTouchEvent ERROR: "..tostring(err), CErr); return end
    task.wait(0.1)
    if touchHits > before then
        log("✅ เกมรับ touch! ("..(touchHits-before).." events)", COk)
    else
        log("❌ ยิง touch แล้วแต่ไม่มี event ตอบ — เกมไม่รับ synthetic touch", CErr)
    end
end

-- ==================== TEST SPIN (พิสูจน์ server-lock) ====================
-- หมุน HRP 90° แล้ววัดว่าอยู่กี่ ms ก่อนถูก reset กลับ
local function testSpin()
    if not LPRoot then log("no HRP", CErr); return end
    log("── TEST SPIN (90°) ──", CInfo)
    local startYaw = yawDeg(LPRoot.CFrame)
    local target = LPRoot.CFrame * CFrame.Angles(0, math.rad(90), 0)
    LPRoot.CFrame = target
    local setYaw = yawDeg(LPRoot.CFrame)
    log(string.format("set yaw %.0f° → %.0f°", startYaw, setYaw), CWarn)

    -- จับเวลาจนกว่ามันจะ drift กลับเข้าใกล้ startYaw
    task.spawn(function()
        local t0 = os.clock()
        local heldMs = 0
        for i = 1, 120 do  -- ~2 วินาที
            RunService.RenderStepped:Wait()
            local cur = yawDeg(LPRoot.CFrame)
            local backDiff = math.abs(cur - startYaw)
            if backDiff < 20 then
                heldMs = (os.clock() - t0) * 1000
                break
            end
        end
        if heldMs > 0 then
            log(string.format("⚠ ถูก reset กลับใน %.0f ms → SERVER ล็อค rotation", heldMs), CErr)
            log("→ หมุน client ได้แต่ server เขียนทับ (ต้องใช้ touch input จริง)", CWarn)
        else
            log("✅ หมุนค้างได้ >2s → rotation เป็น client-side (CFrame hook ใช้ได้)", COk)
        end
    end)
end

-- ==================== SCAN constraints ====================
local function scanConstraints()
    log("── SCAN constraints/motors ──", CInfo)
    if not LPChar then log("no char", CErr); return end
    local found = 0
    local CLASSES = {
        BodyGyro=true, BodyAngularVelocity=true, AlignOrientation=true,
        BodyPosition=true, AlignPosition=true, LinearVelocity=true,
        AngularVelocity=true, VectorForce=true,
    }
    for _, d in ipairs(LPChar:GetDescendants()) do
        if CLASSES[d.ClassName] then
            found += 1
            log(string.format("• %s : %s", d.ClassName, d:GetFullName():gsub(LPChar.Name, "Char")), CWarn)
        end
    end
    -- network owner hint
    if LPRoot then
        log("HRP.ReceiveAge="..string.format("%.3f", LPRoot.ReceiveAge).." (สูง=server คุม)", CInfo)
        log("HRP.Anchored="..tostring(LPRoot.Anchored), CInfo)
    end
    if found == 0 then log("ไม่เจอ constraint คุมการหมุน (rotation อาจคุมจาก server-side script)", CWarn) end
    log("รวม "..found.." constraint", COk)
end

touchB.MouseButton1Click:Connect(testTouch)
spinB.MouseButton1Click:Connect(testSpin)
scanB.MouseButton1Click:Connect(scanConstraints)

bind(UIS.InputBegan, function(i, gp)

    elseif i.KeyCode == Enum.KeyCode.K then testSpin()
    elseif i.KeyCode == Enum.KeyCode.L then scanConstraints()
    end
end)
