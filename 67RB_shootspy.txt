-- ==================== Arg formatter (ละเอียด — เน้น aim data) ====================
local function fmt(v)
    local t = typeof(v)
    if t == "Instance" then
        -- target! แสดง class + name + path สั้น
        local par = v.Parent and v.Parent.Name or "?"
        return "🎯"..v.ClassName..":"..v.Name.."(in "..par..")"
    elseif t == "Vector3" then
        return "📍V3("..math.floor(v.X)..","..math.floor(v.Y)..","..math.floor(v.Z)..")"
    elseif t == "CFrame" then
        local p = v.Position
        return "📐CF("..math.floor(p.X)..","..math.floor(p.Y)..","..math.floor(p.Z)..")"
    elseif t == "string" then
        if #v > 30 then return '"'..v:sub(1,30)..'..."' end
        return '"'..v..'"'
    elseif t == "number" then return tostring(v)
    elseif t == "boolean" then return tostring(v)
    elseif t == "table" then
        local n = 0; for _ in pairs(v) do n += 1 end
        return "{table x"..n.."}"
    else return t end
end
local function fmtArgs(...)
    local n = select("#", ...)
    if n == 0 then return "(ไม่มี args)" end
    local parts = {}
    for i = 1, math.min(n, 8) do parts[i] = fmt(select(i, ...)) end
    return table.concat(parts, "  |  ")
end

-- ==================== Shoot detection ====================
-- กดยิง (mouse1 / touch) → mark ช่วง 350ms ว่า "กำลังยิง"
bind(UIS.InputBegan, function(i, gp)
    local ut = i.UserInputType
    if ut == Enum.UserInputType.MouseButton1 or ut == Enum.UserInputType.Touch then
        S.shootUntil = os.clock() + 0.35
    end
end)

-- ==================== HOOK FireServer ====================
if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local ok, method = pcall(getnc)
        if ok and (method == "FireServer" or method == "InvokeServer") then
            local shooting = os.clock() < S.shootUntil
            if (not S.onlyShoot) or shooting then
                local name = ""
                pcall(function() name = self:GetFullName() end)
                local short = name:gsub("^.-Remotes%.", ""):gsub("^ReplicatedStorage%.", "")
                -- ข้าม remote ที่รู้ว่าไม่ใช่ยิง (state/camera spam)
                if not (name:find("UpdateState") or name:find("CameraRotation") or name:find("Spectating")) then
                    local mark = shooting and "🔫 " or ""
                    S.queue[#S.queue+1] = mark..short.." | "..fmtArgs(...)
                    if #S.queue > MAXQ then table.remove(S.queue, 1) end
                end
            end
        end
        return oldNc(self, ...)
    end))
    S.restore = oldNc
end

local order = 0
local copyLines = {}
local MAXSHOW = 100
local lastText, lastLbl, lastN = nil, nil, 0
local function addLine(txt)
    if txt == lastText and lastLbl and lastLbl.Parent then
        lastN += 1; lastLbl.Text = txt.."   ×"..lastN
        copyLines[#copyLines] = lastLbl.Text
        return
    end
    order += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.AutomaticSize = Enum.AutomaticSize.Y
    l.BackgroundTransparency = 1; l.Font = Enum.Font.Code; l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = txt:find("🔫") and Color3.fromRGB(255,200,120) or Color3.fromRGB(180,200,255)
    l.Text = txt; l.LayoutOrder = order; l.Parent = scroll
    lastText = txt; lastLbl = l; lastN = 1
    copyLines[#copyLines+1] = txt
    local cnt = 0
    for _, c in ipairs(scroll:GetChildren()) do if c:IsA("TextLabel") then cnt += 1 end end
    if cnt > MAXSHOW then
        for _, c in ipairs(scroll:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() break end end
    end
end

-- ==================== Buttons ====================
local function updOnly()
    onlyB.BackgroundColor3 = S.onlyShoot and Color3.fromRGB(180,50,40) or Color3.fromRGB(60,60,90)
    onlyB.Text = S.onlyShoot and "🔫 ONLY SHOOT" or "ALL remotes"
end
onlyB.MouseButton1Click:Connect(function() S.onlyShoot = not S.onlyShoot updOnly() end)

local function toggleFreeze()
    S.frozen = not S.frozen
    freezeB.Text = S.frozen and "FROZEN [F]" or "FREEZE [F]"
    freezeB.BackgroundColor3 = S.frozen and Color3.fromRGB(180,40,40) or Color3.fromRGB(120,90,30)
end
freezeB.MouseButton1Click:Connect(toggleFreeze)
bind(UIS.InputBegan, function(i, gp)
    if not gp and i.KeyCode == Enum.KeyCode.F then toggleFreeze() end
end)

clrB.MouseButton1Click:Connect(function()
    for _, c in pairs(scroll:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end
    order = 0; lastText = nil; table.clear(copyLines)
end)
copyB.MouseButton1Click:Connect(function()
    local clip = setclipboard or toclipboard or write_clipboard
    local ok = clip and pcall(clip, table.concat(copyLines, "\n"))
    copyB.Text = ok and "Copied!" or "ไม่รองรับ"
    task.delay(2, function() if copyB.Parent then copyB.Text = "COPY" end end)
end)

-- ==================== Drain ====================
bind(RunService.Heartbeat, function()
    if S.frozen then return end
    local q = S.queue
    if #q == 0 then return end
    for i = 1, #q do addLine(q[i]) end
    table.clear(q)
    task.defer(function()
        scroll.CanvasSize = UDim2.new(0,0,0,layout.AbsoluteContentSize.Y + 6)
        scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
    end)
end)
