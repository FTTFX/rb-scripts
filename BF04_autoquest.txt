-- ── Log system ─────────────────────────────────────────────────────
local logLines = {}
local logOrder = 0
local logColors = {
    G = Color3.fromRGB(80,255,120),   -- green (สำเร็จ)
    Y = Color3.fromRGB(255,210,0),    -- yellow (action)
    R = Color3.fromRGB(255,80,80),    -- red (error)
    B = Color3.fromRGB(140,160,255),  -- blue (info)
    S = Color3.fromRGB(140,140,140),  -- gray
}
local function log(txt, col)
    local c = logColors[col] or logColors.S
    logOrder += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.BackgroundTransparency = 1; l.Font = Enum.Font.Code
    l.TextSize = 11; l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = c; l.Text = txt; l.LayoutOrder = logOrder; l.Parent = scroll
    table.insert(logLines, txt)
    scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
end

-- ── Helper ────────────────────────────────────────────────────────
local CommF = nil
local function getCommF()
    if CommF and CommF.Parent then return CommF end
    local remotes = RS:FindFirstChild("Remotes")
    if remotes then
        for _, r in ipairs(remotes:GetDescendants()) do
            if r.Name == "CommF_" and r:IsA("RemoteFunction") then
                CommF = r
                return CommF
            end
        end
    end
    return nil
end

local function getChar()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if char and hrp and hum then return char, hrp, hum end
    return nil
end

-- ล็อกตัว (กันฟิสิกส์ดึงกลับ) — ใช้ตอน tween
local function lockChar(hum)
    pcall(function()
        if hum then hum:ChangeState(Enum.HumanoidStateType.Physics) end
    end)
end

-- Tween TP (ลื่น ๆ — ไม่กระตุก)
local tween = nil
local function tp(cframe)
    local char, hrp, hum = getChar()
    if not char then return false end
    local dist = (hrp.Position - cframe.Position).Magnitude
    if dist < 1 then return true end
    if tween then pcall(function() tween:Cancel() end) end
    lockChar(hum)
    local dur = math.clamp(dist / CFG.TWEEN_SPEED, 0.2, 4)
    tween = TweenService:Create(hrp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = cframe})
    tween:Play()
    tween.Completed:Wait()
    return true
end

-- อ่าน progress จาก QuestTitle.Title.Text
local function readProgress()
    local main = pg:FindFirstChild("Main")
    local q = main and main:FindFirstChild("Quest")
    if not q then return nil, nil, nil end
    local ok, txt = pcall(function() return q.Container.QuestTitle.Title.Text end)
    if not ok or not txt then return nil, nil, nil end
    local cur, need = txt:match("%((%d+)/(%d+)%)")
    return tonumber(cur), tonumber(need), txt
end

-- หามอนใกล้สุด
local function findMob()
    local folder = Workspace:FindFirstChild(CFG.MOB_FOLDER)
    if not folder then return nil, nil end
    local char, hrp = getChar()
    local best, bestDist = nil, math.huge
    for _, m in ipairs(folder:GetChildren()) do
        if m:IsA("Model") and m.Name == CFG.MOB_NAME then
            local hum = m:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local mhrp = m:FindFirstChild("HumanoidRootPart")
                if mhrp then
                    if hrp then
                        local d = (mhrp.Position - hrp.Position).Magnitude
                        if d < bestDist then best, bestDist = m, d end
                    else
                        return m, 0
                    end
                end
            end
        end
    end
    return best, bestDist
end

-- หันไปหามอน (ยืนใกล้พอตี)
local function aimAt(mob)
    local char, hrp = getChar()
    local mhrp = mob and mob:FindFirstChild("HumanoidRootPart")
    if not char or not mhrp then return end
    local dir = hrp.Position - mhrp.Position
    local flat = Vector3.new(dir.X, 0, dir.Z)
    if flat.Magnitude < 0.1 then flat = Vector3.new(0,0,-1) end
    hrp.CFrame = CFrame.lookAt(mhrp.Position + flat.Unit * 3 + Vector3.new(0,0.5,0), mhrp.Position)
    local cam = Workspace.CurrentCamera
    if cam then cam.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0,5,0), mhrp.Position) end
end

-- M1 (VirtualUser)
local function attackOn()
    pcall(function() VU:CaptureController() end)
    pcall(function() VU:Button1Down() end)
end
local function attackOff()
    pcall(function() VU:Button1Up() end)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    local down = false
    log("=== START AUTO ===", "Y")
    while CFG.AUTO_ON and _G.BF04_LOOP do
        task.wait(0.1)

        local char, hrp = getChar()
        if not char then
            log("รอตัวละคร spawn...", "S")
            task.wait(1)
            continue
        end

        local cur, need, txt = readProgress()

        if not cur then
            -- ไม่มีเควส → ไป NPC + รับ
            if not getCommF() then
                log("หา CommF_ ไม่เจอ (รอเกมโหลด?)", "R")
                task.wait(1)
            elseif (hrp.Position - CFG.QUEST_CFRAME.Position).Magnitude > 12 then
                log("TP ไป NPC (ระยะ " .. math.floor((hrp.Position - CFG.QUEST_CFRAME.Position).Magnitude) .. ")", "B")
                if down then attackOff() down = false end
                tp(CFG.QUEST_CFRAME)
            else
                log("รับเควส: StartQuest(" .. CFG.QUEST .. ", " .. CFG.LEVEL .. ")", "Y")
                local ok = pcall(function() CommF:InvokeServer("StartQuest", CFG.QUEST, CFG.LEVEL) end)
                log(ok and "  → รับเควสแล้ว" or "  → รับเควส error", ok and "G" or "R")
                task.wait(1.5)
            end
        elseif cur >= need then
            -- เสร็จ → รับใหม่
            log("เควสเสร็จ (" .. cur .. "/" .. need .. ") → Abandon + Start ใหม่", "G")
            if down then attackOff() down = false end
            if getCommF() then
                pcall(function() CommF:InvokeServer("AbandonQuest") end)
                task.wait(0.8)
                pcall(function() CommF:InvokeServer("StartQuest", CFG.QUEST, CFG.LEVEL) end)
            end
            task.wait(1.5)
        else
            -- มีเควส → ตีมอน
            local mob, dist = findMob()
            if mob then
                if dist > 25 then
                    log("TP ไปจุดมอน (ระยะ " .. math.floor(dist) .. ")", "B")
                    if down then attackOff() down = false end
                    tp(CFG.MON_CFRAME)
                    task.wait(0.5)
                else
                    if not down then
                        log("เริ่มตี " .. CFG.MOB_NAME .. " (" .. cur .. "/" .. need .. ")", "Y")
                    end
                    aimAt(mob)
                    attackOn()
                    down = true
                end
            else
                log("หามอนไม่เจอ → TP จุดรวมมอน", "R")
                if down then attackOff() down = false end
                tp(CFG.MON_CFRAME)
                task.wait(1)
            end
        end
    end
    if down then attackOff() end
    log("=== STOP AUTO ===", "R")
end

    end
