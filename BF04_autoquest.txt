-- ── Log ────────────────────────────────────────────────────────────
local logLines = {}
local logOrder = 0
local lastState = ""
local logColors = {
    G = Color3.fromRGB(80,255,120), Y = Color3.fromRGB(255,210,0),
    R = Color3.fromRGB(255,80,80), B = Color3.fromRGB(140,160,255),
    S = Color3.fromRGB(140,140,140),
}
local function log(txt, col)
    local c = logColors[col] or logColors.S
    logOrder += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.BackgroundTransparency = 1; l.Font = Enum.Font.Code
    l.TextSize = 11; l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = c; l.Text = txt; l.LayoutOrder = logOrder; l.Parent = scroll
    table.insert(logLines, txt)
    scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
end
local function setState(txt, col)
    if txt ~= lastState then
        lastState = txt
        log(txt, col)
    end
end
copyBtn.MouseButton1Click:Connect(function()
    local clip = setclipboard or toclipboard or write_clipboard or (getgenv and getgenv().setclipboard)
    local ok = clip and pcall(clip, table.concat(logLines, "\n"))
    copyBtn.Text = ok and "Copied!" or "ไม่รองรับ"
    task.delay(2, function() if copyBtn.Parent then copyBtn.Text = "Copy Log" end end)
end)

-- ── Helper ────────────────────────────────────────────────────────
local function getCommF()
    local remotes = RS:FindFirstChild("Remotes")
    if remotes then
        for _, r in ipairs(remotes:GetDescendants()) do
            if r.Name == "CommF_" and r:IsA("RemoteFunction") then return r end
        end
    end
    return nil
end

local function getChar()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if char and hrp and hum then return char, hrp, hum end
    return nil
end

local function tp(cframe)
    local char, hrp = getChar()
    if not char then return false end
    hrp.CFrame = cframe
    if not _G.BF04_BV or not _G.BF04_BV.Parent then
        local bv = Instance.new("BodyVelocity")
        bv.Name = "BF04_Noclip"
        bv.Parent = hrp
        bv.MaxForce = Vector3.new(9e99, 9e99, 9e99)
        bv.Velocity = Vector3.new(0, 0, 0)
        _G.BF04_BV = bv
    end
    return true
end

local function equipWeapon()
    local char, hrp, hum = getChar()
    if not char then return false end
    local tool = char:FindFirstChild(CFG.WEAPON_NAME)
    if not tool then
        local bp = LP:FindFirstChild("Backpack")
        tool = bp and bp:FindFirstChild(CFG.WEAPON_NAME)
    end
    if tool and tool:IsA("Tool") then
        pcall(function() hum:EquipTool(tool) end)
        return true
    end
    return false
end

local function aimCam(mob)
    local char, hrp = getChar()
    local mhrp = mob and mob:FindFirstChild("HumanoidRootPart")
    if not char or not mhrp then return end
    local cam = Workspace.CurrentCamera
    if cam then
        cam.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0, 2, 0), mhrp.Position)
    end
end

-- ปิดหน้าต่างคุย NPC (ตัวร้ายที่ดูดคลิกไปกดปุ่มเควส)
local function closeDialogue()
    pcall(function()
        local main = pg:FindFirstChild("Main")
        if not main then return end
        for _, name in ipairs({"Dialogue", "Conversations", "Talk", "NPCDialog"}) do
            local d = main:FindFirstChild(name)
            if d and d:IsA("GuiObject") and d.Visible then d.Visible = false end
        end
    end)
end

-- ⚔️ คลิก M1 เสมือน (แบบ v1.9 ที่พิสูจน์แล้วว่าฆ่ามอนได้)
-- VirtualUser คือคลิกระดับ engine ไม่ใช่ดึงเมาส์จริง และไม่ต้องอาศัย CombatFramework
local function getClickPos()
    local ok, pos = pcall(function()
        local vp = Workspace.CurrentCamera.ViewportSize
        return Vector2.new(vp.X * 0.5, vp.Y * 0.55)
    end)
    if ok and pos then return pos end
    return Vector2.new(1280, 672)
end

local Mouse = LP:GetMouse()
local function fireMouseSignal(down)
    local sig = down and Mouse.Button1Down or Mouse.Button1Up
    if typeof(firesignal) == "function" then pcall(firesignal, sig) end
    if typeof(getconnections) == "function" then
        pcall(function()
            for _, c in ipairs(getconnections(sig)) do
                pcall(function() c:Fire() end)
            end
        end)
    end
end

local function attackDown()
    local pos = getClickPos()
    fireMouseSignal(true)
    if typeof(mouse1press) == "function" then pcall(mouse1press) end
    pcall(function()
        game:GetService("VirtualInputManager"):SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 0)
    end)
    pcall(function()
        VU:CaptureController()
        VU:Button1Down(pos)
    end)
end
local function attackUp()
    local pos = getClickPos()
    fireMouseSignal(false)
    if typeof(mouse1release) == "function" then pcall(mouse1release) end
    pcall(function()
        game:GetService("VirtualInputManager"):SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 0)
    end)
    pcall(function() VU:Button1Up() end)
end

local function fastAttack()
    pcall(function()
        local char = LP.Character
        if char then
            if char:FindFirstChild("Stun") then char.Stun.Value = 0 end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum.Sit = false end
        end
    end)
    closeDialogue()   -- สำคัญ: ปิดหน้าต่างคุยก่อนคลิกทุกครั้ง กันคลิกไปโดนปุ่มเควส
    attackDown()
    task.delay(0.07, attackUp)
end
local function attackOff()
    pcall(function() VU:Button1Up() end)
    fireMouseSignal(false)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    log("=== START AUTO v2.1 ===", "Y")
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ! (รอเกมโหลดแล้วรันใหม่)", "R") return end
    log("CommF_ เจอ ✓", "G")
    log("โหมดตี: คลิกเสมือน v1.9 (ที่ฆ่ามอนได้) + ปิดหน้าต่างคุยทุก tick", "G")

    while CFG.AUTO_ON and _G.BF04_LOOP and not getChar() do
        setState("รอตัวละคร spawn...", "S")
        task.wait(1)
    end

    while CFG.AUTO_ON and _G.BF04_LOOP do
        task.wait(0.1)
        pcall(function()
            local char, hrp = getChar()
            if not char then setState("รอตัวละคร...", "S") return end

            local UIQ = pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Quest")
            if not UIQ then setState("หา Quest UI ไม่เจอ", "R") return end

            local titleTxt = ""
            pcall(function() titleTxt = UIQ.Container.QuestTitle.Title.Text end)

            if UIQ.Visible and titleTxt ~= "" and not string.find(titleTxt, CFG.MOB_NAME) then
                setState("เควสไม่ใช่ Bandit → AbandonQuest", "Y")
                CommF:InvokeServer("AbandonQuest")
                task.wait(0.5)
                return
            end

            -- 1. ยังไม่มีเควส → เทเลไป NPC → รับครั้งเดียว → รอ UI → บินหนีออกจาก NPC
            if not UIQ.Visible then
                local dist = (CFG.QUEST_CFRAME.Position - hrp.Position).Magnitude
                if dist > 15 then
                    setState("เทเลไป NPC (ระยะ " .. math.floor(dist) .. ")", "B")
                    tp(CFG.QUEST_CFRAME)
                else
                    setState("รับเควส: StartQuest(" .. CFG.QUEST_NAME .. "," .. CFG.QUEST_NUM .. ")", "Y")
                    pcall(function()
                        CommF:InvokeServer("StartQuest", CFG.QUEST_NAME, CFG.QUEST_NUM)
                    end)

                    -- รอให้ UI เควสขึ้น (สูงสุด 2 วิ) — ไม่ยิง StartQuest เบิ้ลเด็ดขาด
                    local waited = 0
                    while not UIQ.Visible and waited < 20 and CFG.AUTO_ON and _G.BF04_LOOP do
                        task.wait(0.1)
                        waited += 1
                        closeDialogue()
                    end

                    if UIQ.Visible then
                        setState("รับเควสสำเร็จ ✓ → บินไปจุดมอนทันที", "G")
                        tp(CFG.MOB_SPAWN_CF)   -- หนีออกจาก NPC กันหน้าต่างคุยเด้งซ้ำ
                        task.wait(0.3)
                    end
                end

            -- 2. มีเควสแล้ว → ล็อกมอน → คลิกเสมือนตี
            else
                closeDialogue()

                local enemies = Workspace:FindFirstChild("Enemies")
                local target = nil
                if enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        if v.Name == CFG.MOB_NAME and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            target = v
                            break
                        end
                    end
                end

                if not target then
                    attackOff()
                    setState("หามอนไม่เจอ (บินรอที่จุดเกิดมอน)...", "Y")
                    tp(CFG.MOB_SPAWN_CF)
                    task.wait(0.5)
                    return
                end

                local v = target
                setState("ล็อกมอน: " .. v.Name .. " (HP " .. math.floor(v.Humanoid.Health) .. ") — คลิกเสมือน", "Y")

                equipWeapon()

                while v.Parent and v.Humanoid.Health > 0 and UIQ.Visible and CFG.AUTO_ON and _G.BF04_LOOP do
                    local mhrp = v:FindFirstChild("HumanoidRootPart")
                    if mhrp then
                        tp(mhrp.CFrame * CFrame.new(0, 0, 5))
                        aimCam(v)
                    end
                    fastAttack()
                    task.wait(0.15)
                end
                attackOff()

                if v.Parent and v.Humanoid.Health <= 0 then
                    setState("มอนตาย ✓ (HP 0)", "G")
                end
            end
        end)
    end
    attackOff()
    pcall(function() if _G.BF04_BV then _G.BF04_BV:Destroy() _G.BF04_BV = nil end end)
    log("=== STOP AUTO ===", "R")
end

autoBtn.MouseButton1Click:Connect(function()
    CFG.AUTO_ON = not CFG.AUTO_ON
    autoBtn.Text = "AUTO: " .. (CFG.AUTO_ON and "ON" or "OFF")
    autoBtn.BackgroundColor3 = CFG.AUTO_ON and Color3.fromRGB(180,40,40) or Color3.fromRGB(30,120,50)
    if CFG.AUTO_ON then
        _G.BF04_LOOP = true
        lastState = ""
        task.spawn(mainLoop)
    else
        _G.BF04_LOOP = nil
        attackOff()
        log("หยุดโดยผู้ใช้", "R")
    end
end)
