local logLines = {}

    G = Color3.fromRGB(80,255,120), Y = Color3.fromRGB(255,210,0),
    R = Color3.fromRGB(255,80,80), B = Color3.fromRGB(140,160,255),

local function log(txt, col)
    local c = logColors[col] or logColors.S
    logOrder += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.BackgroundTransparency = 1; l.Font = Enum.Font.Code
    l.TextSize = 11; l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true

    table.insert(logLines, txt)
    scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
end
local function setState(txt, col)
    if txt ~= lastState then
        lastState = txt
        log(txt, col)
    end
end

-- ── Helper ────────────────────────────────────────────────────────
local function getCommF()
    local remotes = RS:FindFirstChild("Remotes")
    if remotes then
        for _, r in ipairs(remotes:GetDescendants()) do
            if r.Name == "CommF_" and r:IsA("RemoteFunction") then return r end
        end
    end
    return nil
end

local function getChar()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if char and hrp and hum then return char, hrp, hum end
    return nil
end

local TweenService = game:GetService("TweenService")
local currentTween = nil

local function stopTween()
    if currentTween then
        pcall(function() currentTween:Cancel() end)
        currentTween = nil
    end
end

local function applyBodyVelocity(hrp)
    if not _G.BF04_BV or not _G.BF04_BV.Parent then
        local bv = Instance.new("BodyVelocity")
        bv.Name = "BF04_Noclip"
        bv.Parent = hrp
        bv.MaxForce = Vector3.new(9e99, 9e99, 9e99)
        bv.Velocity = Vector3.new(0, 0, 0)
        _G.BF04_BV = bv
    end
end

-- ระบบเคลื่อนที่อัจฉริยะ (Smart Travel):
-- 1. ใกล้ (< 50 studs): วาร์ปประชิดตรงๆ รวดเร็ว
-- 2. ไกล (>= 50 studs เช่น ข้ามเกาะ): ยกตัวขึ้นฟ้า (High Altitude) แล้ว Tween บินไปแบบปลอดภัย ไม่เร็วเกินไป ไม่ติดต้นไม้/สิ่งกีดขวาง ไม่โดน Anti-Cheat ดีดกลับ
local function toTarget(targetCF)
    local char, hrp, hum = getChar()
    if not char then return false end
    
    applyBodyVelocity(hrp)
    
    local dist = (targetCF.Position - hrp.Position).Magnitude
    
    -- ระยะประชิด (ตีมอน) → วาร์ปตรงได้เลย
    if dist < 50 then
        stopTween()
        hrp.CFrame = targetCF
        return true
    end
    
    -- ระยะไกล (ข้ามเกาะ / ไป NPC) → บินสูง + Tween แบบ Linear (ความเร็วปลอดภัย ~250 studs/sec)
    stopTween()
    
    -- จุดบินยกสูงข้ามสิ่งกีดขวาง (ระดับความสูงปลอดภัย ~120-150 studs)
    local highY = math.max(hrp.Position.Y, targetCF.Position.Y) + 120
    if targetCF.Position.Y > 300 then -- เผื่อเกาะลอยฟ้า
        highY = targetCF.Position.Y + 60
    end
    
    local skyStartCF = CFrame.new(hrp.Position.X, highY, hrp.Position.Z)
    local skyEndCF   = CFrame.new(targetCF.Position.X, highY, targetCF.Position.Z)
    
    -- Step 1: ลอยตัวขึ้นฟ้าตรงๆ ก่อน (เร็ว)
    hrp.CFrame = skyStartCF
    task.wait(0.05)
    
    -- Step 2: บินเลื่อนไปบนฟ้า (ความเร็วคงที่ ไม่เร็วเกินไป)
    local flyDist = (skyEndCF.Position - skyStartCF.Position).Magnitude
    local flySpeed = 260 -- studs per second (ความเร็วมาตรฐานของ Auto Farm ปลอดภัย)
    local flyTime = math.max(flyDist / flySpeed, 0.3)
    
    local tweenInfo = TweenInfo.new(flyTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
    local tw = TweenService:Create(hrp, tweenInfo, {CFrame = skyEndCF})
    currentTween = tw
    tw:Play()
    
    local waited = 0
    while AUTO_ON and _G.BF04_LOOP and currentTween == tw and waited < (flyTime + 0.5) do
        local curDist = (skyEndCF.Position - hrp.Position).Magnitude
        if curDist <= 15 then break end
        task.wait(0.1)
        waited += 0.1
    end
    
    -- Step 3: ร่อนลงสู่จุดเป้าหมาย
    if AUTO_ON and _G.BF04_LOOP then
        hrp.CFrame = targetCF
    end
    
    stopTween()
    return true
end

local function tp(cframe)
    return toTarget(cframe)
end

-- ถืออาวุธ Melee/Combat หรืออาวุธอะไรก็ได้ที่มี
local function equipWeapon()
    local char, hrp, hum = getChar()
    if not char then return false end
    
    -- ถ้าถืออยู่แล้วไม่ต้องทำอะไร
    if char:FindFirstChildOfClass("Tool") then return true end
    
    local bp = LP:FindFirstChild("Backpack")
    if not bp then return false end
    
    -- หา Combat ก่อน หรือหยิบ Tool แรกในกระเป๋า
    local tool = bp:FindFirstChild("Combat") or bp:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() hum:EquipTool(tool) end)
        return true
    end
    return false
end

local function aimCam(mob)
    local char, hrp = getChar()
    local mhrp = mob and mob:FindFirstChild("HumanoidRootPart")
    if not char or not mhrp then return end
    local cam = Workspace.CurrentCamera
    if cam then
        cam.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0, 2, 0), mhrp.Position)
    end
end

local function closeDialogue()
    pcall(function()
        local main = pg:FindFirstChild("Main")
        if not main then return end
        for _, name in ipairs({"Dialogue", "Conversations", "Talk", "NPCDialog"}) do
            local d = main:FindFirstChild(name)
            if d and d:IsA("GuiObject") and d.Visible then d.Visible = false end
        end
    end)
end

local function getClickPos()
    local ok, pos = pcall(function()
        local vp = Workspace.CurrentCamera.ViewportSize
        return Vector2.new(vp.X * 0.5, vp.Y * 0.55)
    end)
    if ok and pos then return pos end
    return Vector2.new(1280, 672)
end

local Mouse = LP:GetMouse()
local function fireMouseSignal(down)
    local sig = down and Mouse.Button1Down or Mouse.Button1Up
    if typeof(firesignal) == "function" then pcall(firesignal, sig) end
    if typeof(getconnections) == "function" then
        pcall(function()
            for _, c in ipairs(getconnections(sig)) do
                pcall(function() c:Fire() end)
            end
        end)
    end
end

local function attackDown()
    local pos = getClickPos()
    fireMouseSignal(true)
    if typeof(mouse1press) == "function" then pcall(mouse1press) end
    pcall(function()
        game:GetService("VirtualInputManager"):SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 0)
    end)
    pcall(function()
        VU:CaptureController()
        VU:Button1Down(pos)
    end)
end

local function attackUp()
    local pos = getClickPos()
    fireMouseSignal(false)
    if typeof(mouse1release) == "function" then pcall(mouse1release) end
    pcall(function()
        game:GetService("VirtualInputManager"):SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 0)
    end)
    pcall(function() VU:Button1Up() end)
end

local function fastAttack()
    pcall(function()
        local char = LP.Character
        if char then
            if char:FindFirstChild("Stun") then char.Stun.Value = 0 end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum.Sit = false end
        end
    end)
    closeDialogue()
    attackDown()
    task.delay(0.07, attackUp)
end

local function attackOff()
    pcall(function() VU:Button1Up() end)
    fireMouseSignal(false)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    log("=== START AUTO v3.0 (Dynamic Quest) ===", "Y")
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ! (รอเกมโหลดแล้วรันใหม่)", "R") return end
    log("CommF_ เจอ ✓", "G")

    while AUTO_ON and _G.BF04_LOOP and not getChar() do
        setState("รอตัวละคร spawn...", "S")
        task.wait(1)
    end

    while AUTO_ON and _G.BF04_LOOP do
        task.wait(0.1)
        pcall(function()
            local char, hrp = getChar()
            if not char then setState("รอตัวละคร...", "S") return end

            local currentLevel = getMyLevel()
            local qdata = getQuestData(currentLevel)

            local UIQ = pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Quest")
            if not UIQ then setState("หา Quest UI ไม่เจอ", "R") return end

            local titleTxt = ""
            pcall(function() titleTxt = UIQ.Container.QuestTitle.Title.Text end)

            -- ถ้ามีเควสค้างอยู่แต่ไม่ตรงกับเลเวลปัจจุบัน หรือไม่ใช่เควสที่ต้องการ → ยกเลิกเพื่อรับเควสใหม่ที่ดีกว่า
            if UIQ.Visible and titleTxt ~= "" and not string.find(titleTxt, qdata.mob) then
                setState("เควสไม่ตรงกับเลเวล " .. currentLevel .. " (" .. qdata.mob .. ") → AbandonQuest", "Y")
                CommF:InvokeServer("AbandonQuest")
                task.wait(0.5)
                return
            end

            -- 1. ยังไม่มีเควส → บินไปหา NPC ตามเลเวล → รับเควส
            if not UIQ.Visible then
                attackOff()
                local dist = (qdata.questCF.Position - hrp.Position).Magnitude
                if dist > 15 then
                    setState("บินไปหา NPC Lv." .. currentLevel .. ": " .. qdata.mob .. " (ระยะ " .. math.floor(dist) .. ")", "B")
                    tp(qdata.questCF)
                else
                    setState("รับเควส: StartQuest(" .. qdata.questName .. "," .. qdata.questNum .. ") [Lv." .. currentLevel .. "]", "Y")
                    pcall(function()
                        CommF:InvokeServer("StartQuest", qdata.questName, qdata.questNum)
                    end)

                    local waited = 0
                    while not UIQ.Visible and waited < 20 and AUTO_ON and _G.BF04_LOOP do
                        task.wait(0.1)
                        waited += 1
                        closeDialogue()
                    end

                    if UIQ.Visible then
                        setState("รับเควสสำเร็จ ✓ → บินไปจุดเกิดมอน", "G")
                        tp(qdata.mobCF)
                        task.wait(0.3)
                    end
                end

            -- 2. มีเควสแล้ว → ล็อกมอนตามเลเวลและตีจนตาย
            else
                closeDialogue()

                local enemies = Workspace:FindFirstChild("Enemies")
                local target = nil
                if enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        -- หาชื่อมอนที่ตรงกับเควส:
                        -- ต้องไม่เป็น Boss (เช่น The Gorilla King ไม่ใช่ Gorilla ธรรมดา)
                        local isMatch = string.find(v.Name, qdata.mob)
                        local isBoss = string.find(v.Name, "Boss") or string.find(v.Name, "King")
                        
                        -- ถ้าเควสไม่ใช่บอส แต่ตัวมอนเป็นบอส ให้ข้าม
                        if not string.find(qdata.mob, "King") and not string.find(qdata.mob, "Boss") and isBoss then
                            isMatch = false
                        end

                        if isMatch and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            target = v
                            break
                        end
                    end
                end

                if not target then
                    attackOff()
                    setState("หามอน " .. qdata.mob .. " ไม่เจอ (รอที่จุดเกิด)...", "Y")
                    tp(qdata.mobCF)
                    task.wait(0.5)
                    return
                end

                local v = target
                setState("ล็อกมอน: " .. v.Name .. " (HP " .. math.floor(v.Humanoid.Health) .. ") [Lv." .. currentLevel .. "]", "Y")

                equipWeapon()

                while v.Parent and v.Humanoid.Health > 0 and UIQ.Visible and AUTO_ON and _G.BF04_LOOP do
                    local mhrp = v:FindFirstChild("HumanoidRootPart")
                    if mhrp then
                        tp(mhrp.CFrame * CFrame.new(0, 0, 5))
                        aimCam(v)
                    end
                    fastAttack()
                    task.wait(0.15)
                end
                attackOff()

                if v.Parent and v.Humanoid.Health <= 0 then
                    setState("มอนตาย ✓ (HP 0)", "G")
                end
            end
        end)
    end
    attackOff()
    pcall(function() if _G.BF04_BV then _G.BF04_BV:Destroy() _G.BF04_BV = nil end end)
    log("=== STOP AUTO ===", "R")
end
