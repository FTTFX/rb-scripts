    AUTO_ON    = false,
}

local function refreshAuto()
    autoBtn.Text = "AUTO: " .. (CFG.AUTO_ON and "ON" or "OFF")
    autoBtn.BackgroundColor3 = CFG.AUTO_ON and Color3.fromRGB(180,40,40) or Color3.fromRGB(30,120,50)
end

-- ── Helper ────────────────────────────────────────────────────────
local function setStatus(txt)
    statusLbl.Text = string.format(
        "สถานะ: %s\nเควส: %s\nความคืบหน้า: %s\nเป้าหมาย: %s",
        txt, CFG.QUEST,
        (CFG.cur and CFG.need) and (CFG.cur.."/"..CFG.need) or "-",
        CFG.MOB_NAME
    )
end

local CommF = nil
local function getCommF()
    if CommF and CommF.Parent then return CommF end
    local remotes = RS:FindFirstChild("Remotes")
    if remotes then
        for _, r in ipairs(remotes:GetDescendants()) do
            if r.Name == "CommF_" and r:IsA("RemoteFunction") then
                CommF = r
                return CommF
            end
        end
    end
    return nil
end

-- อ่าน progress จาก QuestTitle.Title.Text เช่น "DEFEAT 5 BANDITS (2/5)"
local function readProgress()
    local main = pg:FindFirstChild("Main")
    local q = main and main:FindFirstChild("Quest")
    if not q then return nil, nil, nil end
    local ok, txt = pcall(function()
        return q.Container.QuestTitle.Title.Text
    end)
    if not ok or not txt then return nil, nil, nil end
    local cur, need = txt:match("%((%d+)/(%d+)%)")
    return tonumber(cur), tonumber(need), txt
end

-- หามอนใกล้สุด (ยังไม่ตาย)
local function findMob()
    local folder = Workspace:FindFirstChild(CFG.MOB_FOLDER)
    if not folder then return nil end
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local best, bestDist = nil, math.huge
    for _, m in ipairs(folder:GetChildren()) do
        if m:IsA("Model") and m.Name == CFG.MOB_NAME then
            local hum = m:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local mhrp = m:FindFirstChild("HumanoidRootPart")
                if mhrp then
                    if hrp then
                        local d = (mhrp.Position - hrp.Position).Magnitude
                        if d < bestDist then best, bestDist = m, d end
                    else
                        return m, 0
                    end
                end
            end
        end
    end
    return best, bestDist
end

-- เทเลพอร์ต (TP)
local function tp(cframe)
    local char = LP.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    hrp.CFrame = cframe
    return true
end

-- เทเล + หันไปหามอน (ยืนหลังมอนระยะตี)
local function aimAt(mob)
    local char = LP.Character
    local cam = Workspace.CurrentCamera
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local mhrp = mob:FindFirstChild("HumanoidRootPart")
    if not hrp or not mhrp then return end
    local dir = hrp.Position - mhrp.Position
    local flat = Vector3.new(dir.X, 0, dir.Z)
    if flat.Magnitude < 0.1 then flat = Vector3.new(0,0,-1) end
    hrp.CFrame = CFrame.lookAt(mhrp.Position + flat.Unit * 3 + Vector3.new(0,0.5,0), mhrp.Position)
    if cam then
        cam.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0,5,0), mhrp.Position)
    end
end

-- M1 (VirtualUser — มาตรฐาน Blox Fruits)
local function attackOn()
    pcall(function() VU:CaptureController() end)
    pcall(function() VU:Button1Down() end)
end
local function attackOff()
    pcall(function() VU:Button1Up() end)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    local down = false
    while CFG.AUTO_ON and _G.BF04_LOOP do
        task.wait(0.1)

        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then
            setStatus("รอตัวละคร spawn")
            task.wait(1)
            continue
        end

        local cur, need, txt = readProgress()
        CFG.cur, CFG.need = cur, need

        local qVisible = false
        pcall(function()
            local main = pg:FindFirstChild("Main")
            local q = main and main:FindFirstChild("Quest")
            qVisible = q and q.Visible
        end)

        if not cur then
            -- ไม่มีเควส
            if not qVisible then
                -- ไป NPC ก่อน
                local d = (hrp.Position - CFG.QUEST_CFRAME.Position).Magnitude
                if d > 12 then
                    setStatus("ไปหา NPC (ระยะ " .. math.floor(d) .. ")")
                    if down then attackOff() down = false end
                    tp(CFG.QUEST_CFRAME)
                    task.wait(1)
                else
                    setStatus("รับเควส...")
                    local c = getCommF()
                    if c then
                        local ok = pcall(function() c:InvokeServer("StartQuest", CFG.QUEST, CFG.LEVEL) end)
                        setStatus(ok and "รับเควสแล้ว (รอ UI)" or "รับเควสไม่สำเร็จ")
                    else
                        setStatus("หา CommF_ ไม่เจอ")
                    end
                    task.wait(1.5)
                end
            else
                -- มี UI แต่ parse ไม่ได้ → รอ
                setStatus("รออ่านเควส...")
                task.wait(1)
            end
        elseif cur >= need then
            -- เควสเสร็จ
            setStatus("เควสเสร็จ! ยกเลิก + รับใหม่")
            if down then attackOff() down = false end
            local c = getCommF()
            if c then
                pcall(function() c:InvokeServer("AbandonQuest") end)
                task.wait(0.8)
                pcall(function() c:InvokeServer("StartQuest", CFG.QUEST, CFG.LEVEL) end)
            end
            task.wait(1.5)
        else
            -- มีเควส → ตีมอน
            local mob, dist = findMob()
            if mob then
                if dist > 20 then
                    setStatus("เทเลไปมอน (ระยะ " .. math.floor(dist) .. ")")
                    if down then attackOff() down = false end
                    tp(CFG.MON_CFRAME)
                    task.wait(0.5)
                else
                    setStatus("ตี " .. CFG.MOB_NAME .. " (" .. cur .. "/" .. need .. ")")
                    aimAt(mob)
                    if not down then attackOn() down = true end
                end
            else
                setStatus("หามอนไม่เจอ (ไปจุดรวมมอน)")
                if down then attackOff() down = false end
                tp(CFG.MON_CFRAME)
                task.wait(1)
            end
        end
    end
    if down then attackOff() end
    setStatus("หยุดแล้ว")
end

autoBtn.MouseButton1Click:Connect(function()
    CFG.AUTO_ON = not CFG.AUTO_ON
    refreshAuto()
    if CFG.AUTO_ON then
        _G.BF04_LOOP = true
        task.spawn(mainLoop)
        setStatus("เริ่มทำงาน...")
    else
        _G.BF04_LOOP = nil
        pcall(function() VU:Button1Up() end)
        setStatus("หยุดแล้ว")
    end
end)

setStatus("พร้อม — กด AUTO เพื่อเริ่ม")
refreshAuto()
warn("[BF04] Auto Quest v1.1 loaded")
