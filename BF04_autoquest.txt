-- ── Log (state-based — log แค่ตอนเปลี่ยน) ──────────────────────────
local logLines = {}
local logOrder = 0
local lastState = ""
local logColors = {
    G = Color3.fromRGB(80,255,120), Y = Color3.fromRGB(255,210,0),
    R = Color3.fromRGB(255,80,80), B = Color3.fromRGB(140,160,255),
    S = Color3.fromRGB(140,140,140),
}
local function log(txt, col)
    local c = logColors[col] or logColors.S
    logOrder += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.BackgroundTransparency = 1; l.Font = Enum.Font.Code
    l.TextSize = 11; l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = c; l.Text = txt; l.LayoutOrder = logOrder; l.Parent = scroll
    table.insert(logLines, txt)
    scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
end
local function setState(txt, col)
    if txt ~= lastState then
        lastState = txt
        log(txt, col)
    end
end
copyBtn.MouseButton1Click:Connect(function()
    local clip = setclipboard or toclipboard or write_clipboard or (getgenv and getgenv().setclipboard)
    local ok = clip and pcall(clip, table.concat(logLines, "\n"))
    copyBtn.Text = ok and "Copied!" or "ไม่รองรับ"
    task.delay(2, function() if copyBtn.Parent then copyBtn.Text = "Copy Log" end end)
end)

-- ── Helper ────────────────────────────────────────────────────────
local function getCommF()
    local remotes = RS:FindFirstChild("Remotes")
    if remotes then
        for _, r in ipairs(remotes:GetDescendants()) do
            if r.Name == "CommF_" and r:IsA("RemoteFunction") then return r end
        end
    end
    return nil
end

local function getChar()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if char and hrp and hum then return char, hrp, hum end
    return nil
end

-- เทเลพอร์ตตรง ๆ (เร็ว เชื่อถือได้) + BodyVelocity กันกระเด็น
local function tp(cframe)
    local char, hrp = getChar()
    if not char then return false end
    hrp.CFrame = cframe
    -- กันกระเด็น
    if not _G.BF04_BV or not _G.BF04_BV.Parent then
        local bv = Instance.new("BodyVelocity")
        bv.Name = "BF04_Noclip"
        bv.Parent = hrp
        bv.MaxForce = Vector3.new(9e99, 9e99, 9e99)
        bv.Velocity = Vector3.new(0, 0, 0)
        _G.BF04_BV = bv
    end
    return true
end

local function clickM1()
    pcall(function() VU:CaptureController() end)
    pcall(function() VU:Button1Down(Vector2.new(1280, 672)) end)
    task.wait(0.05)
    pcall(function() VU:Button1Up(Vector2.new(1280, 672)) end)
end

local function readProgress()
    local main = pg:FindFirstChild("Main")
    local q = main and main:FindFirstChild("Quest")
    if not q then return nil, nil end
    local ok, txt = pcall(function() return q.Container.QuestTitle.Title.Text end)
    if not ok or not txt then return nil, nil end
    local cur, need = txt:match("%((%d+)/(%d+)%)")
    return tonumber(cur), tonumber(need)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    log("=== START AUTO ===", "Y")
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ! (รอเกมโหลดแล้วรันใหม่)", "R") return end
    log("CommF_ เจอ ✓", "G")

    -- รอตัวละคร (กันเงียบพัง)
    while CFG.AUTO_ON and _G.BF04_LOOP and not getChar() do
        setState("รอตัวละคร spawn...", "S")
        task.wait(1)
    end

    while CFG.AUTO_ON and _G.BF04_LOOP do
        task.wait(0.1)
        pcall(function()
            local char, hrp = getChar()
            if not char then setState("รอตัวละคร...", "S") return end

            local UIQ = pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Quest")
            if not UIQ then setState("หา Quest UI ไม่เจอ (ยังไม่มีเควส UI)", "R") return end

            local titleTxt = ""
            pcall(function() titleTxt = UIQ.Container.QuestTitle.Title.Text end)

            -- เควสค้างไม่ใช่ Bandit → ทิ้ง
            if UIQ.Visible and titleTxt ~= "" and not string.find(titleTxt, CFG.MOB_NAME) then
                setState("เควสไม่ใช่ Bandit → AbandonQuest", "Y")
                CommF:InvokeServer("AbandonQuest")
                task.wait(0.5)
                return
            end

            if not UIQ.Visible then
                -- ไม่มีเควส → เทเลไป NPC → รับ
                local dist = (CFG.QUEST_CFRAME.Position - hrp.Position).Magnitude
                if dist > 15 then
                    setState("เทเลไป NPC (ระยะ " .. math.floor(dist) .. ")", "B")
                    tp(CFG.QUEST_CFRAME)
                else
                    setState("รับเควส: StartQuest(" .. CFG.QUEST_NAME .. "," .. CFG.QUEST_NUM .. ")", "Y")
                    local ok = pcall(function()
                        CommF:InvokeServer("StartQuest", CFG.QUEST_NAME, CFG.QUEST_NUM)
                    end)
                    if ok then setState("รับเควสแล้ว (รอ UI ขึ้น)", "G") end
                    task.wait(1)
                end
            else
                -- มีเควส → หามอน + ล็อกตัวเดียวจนตาย
                local enemies = Workspace:FindFirstChild("Enemies")
                local target = nil
                if enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        if v.Name == CFG.MOB_NAME and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            target = v
                            break
                        end
                    end
                end

                if not target then
                    setState("หามอนไม่เจอ (รอ spawn)...", "R")
                    return
                end

                local v = target
                local mhrp = v:FindFirstChild("HumanoidRootPart")
                setState("ล็อกมอน: " .. v.Name .. " (HP " .. math.floor(v.Humanoid.Health) .. ")", "Y")
                while v.Parent and v.Humanoid.Health > 0 and UIQ.Visible and CFG.AUTO_ON and _G.BF04_LOOP do
                    if v:FindFirstChild("HumanoidRootPart") then
                        tp(v.HumanoidRootPart.CFrame * CFrame.new(0, 0, 5))
                    end
                    clickM1()
                    task.wait(0.15)
                end
                if v.Parent and v.Humanoid.Health <= 0 then
                    setState("มอนตาย ✓ (HP 0)", "G")
                end
            end
        end)
    end
    pcall(function() VU:Button1Up() end)
    pcall(function() if _G.BF04_BV then _G.BF04_BV:Destroy() _G.BF04_BV = nil end end)
    log("=== STOP AUTO ===", "R")
end

autoBtn.MouseButton1Click:Connect(function()
    CFG.AUTO_ON = not CFG.AUTO_ON
    autoBtn.Text = "AUTO: " .. (CFG.AUTO_ON and "ON" or "OFF")
    autoBtn.BackgroundColor3 = CFG.AUTO_ON and Color3.fromRGB(180,40,40) or Color3.fromRGB(30,120,50)
    if CFG.AUTO_ON then
        _G.BF04_LOOP = true
        lastState = ""
        task.spawn(mainLoop)
    else
        _G.BF04_LOOP = nil
        pcall(function() VU:Button1Up() end)
        log("หยุดโดยผู้ใช้", "R")
    end
end)
