-- ── Helper ────────────────────────────────────────────────────────
local function findRemote(name)
    local remotes = RS:FindFirstChild("Remotes")
    if remotes then
        for _, r in ipairs(remotes:GetDescendants()) do
            if r.Name == name and r:IsA("RemoteFunction") then return r end
        end
    end
    return nil
end
local CommF = findRemote("CommF_")

local function setStatus(txt)
    CFG.STATUS = txt
    statusLbl.Text = string.format(
        "สถานะ: %s\nเควส: %s\nความคืบหน้า: %s\nเป้าหมาย: %s",
        txt,
        CFG.QUEST,
        CFG.cur and (CFG.cur.."/"..CFG.need) or "-",
        CFG.MOB_NAME
    )
end

-- รับเควสผ่าน CommF_
local function startQuest()
    if not CommF then return false end
    local ok, err = pcall(function()
        CommF:InvokeServer("StartQuest", CFG.QUEST, CFG.LEVEL)
    end)
    return ok
end
local function abandonQuest()
    if not CommF then return false end
    local ok, err = pcall(function()
        CommF:InvokeServer("AbandonQuest")
    end)
    return ok
end

-- อ่าน progress จาก UI "DEFEAT 5 BANDITS (n/5)"
local function readProgress()
    local main = pg:FindFirstChild("Main")
    local q = main and main:FindFirstChild("Quest")
    if not q then return nil end
    local function walk(node, depth)
        depth = depth or 0
        if depth > 12 then return nil end
        for _, c in ipairs(node:GetChildren()) do
            if c:IsA("TextLabel") or c:IsA("TextButton") then
                local cur, need = tostring(c.Text):match("%((%d+)/(%d+)%)")
                if cur and need then
                    return tonumber(cur), tonumber(need)
                end
            end
            local r = walk(c, depth + 1)
            if r then return r end
        end
        return nil
    end
    return walk(q)
end

-- หามอนที่ใกล้สุด (ยังไม่ตาย)
local function findMob()
    local folder = Workspace:FindFirstChild(CFG.MOB_FOLDER)
    if not folder then return nil end
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local best, bestDist = nil, math.huge
    for _, m in ipairs(folder:GetChildren()) do
        if m:IsA("Model") and m.Name == CFG.MOB_NAME then
            local hum = m:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local mhrp = m:FindFirstChild("HumanoidRootPart")
                if mhrp then
                    if hrp then
                        local d = (mhrp.Position - hrp.Position).Magnitude
                        if d < bestDist then best, bestDist = m, d end
                    else
                        return m
                    end
                end
            end
        end
    end
    return best, bestDist
end

-- เทเลพอร์ต + หันกล้อง/ตัวละครไปหามอน
local function aimAt(mob)
    local char = LP.Character
    local cam = Workspace.CurrentCamera
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local mhrp = mob:FindFirstChild("HumanoidRootPart")
    if not hrp or not mhrp then return end
    -- ยืนหลังมอน 3 studs (ระยะตี) + หันเข้าหา
    local dir = (hrp.Position - mhrp.Position)
    local flat = Vector3.new(dir.X, 0, dir.Z)
    if flat.Magnitude < 0.1 then flat = Vector3.new(0,0,-1) end
    local standPos = mhrp.Position + flat.Unit * 3
    hrp.CFrame = CFrame.lookAt(standPos + Vector3.new(0,0.5,0), mhrp.Position)
    if cam then
        cam.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0,5,0), mhrp.Position)
    end
end

-- M1 คลิก (client จะ fire remote โจมตีให้เอง)
local function attack()
    pcall(function()
        VIM:SendMouseButtonEvent(0, 0, 0, true, game, 1)
        task.wait(0.06)
        VIM:SendMouseButtonEvent(0, 0, 0, false, game, 1)
    end)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    while CFG.AUTO_ON and _G.BF04_LOOP do
        task.wait(0.05)

        if not LP.Character or not LP.Character:FindFirstChild("HumanoidRootPart") then
            setStatus("รอตัวละคร spawn")
            task.wait(1)
            continue
        end

        local cur, need = readProgress()
        CFG.cur, CFG.need = cur, need

        if not cur then
            -- ไม่มีเควส → รับเควส
            setStatus("รับเควส...")
            if startQuest() then
                setStatus("รับเควสสำเร็จ")
            else
                setStatus("รับเควสไม่สำเร็จ (ไม่มี CommF_?)")
            end
            task.wait(1)
        elseif cur >= need then
            -- เควสเสร็จ → รับใหม่ (ถ้าจำเป็น abandon ก่อน)
            setStatus("เควสเสร็จ! รับใหม่...")
            abandonQuest()
            task.wait(0.5)
            startQuest()
            task.wait(1)
        else
            -- มีเควส → หามอน ตี
            local mob, dist = findMob()
            if mob then
                setStatus("ตี " .. CFG.MOB_NAME .. " (ระยะ " .. math.floor(dist) .. ")")
                aimAt(mob)
                attack()
                task.wait(0.1)
            else
                setStatus("หามอนไม่เจอ (รอ spawn)")
                task.wait(1)
            end
        end
    end
    setStatus("หยุดแล้ว")
end

autoBtn.MouseButton1Click:Connect(function()
    if CFG.AUTO_ON then
        _G.BF04_LOOP = true
        task.spawn(mainLoop)
        setStatus("เริ่มทำงาน...")
    else
        _G.BF04_LOOP = nil
        setStatus("หยุดแล้ว")
    end
end)

setStatus("พร้อม — กด AUTO เพื่อเริ่ม (รัน BF03 ก่อนเพื่อให้แน่ใจว่า CommF_ มี)")
refreshAuto()
warn("[BF04] Auto Quest v1.0 loaded | CommF_: " .. tostring(CommF ~= nil and "เจอ ✅" or "ไม่เจอ ❌"))
