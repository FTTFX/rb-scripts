-- ── Log ────────────────────────────────────────────────────────────
local logLines = {}
local logOrder = 0
local lastState = ""
local logColors = {
    G = Color3.fromRGB(80,255,120), Y = Color3.fromRGB(255,210,0),
    R = Color3.fromRGB(255,80,80), B = Color3.fromRGB(140,160,255),
    S = Color3.fromRGB(140,140,140),
}
-- เวลาเริ่มสคริปต์ ใช้คำนวณ [เวลานาฬิกา +วินาทีที่ผ่านไป] หน้าทุกบรรทัด log
local SCRIPT_T0 = os.clock()
local lastLogT = SCRIPT_T0
local function log(txt, col)
    local c = logColors[col] or logColors.S
    local now = os.clock()
    txt = string.format("[%s +%.2fs] %s", os.date("%H:%M:%S"), now - lastLogT, txt)
    lastLogT = now
    logOrder += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.BackgroundTransparency = 1; l.Font = Enum.Font.Code
    l.TextSize = 11; l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = c; l.Text = txt; l.LayoutOrder = logOrder; l.Parent = scroll
    table.insert(logLines, txt)
    scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
end
local function setState(txt, col)
    if txt ~= lastState then
        lastState = txt
        log(txt, col)
    end
end

-- ── ระบบเก็บผลปีศาจ (Devil Fruit Finder) ──────────────────────────
local FRUIT_MODE = false
-- ธงหยุด AUTO ชั่วคราวระหว่างบินไปเก็บผล (กันสองระบบดึงตัวละครสวนทางกัน = กระตุก)
local PAUSED_FOR_FRUIT = false

local function findNearestFruit()
    local char, hrp = getChar()
    if not char or not hrp then return nil end
    local best, bestDist = nil, math.huge
    -- ผลที่ตกบนแมพ = Tool ที่ลอยอยู่ใน Workspace โดยตรง และมี Handle
    -- (ไม่นับ Tool ใน Character/Backpack ของผู้เล่น = ผลที่คนอื่นถืออยู่)
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Tool") and obj:FindFirstChild("Handle") then
            local handle = obj.Handle
            local d = (handle.Position - hrp.Position).Magnitude
            if d < bestDist and d < 8000 then
                best = handle
                bestDist = d
            end
        end
    end
    return best, bestDist
end

fruitBtn.MouseButton1Click:Connect(function()
    FRUIT_MODE = not FRUIT_MODE
    fruitBtn.Text = "FRUIT: " .. (FRUIT_MODE and "ON..." or "OFF")
    fruitBtn.BackgroundColor3 = FRUIT_MODE and Color3.fromRGB(140,60,200) or Color3.fromRGB(60,30,90)
    if FRUIT_MODE then
        task.spawn(function()
            while FRUIT_MODE do
                local gotOne = false
                pcall(function()
                    local fruit, dist = findNearestFruit()
                    if fruit and fruit.Parent then
                        gotOne = true
                        PAUSED_FOR_FRUIT = true -- พัก AUTO ระหว่างบินไปเก็บ
                        setState("เจอผล: " .. fruit.Name .. " (ระยะ " .. math.floor(dist or 0) .. ") → บินไปเก็บ (AUTO พักชั่วคราว)", "Y")
                        if dist and dist > 25 then
                            tp(CFrame.new(fruit.Position))
                        else
                            local hrp = select(2, getChar())
                            if hrp then hrp.CFrame = fruit.CFrame end
                        end
                        task.wait(0.6)
                        -- เช็คว่าเก็บสำเร็จ (ผลหายไปจากแมพ) → ปลดพัก ให้ AUTO ทำงานต่อ
                        if not fruit.Parent or not findNearestFruit() then
                            setState("เก็บผลสำเร็จ ✓ → AUTO กลับมาทำงาน", "G")
                        end
                        PAUSED_FOR_FRUIT = false
                    else
                        PAUSED_FOR_FRUIT = false
                        setState("ไม่มีผลเกิดบนแผนที่ (รอ 5 วิ...)", "S")
                        task.wait(5)
                    end
                end)
                if not gotOne then task.wait(0.5) end
                task.wait(0.2)
            end
            PAUSED_FOR_FRUIT = false
            setState("FRUIT OFF", "S")
        end)
    else
        PAUSED_FOR_FRUIT = false
    end
end)

-- ล้างสถานะทั้งหมดที่สคริปต์ไปยุ่งกับตัวละคร แล้วคืนการควบคุมให้ผู้เล่น
-- ต้องล้างให้ครบทุกอย่าง ไม่ใช่แค่ BodyVelocity ตัวเดียว
-- (tween ที่ค้าง / Anchored / PlatformStand / ControlModule ที่ถูกปิด ทำให้ขยับไม่ได้ทั้งนั้น)
local BLOCKING_CLASSES = {
    BodyVelocity = true, BodyGyro = true, BodyPosition = true,
    BodyThrust = true, BodyAngularVelocity = true,
    AlignPosition = true, AlignOrientation = true,
    LinearVelocity = true, AngularVelocity = true, VectorForce = true,
}
local function releaseControl()
    pcall(function() if currentTween then currentTween:Cancel() currentTween = nil end end)

    pcall(function()
        if _G.BF04_BV then _G.BF04_BV:Destroy() _G.BF04_BV = nil end
    end)

    local char = LP.Character
    if not char then return end

    -- ลบตัวบังคับฟิสิกส์ทุกชิ้นที่ค้างอยู่ในตัวละคร
    pcall(function()
        for _, obj in ipairs(char:GetDescendants()) do
            if BLOCKING_CLASSES[obj.ClassName] then
                obj:Destroy()
            end
        end
    end)

    -- ปลด Anchored และคืน CanCollide ให้ทุกชิ้นส่วน
    pcall(function()
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Anchored = false
                if part.Name ~= "HumanoidRootPart" then part.CanCollide = true end
            end
        end
    end)

    pcall(function()
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.Sit = false
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            if hum.WalkSpeed <= 0 then hum.WalkSpeed = 16 end
            if hum.JumpPower <= 0 then hum.JumpPower = 50 end
            hum.AutoRotate = true
        end
    end)

    -- เปิดระบบควบคุมของเกมกลับ (บางกรณีถูกปิดไว้ตอนวาร์ป)
    pcall(function()
        local pm = LP:FindFirstChild("PlayerScripts") and LP.PlayerScripts:FindFirstChild("PlayerModule")
        if pm then
            local controls = require(pm):GetControls()
            controls:Enable()
        end
    end)
end

-- ระบบ Smart Travel (บินสูงข้ามเกาะ)
-- ⚠️ กัน Anti-Cheat: ระยะประชิดวาร์ปได้แค่ < 25 studs (เกินนี้โดนเด้งกลับ)
--    บินด้วยความเร็ว 150 studs/sec (เกมรับได้ ไม่ trigger anti-teleport)
local FLY_SPEED = 380
-- ระบบไหนก็ได้ที่กำลังทำงาน = อนุญาตให้บินต่อ
-- (เดิมเช็ค AUTO_ON เท่านั้น พอ AUTO SABER ทำงานลูปบินจะหลุดกลางทาง = ค้างบนฟ้า)
local function moveOK()
    return (AUTO_ON and _G.BF04_LOOP) or (SABER_ON and _G.BF04_SABER) or FRUIT_MODE
end
-- บินแบบขยับทีละเฟรม แทน Tween ยาวๆ
-- Tween ตั้งค่า CFrame สวนกับ physics ของเกม ทำให้ภาพกระตุกและถูกดึงกลับเป็นช่วงๆ
-- พุ่งเป็นช่วง: ขยับ BURST_MOVE วิ แล้วหยุดนิ่ง BURST_REST วิ
-- การขยับต่อเนื่องยาวๆ คือสิ่งที่ anti-cheat จับ การหยุดสั้นๆ ให้เซิร์ฟเวอร์
-- ยืนยันตำแหน่งก่อนพุ่งต่อ ทำให้ไม่ถูกดึงกลับ
local BURST_MOVE = 1.0
local BURST_REST = 0.2

-- รัศมีค้นหามอน และระยะที่ยังบินตรงได้ (ไม่ต้องไต่ขึ้นฟ้า)
SEARCH_R = 3000
local DIRECT_FLY_R = 3000

local function flySmooth(destCF, speed, report)
    local total
    local nextReport = os.clock() + 5
    local burstT0 = os.clock()
    local deadline
    while moveOK() do
        local char, hrp = getChar()
        if not char or not hrp then return false end
        local delta = destCF.Position - hrp.Position
        local left = delta.Magnitude
        total = total or left
        -- ถ้าถูกดึงกลับตลอด ระยะจะไม่ลด วนไม่จบ ต้องมีเวลาสูงสุด
        deadline = deadline or (os.clock() + (total / speed) * 2.5 + 5)
        if left <= 8 then break end
        if os.clock() > deadline then
            log(string.format("บินไม่ถึง (เหลือ %.0f studs) → เริ่มรอบใหม่", left), "Y")
            return false
        end

        if os.clock() - burstT0 >= BURST_MOVE then
            local restEnd = os.clock() + BURST_REST
            while os.clock() < restEnd do
                local _, h = getChar()
                if h then h.Velocity = Vector3.new(0, 0, 0) end
                RunService.Heartbeat:Wait()
            end
            burstT0 = os.clock()
        end

        local dt = RunService.Heartbeat:Wait()
        local step = math.min(speed * dt, left)
        hrp.CFrame = CFrame.new(hrp.Position + delta.Unit * step)
        hrp.Velocity = Vector3.new(0, 0, 0)
        if report and total > 1500 and os.clock() >= nextReport then
            nextReport = os.clock() + 5
            log(string.format("  ...เหลือ %.0f studs", left), "S")
        end
    end
    return true
end

local function toTarget(targetCF)
    local char, hrp, hum = getChar()
    if not char then return false end
    
    applyBodyVelocity(hrp)
    
    local dist = (targetCF.Position - hrp.Position).Magnitude
    
    -- ระยะประชิด (< 25 studs) → วาร์ปตรง (ไม่เกินเพดานที่เกมยอมรับ)
    if dist < 25 then
        stopTween()
        hrp.CFrame = targetCF
        return true
    end
    
    -- ระยะกลาง (< 650 studs, อยู่เกาะเดียวกัน) → พุ่งตรงไปเลย ไม่ต้องไต่ขึ้นฟ้า
    -- ตัดขั้นตอนไต่ขึ้น-ลงออก ประหยัดเวลาต่อมอนได้หลายวินาที
    if dist < DIRECT_FLY_R then
        stopTween()
        flySmooth(targetCF, 420, dist > 1500)
        return true
    end

    -- ระยะไกล (ข้ามเกาะ) → บินขึ้นสูงก่อน (ระยะสั้นๆ ไม่วาร์ป)
    stopTween()
    
    local highY = math.max(hrp.Position.Y, targetCF.Position.Y) + 120
    if targetCF.Position.Y > 300 then highY = targetCF.Position.Y + 60 end
    
    -- ไต่ระดับขึ้นเป็นช่วงๆ ไม่กระโดดสูงทีเดียว (กัน anti-teleport แนวตั้ง)
    local stepY = 40
    local curY = hrp.Position.Y
    while curY < highY - 5 do
        curY = math.min(curY + stepY, highY)
        hrp.CFrame = CFrame.new(hrp.Position.X, curY, hrp.Position.Z)
        task.wait(0.08)
    end
    
    local skyStartCF = CFrame.new(hrp.Position.X, highY, hrp.Position.Z)
    local skyEndCF   = CFrame.new(targetCF.Position.X, highY, targetCF.Position.Z)
    
    local flyDist = (skyEndCF.Position - skyStartCF.Position).Magnitude

    -- บินไกลใช้เวลาหลายสิบวินาที ถ้าไม่รายงานความคืบหน้าจะดูเหมือนสคริปต์ค้าง
    if flyDist > 1500 then
        log(string.format("บินไกล %.0f studs (~%.0f วิ)", flyDist, flyDist / FLY_SPEED), "B")
    end

    flySmooth(skyEndCF, FLY_SPEED, true)
    
    -- ไต่ระดับลงเป็นช่วงๆ เช่นกัน
    if moveOK() then
        local landY = hrp.Position.Y
        while landY > targetCF.Position.Y + 25 do
            landY = math.max(landY - stepY, targetCF.Position.Y + 25)
            hrp.CFrame = CFrame.new(targetCF.Position.X, landY, targetCF.Position.Z)
            task.wait(0.08)
        end
        hrp.CFrame = targetCF
    end
    
    stopTween()
    return true
end

function tp(cframe)
    return toTarget(cframe)
end

-- ของเควสที่ถือได้แต่ตีไม่ออก ต้องถอดก่อนสู้
-- (เดิมเช็คแค่ว่า "มี Tool ในมือ" พอถือ Relic อยู่ก็ไม่สลับอาวุธ = ตีไม่โดน)
local QUEST_ITEMS = { Relic = true, Cup = true, Torch = true }

local function equipWeapon()
    local char, hrp, hum = getChar()
    if not char or not hum then return false end

    local held = char:FindFirstChildOfClass("Tool")
    if held and not QUEST_ITEMS[held.Name] then return true end

    -- ถือของเควสอยู่ → เก็บเข้ากระเป๋าก่อน
    if held then
        pcall(function() hum:UnequipTools() end)
        task.wait(0.1)
    end

    local bp = LP:FindFirstChild("Backpack")
    if not bp then return false end

    local tool = bp:FindFirstChild("Combat")
    if not tool then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and not QUEST_ITEMS[t.Name] then tool = t break end
        end
    end
    if tool then
        pcall(function() hum:EquipTool(tool) end)
        return true
    end
    return false
end

-- 🧲 ระบบ Mob Magnet / รวมมอนสเตอร์รอบข้าง
-- ขยาย SimulationRadius เพื่อให้เราเป็นเจ้าของฟิสิกส์มอน (Network Ownership)
-- ถ้าไม่ทำเซาะตรงนี้ CFrame ที่เราตั้งจะไม่ replicate ไปเซิร์ฟ = มอนไม่ขยับจริง
local function boostSimulationRadius()
    pcall(function()
        -- วิธีของสคริปต์ใหญ่ (MUXHUB): setsimulationradius ทรงพลังกว่า sethiddenproperty เฉยๆ
        if setsimulationradius then
            setsimulationradius(math.huge, math.huge)
        end
        if sethiddenproperty then
            sethiddenproperty(LP, "MaximumSimulationRadius", math.huge)
            sethiddenproperty(LP, "SimulationRadius", math.huge)
            sethiddenproperty(Workspace, "PhysicsSteppedMethod", "Transient")
        end
    end)
end

-- รวมมอนสเตอร์ทุกตัวชื่อเดียวกัน "ทั่วทั้งแมพ" มาไว้ที่จุดเดียว (Bring All Mobs)
-- ไม่จำกัดระยะ → ตี AoE ได้ครั้งละ 5-8 ตัว
local function bringMobsToTarget(primaryTarget, mobName)
    if not primaryTarget or not primaryTarget.Parent then return end
    -- ยึดตำแหน่งตัวละครเราเป็นจุดรวมกอง (เดิมใช้ตำแหน่งมอนเป้าหมายซึ่งขยับหนีตลอด)
    local _, myHrp = getChar()
    local pHrp = myHrp or primaryTarget:FindFirstChild("HumanoidRootPart")
    if not pHrp then return end
    
    boostSimulationRadius()
    
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return end
    
    for _, otherMob in ipairs(enemies:GetChildren()) do
        if otherMob ~= primaryTarget and string.find(otherMob.Name, mobName) then
            local isBoss = string.find(otherMob.Name, "Boss") or string.find(otherMob.Name, "King")
            if not isBoss and otherMob:FindFirstChild("Humanoid") and otherMob:FindFirstChild("HumanoidRootPart") and otherMob.Humanoid.Health > 0 then
                local oHrp = otherMob.HumanoidRootPart
                pcall(function()
                    otherMob.Humanoid.PlatformStand = true
                    otherMob.Humanoid.Sit = true
                    oHrp.CanCollide = false
                    oHrp.Velocity = Vector3.new(0, 0, 0)
                    oHrp.Anchored = false
                    oHrp.CFrame = pHrp.CFrame * CFrame.new(
                        math.random(-6, 6), 0, math.random(-6, 6)
                    )
                end)
            end
        end
    end
end

-- ⚡ AttackNoCoolDown: รูปแบบที่พิสูจน์แล้วจากสคริปต์ที่ใช้งานได้จริง
--   RegisterAttack:FireServer(0) → (1) → (2) → (3)   [อาร์กิวเมนต์เดียว = เลขคอมโบ]
--   RegisterHit:FireServer(EnemyHead, OthersEnemies)
-- การยิงคอมโบครบ 0-3 คือกุญแจที่ทำให้ผ่านการเช็คคูลดาวน์ฝั่งเซิร์ฟเวอร์
local function directRemoteAttack(mobTarget, mobName)
    if not mobTarget or not mobTarget.Parent then return end
    
    local rAttack, rHit = getCombatRemotes()
    
    -- ล้างสถานะ stun
    pcall(function()
        local char = LP.Character
        if char then
            if char:FindFirstChild("Stun") then char.Stun.Value = 0 end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum.Sit = false end
        end
    end)
    
    -- ต้องถืออาวุธอยู่ในมือ ไม่งั้นเซิร์ฟเวอร์ปฏิเสธทุกกรณี
    local equipped = LP.Character and LP.Character:FindFirstChildOfClass("Tool")
    if not equipped then return end

    -- เก็บเป้าหมายทุกตัวในระยะ 100 studs ตามสคริปต์ต้นแบบ (ใช้ Head เป็นหลัก)
    local mainPart = nil
    local othersEnemies = {}
    local enemies = Workspace:FindFirstChild("Enemies")
    if enemies then
        for _, enemy in ipairs(enemies:GetChildren()) do
            local eHum = enemy:FindFirstChild("Humanoid")
            local eHead = enemy:FindFirstChild("Head") or enemy:FindFirstChild("HumanoidRootPart")
            if eHum and eHead and eHum.Health > 0 and enemy ~= LP.Character then
                if LP:DistanceFromCharacter(eHead.Position) < 100 then
                    table.insert(othersEnemies, { enemy, eHead })
                    mainPart = eHead
                end
            end
        end
    end
    if not mainPart or #othersEnemies == 0 then return end

    -- 1. ไล่คอมโบ 0-3 (อาร์กิวเมนต์เดียว) เพื่อรีเซ็ตคูลดาวน์ฝั่งเซิร์ฟเวอร์
    pcall(function()
        if rAttack then
            rAttack:FireServer(0)
            rAttack:FireServer(1)
            rAttack:FireServer(2)
            rAttack:FireServer(3)
        end
    end)

    -- 2. ยิง RegisterHit ส่งทั้งกองใน call เดียว
    pcall(function()
        if rHit then
            rHit:FireServer(mainPart, othersEnemies)
        end
    end)
end

-- ── Main Loop ─────────────────────────────────────────────────────
local function mainLoop()
    log("=== START AUTO v8.2 ===", "Y")
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ! (รอเกมโหลดแล้วรันใหม่)", "R") return end
    log("CommF_ เจอ ✓", "G")
    
    local rAttack, rHit = getCombatRemotes()
    if rAttack and rHit then
        log("Net Remotes (RegisterAttack / Hit) เจอ ✓", "G")
    else
        log("⚠️ หา Net Remotes ไม่เจอ (กำลังลองค้นหาใหม่)...", "R")
    end

    while AUTO_ON and _G.BF04_LOOP and not getChar() do
        setState("รอตัวละคร spawn...", "S")
        task.wait(1)
    end

    while AUTO_ON and _G.BF04_LOOP do
        task.wait(0.1)
        -- ⏸️ ถ้า FRUIT กำลังบินไปเก็บผล → AUTO หยุดรอ (ไม่ดึงตัวละครสวนทาง)
        if PAUSED_FOR_FRUIT then
            setState("AUTO พักชั่วคราว (FRUIT กำลังเก็บผล...)", "S")
            task.wait(0.3)
        else
        pcall(function()
            local char, hrp = getChar()
            if not char then setState("รอตัวละคร...", "S") return end

            local currentLevel = getMyLevel()
            local qdata = getQuestData(currentLevel)

            local UIQ = pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Quest")
            if not UIQ then setState("หา Quest UI ไม่เจอ", "R") return end

            local titleTxt = ""
            pcall(function() titleTxt = UIQ.Container.QuestTitle.Title.Text end)

            -- ถ้ามีเควสค้างอยู่แต่ไม่ตรงกับเลเวลปัจจุบัน หรือไม่ใช่เควสที่ต้องการ → ยกเลิกเพื่อรับเควสใหม่ที่ดีกว่า
            if UIQ.Visible and titleTxt ~= "" and not string.find(titleTxt, qdata.mob) then
                setState("เควสไม่ตรงกับเลเวล " .. currentLevel .. " (" .. qdata.mob .. ") → AbandonQuest", "Y")
                CommF:InvokeServer("AbandonQuest")
                task.wait(0.5)
                return
            end

            -- 1. ยังไม่มีเควส → บินไปหา NPC ตามเลเวล → รับเควส
            if not UIQ.Visible then
                local dist = (qdata.questCF.Position - hrp.Position).Magnitude
                if dist > 15 then
                    setState("บินไปหา NPC Lv." .. currentLevel .. ": " .. qdata.mob .. " (ระยะ " .. math.floor(dist) .. ")", "B")
                    tp(qdata.questCF)
                else
                    setState("รับเควส: StartQuest(" .. qdata.questName .. "," .. qdata.questNum .. ") [Lv." .. currentLevel .. "]", "Y")
                    local invokeOk, invokeResult = pcall(function()
                        return CommF:InvokeServer("StartQuest", qdata.questName, qdata.questNum)
                    end)
                    if not invokeOk then
                        log("❌ StartQuest error: " .. tostring(invokeResult), "R")
                    end

                    local waited = 0
                    while not UIQ.Visible and waited < 20 and AUTO_ON and _G.BF04_LOOP do
                        task.wait(0.1)
                        waited += 1
                        closeDialogue()
                    end

                    if UIQ.Visible then
                        setState("รับเควสสำเร็จ ✓ → บินไปจุดเกิดมอน", "G")
                        tp(qdata.mobCF)
                        task.wait(0.3)
                    else
                        log("❌ เซิร์ฟเวอร์ไม่รับเควส " .. qdata.questName
                            .. "," .. qdata.questNum .. " | ตอบ: " .. tostring(invokeResult), "R")
                        task.wait(1)
                    end
                end

            -- 2. มีเควสแล้ว → ล็อกมอนตามเลเวลและยิงดาเมจตรงๆ
            else
                closeDialogue()

                local enemies = Workspace:FindFirstChild("Enemies")
                local target = nil
                local bestDist = math.huge
                if enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        local isMatch = string.find(v.Name, qdata.mob)
                        local isBoss = string.find(v.Name, "Boss") or string.find(v.Name, "King")
                        
                        -- กรองบอสออกถ้าเควสไม่ใช่บอส
                        if not string.find(qdata.mob, "King") and not string.find(qdata.mob, "Boss") and isBoss then
                            isMatch = false
                        end

                        -- เลือกตัวที่ใกล้ที่สุด ไม่ใช่ตัวแรกที่เจอ (เดิมบินข้ามแมพ 300-650 studs เสียเวลา)
                        if isMatch and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            local d = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
                            if d < bestDist and d < SEARCH_R then
                                bestDist = d
                                target = v
                            end
                        end
                    end
                end

                -- ไม่มีตัวไหนในระยะค้นหา → หาใหม่แบบไม่จำกัดระยะ แล้วบินไปหา
                -- (จุดเกิดที่บันทึกไว้อาจไม่ตรง ยืนรอเฉยๆ จะไม่เจอมอนเลย)
                if not target and enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        local isBoss = string.find(v.Name, "Boss") or string.find(v.Name, "King")
                        local isMatch = string.find(v.Name, qdata.mob) and not
                            (isBoss and not string.find(qdata.mob, "King") and not string.find(qdata.mob, "Boss"))
                        if isMatch and v:FindFirstChild("Humanoid")
                           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            local d = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
                            if d < bestDist then bestDist = d; target = v end
                        end
                    end
                    if target then
                        setState("เจอ " .. qdata.mob .. " ไกล " .. math.floor(bestDist) .. " → บินไปหา", "B")
                        tp(CFrame.new(target.HumanoidRootPart.Position + Vector3.new(0, 20, 0)))
                    end
                end

                -- ไม่มีตัวหลักเลย → ตีตัวอะไรก็ได้ที่อยู่ในระยะไปก่อน (รวมบอส)
                -- ไม่เสียเวลายืนรอ respawn เฉยๆ พอตัวหลักเกิดจะกลับไปตีตัวหลักเอง
                local fillerMode = false
                if not target and enemies then
                    for _, v in ipairs(enemies:GetChildren()) do
                        if v:FindFirstChild("Humanoid") and v:FindFirstChild("HumanoidRootPart")
                           and v.Humanoid.Health > 0 then
                            local d = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
                            if d < bestDist and d < SEARCH_R then
                                bestDist = d
                                target = v
                            end
                        end
                    end
                    if target then
                        fillerMode = true
                        setState("ไม่มี " .. qdata.mob .. " → ตีตัวรอง: " .. target.Name
                            .. " (ห่าง " .. math.floor(bestDist) .. ")", "S")
                    end
                end

                if not target then
                    -- ไม่มีมอนใน Workspace เลย = ตายหมดกำลังรอ respawn
                    -- ตำแหน่งมอนตัวล่าสุดที่เจอ = จุดเกิดจริง เชื่อถือได้กว่าพิกัดที่ฝังไว้
                    -- (พิกัดฝังไว้บางเควสตกทะเล ตระเวนวนเปล่าเสียเวลาหลายวินาที)
                    local sameMob = (LAST_MOB_NAME == qdata.mob)
                    local anchor = (sameMob and LAST_MOB_POS) or qdata.mobCF.Position
                    if sameMob and LAST_MOB_POS then
                        setState("มอนตายหมด → ลอยรอ respawn ที่จุดเกิดจริง", "S")
                        tp(CFrame.new(anchor + Vector3.new(0, 20, 0)))
                    else
                        PATROL_I = (PATROL_I or 0) + 1
                        -- แมพนี้มอนจะโหลดเฉพาะเมื่อเราอยู่ใกล้ → ตระเวนเป็นก้นหอย
                        -- วนใกล้ก่อนแล้วขยายวงออก และบินต่ำ ให้ระบบ streaming โหลดมอนขึ้นมา
                        local ang = PATROL_I * 1.2
                        local r = 60 + (PATROL_I % 6) * 50
                        local off = Vector3.new(math.cos(ang) * r, 15, math.sin(ang) * r)
                        setState("ยังไม่รู้จุดเกิดจริงของ " .. qdata.mob .. " → ตระเวนหา (" .. PATROL_I .. ")", "Y")
                        tp(CFrame.new(anchor + off))
                    end
                    task.wait(0.5)
                    return
                end

                local v = target
                -- ลืมจุดเกิดเก่าเมื่อเปลี่ยนชนิดมอน (เลเวลขึ้น = เควสใหม่ = เกาะใหม่)
                if LAST_MOB_NAME ~= qdata.mob then
                    LAST_MOB_NAME = qdata.mob
                    PATROL_I = 0
                end
                -- จำจุดเกิดเฉพาะตัวหลัก ตัวรองอยู่คนละจุด จะทำให้จุดเกิดเพี้ยน
                if not fillerMode then
                    LAST_MOB_POS = v.HumanoidRootPart.Position
                end
                setState("ล็อกมอน: " .. v.Name .. " (HP " .. math.floor(v.Humanoid.Health)
                    .. " | ห่าง " .. math.floor(bestDist) .. ")"
                    .. (MAGNET_ON and " — Magnet + Remote" or " — ตีระยะไกล (ไม่วาป)"), "Y")

                equipWeapon()

                local killT0 = os.clock()
                local hitCount = 0
                -- ระยะตอนล็อกเป้า ใช้เทียบว่ามอน "กระเด็น" (ถูกย้ายกลับจุดเกิด) จริงหรือแค่อยู่ไกล
                local lockDist = bestDist
                local nextMainCheck = os.clock() + 1
                while v.Parent and v.Humanoid.Health > 0 and UIQ.Visible and AUTO_ON and _G.BF04_LOOP do
                    -- ตายแล้วรอเกิดใหม่ แล้วค่อยเริ่มรอบใหม่ (ไม่ถอย ไม่หนี)
                    local myChar, _, myHum = getChar()
                    if not myChar or (myHum and myHum.Health <= 0) then
                        setState("ตาย → รอเกิดใหม่แล้วกลับไปตีต่อ", "R")
                        LP.CharacterAdded:Wait()
                        task.wait(3)
                        break
                    end

                    -- ตีตัวรองอยู่ → คอยดูว่าตัวหลักเกิดแล้วหรือยัง เกิดแล้วสลับกลับทันที
                    if fillerMode and os.clock() >= nextMainCheck then
                        nextMainCheck = os.clock() + 1
                        local found = false
                        for _, e in ipairs(enemies:GetChildren()) do
                            if string.find(e.Name, qdata.mob) and e:FindFirstChild("Humanoid")
                               and e.Humanoid.Health > 0 then
                                found = true
                                break
                            end
                        end
                        if found then
                            setState(qdata.mob .. " เกิดแล้ว → สลับกลับไปตีตัวหลัก", "G")
                            break
                        end
                    end

                    local mhrp = v:FindFirstChild("HumanoidRootPart")
                    if mhrp then
                        local char, hrp = getChar()
                        if hrp then
                            applyBodyVelocity(hrp)
                            -- ❌ ไม่วาปเข้าหามอนทุกเฟรมแล้ว (โดน anti-cheat)
                            -- RegisterHit เช็คระยะที่เซิร์ฟเวอร์ 100 studs → ยืนอยู่ไกลๆ ก็ตีโดน
                            -- ขยับเข้าไปแค่ตอนหลุดระยะจริงๆ และใช้ tween ช้าๆ ครั้งเดียว
                            local dist = (mhrp.Position - hrp.Position).Magnitude
                            -- ทิ้งเป้าเฉพาะตอน "ระยะพุ่งขึ้นกะทันหัน" = ศพถูกย้ายกลับจุดเกิด
                            -- ถ้าแค่อยู่ไกลตั้งแต่แรก ให้บินไปหาปกติ (เดิมเด้งออกทันทีจนวนไม่จบ)
                            if dist > lockDist + 150 then
                                setState("เป้าหมายกระเด็นไป " .. math.floor(dist) .. " → เปลี่ยนเป้าใกล้สุด", "Y")
                                break
                            end
                            -- เกาะติดที่ 25 studs: ยิง 283 ครั้งที่ระยะ 70 ได้แค่ 158 dmg/s
                            -- แต่ยิง 49 ครั้งตอนประชิดได้ 932 dmg/s → เซิร์ฟเวอร์กินดาเมจเฉพาะระยะใกล้
                            -- (tp จะวาร์ปตรงเมื่อ < 25 studs อยู่แล้ว ไม่เข้าเกณฑ์ anti-cheat)
                            if dist > 25 then
                                setState("ขยับเข้าระยะ (" .. math.floor(dist) .. ")", "B")
                                tp(CFrame.new(mhrp.Position + Vector3.new(0, 12, 0)))
                            end
                        end
                        aimCam(v)

                        if MAGNET_ON then
                            bringMobsToTarget(v, qdata.mob)
                        end
                    end

                    directRemoteAttack(v, fillerMode and v.Name or qdata.mob)
                    hitCount += 1
                    task.wait(0.05)
                end

                if v.Parent and v.Humanoid.Health <= 0 then
                    local dt = os.clock() - killT0
                    setState(string.format("มอนตาย ✓ | ใช้เวลา %.2fs | ยิง %d ครั้ง | %.0f dmg/s",
                        dt, hitCount, 3000 / math.max(dt, 0.01)), "G")
                end
            end
        end)
        end
    end
    stopTween()
    pcall(function() if _G.BF04_BV then _G.BF04_BV:Destroy() _G.BF04_BV = nil end end)
    log("=== STOP AUTO ===", "R")
end

-- ── AUTO SABER (ดาบดีที่สุดของ First Sea) ─────────────────────────
-- ลำดับเควสจากสคริปต์อ้างอิงที่ใช้งานได้จริง:
--   1. กดแป้นลับ 5 จุด: Workspace.Map.Jungle.QuestPlates.Plate1-5.Button
--   2. เอา Torch ไปจุดไฟที่ทะเลทราย (Map.Desert.Burn.Part)
--   3. เชนเควส ProQuestProgress: SickMan → GetCup → FillCup → RichSon
--   4. ฆ่า Mob Leader เอา Relic → ไปวางที่ Jungle → ฆ่า Saber Expert → PlaceRelic
local function equipByName(name)
    local char, _, hum = getChar()
    if not char or not hum then return false end
    if char:FindFirstChild(name) then return true end
    local bp = LP:FindFirstChild("Backpack")
    local tool = bp and bp:FindFirstChild(name)
    if tool then
        pcall(function() hum:EquipTool(tool) end)
        return true
    end
    return false
end

local function killNamedBoss(bossName)
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return false end
    for _, v in ipairs(enemies:GetChildren()) do
        if v.Name == bossName and v:FindFirstChild("Humanoid")
           and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
            setState("สู้ " .. bossName .. " (HP " .. math.floor(v.Humanoid.Health) .. ")", "Y")
            equipWeapon()
            local lastHpLog = 0
            while SABER_ON and v.Parent and v.Humanoid.Health > 0 do
                local char, hrp, myHum = getChar()

                -- ตายแล้ว → รอเกิดใหม่แล้วค่อยกลับมาสู้ต่อ (ไม่ใช่ค้างอยู่เฉยๆ)
                if not char or (myHum and myHum.Health <= 0) then
                    log("💀 ตายระหว่างสู้ " .. bossName .. " → รอเกิดใหม่", "R")
                    LP.CharacterAdded:Wait()
                    task.wait(3)
                    return false
                end

                local mhrp = v:FindFirstChild("HumanoidRootPart")
                if not (hrp and mhrp) then break end

                -- ปิดความสามารถของบอสให้มากที่สุด: หยุดเดิน หยุดกระโดด ย่อฮิตบ็อกซ์
                pcall(function()
                    v.Humanoid.WalkSpeed = 0
                    v.Humanoid.JumpPower = 0
                    mhrp.CanCollide = false
                    mhrp.Size = Vector3.new(1, 1, 1)
                end)

                -- ลอยเหนือหัวบอส 14 studs: ยังอยู่ในระยะที่ดาเมจเข้า
                -- แต่พ้นระยะท่าโจมตีพื้นของบอสส่วนใหญ่
                if (mhrp.Position - hrp.Position).Magnitude > 25 then
                    tp(CFrame.new(mhrp.Position + Vector3.new(0, 12, 0)))
                end

                -- ไม่ถอยพักฟื้นแล้ว (ถอย 42 วิ เลือดขึ้นแค่ไม่ถึง 30 หน่วย = ไม่คุ้ม)
                -- สู้จนตาย แล้วเกิดใหม่กลับมาสู้ต่อ HP บอสไม่รีเซ็ตถ้ายังไม่หายไปจากแมพ
                if v.Humanoid.Health < lastHpLog - 500 or lastHpLog == 0 then
                    lastHpLog = v.Humanoid.Health
                    setState(bossName .. " HP " .. math.floor(v.Humanoid.Health), "S")
                end

                -- เช็คทุกรอบ กันกรณีเกิดใหม่แล้วอาวุธหลุด หรือเควสยัดของกลับมาในมือ
                equipWeapon()
                aimCam(v)
                directRemoteAttack(v, bossName)
                task.wait(0.05)
            end
            if v.Humanoid.Health <= 0 then
                setState(bossName .. " ตาย ✓", "G")
                return true
            end
        end
    end
    return false
end

local SABER_CF = {
    plates = CFrame.new(-1612.55884, 36.9774132, 148.719543),
    torchPickup = CFrame.new(-1610.00757, 11.5049858, 164.001587),
    torchBurn = CFrame.new(1114.61475, 5.04679728, 4350.22803),
    mobLeader = CFrame.new(-2967.59521, -4.91089821, 5328.70703),
    relicPlace = CFrame.new(-1404.91504, 29.9773273, 3.80598116),
    saberExpert = CFrame.new(-1401.85046, 29.9773273, 8.81916237),
}

local function saberLoop()
    log("=== START AUTO SABER ===", "Y")
    local CommF = getCommF()
    if not CommF then log("หา CommF_ ไม่เจอ", "R") return end

    local lv = getMyLevel()
    if lv < 200 then
        log("❌ ต้อง Lv.200 ขึ้นไป (ตอนนี้ " .. lv .. ")", "R")
        SABER_ON = false
        saberBtn.Text = "AUTO SABER: OFF (ต้อง Lv.200+)"
        return
    end

    while SABER_ON and _G.BF04_SABER do
        task.wait(0.3)
        pcall(function()
            local char, hrp = getChar()
            if not char then setState("รอตัวละคร...", "S") return end

            local jungle = Workspace.Map:FindFirstChild("Jungle")
            local plates = jungle and jungle:FindFirstChild("QuestPlates")
            local final = jungle and jungle:FindFirstChild("Final")

            -- จบงานแล้ว: มี Saber ในกระเป๋า
            local bp = LP:FindFirstChild("Backpack")
            if (bp and bp:FindFirstChild("Saber")) or char:FindFirstChild("Saber") then
                log("🎉 ได้ Saber แล้ว! ปิด AUTO SABER", "G")
                SABER_ON = false
                _G.BF04_SABER = nil
                saberBtn.Text = "AUTO SABER: สำเร็จ ✓ (ได้ Saber แล้ว)"
                saberBtn.BackgroundColor3 = Color3.fromRGB(30,120,50)
                return
            end

            -- ⭐ ลำดับความสำคัญสูงสุด: ถ้า Saber Expert อยู่บนแมพแล้ว = วาง Relic สำเร็จ
            -- ข้ามการเช็คขั้นตอนอื่นทั้งหมด บินไปตีเลยด้วยระบบเดียวกับ AUTO ฟาร์ม
            local enemiesNow = Workspace:FindFirstChild("Enemies")
            if enemiesNow and enemiesNow:FindFirstChild("Saber Expert") then
                if killNamedBoss("Saber Expert") then
                    setState("ฆ่า Saber Expert สำเร็จ → PlaceRelic", "G")
                    CommF:InvokeServer("ProQuestProgress", "PlaceRelic")
                end
                return
            end

            -- ขั้นที่ 1: ยังไม่ได้กดแป้น 5 จุด (ประตูยังปิด)
            if plates and plates:FindFirstChild("Door") and plates.Door.Transparency == 0 then
                if (SABER_CF.plates.Position - hrp.Position).Magnitude > 100 then
                    setState("ขั้น 1: บินไป Jungle เพื่อกดแป้นลับ", "B")
                    tp(SABER_CF.plates)
                else
                    setState("ขั้น 1: กดแป้นลับ 5 จุด...", "Y")
                    for i = 1, 5 do
                        local plate = plates:FindFirstChild("Plate" .. i)
                        local btn = plate and plate:FindFirstChild("Button")
                        if btn then
                            local _, h = getChar()
                            if h then h.CFrame = btn.CFrame end
                            log("  กดแป้น " .. i .. "/5 ✓", "G")
                        else
                            log("  ❌ หา Plate" .. i .. " ไม่เจอ", "R")
                        end
                        task.wait(1)
                    end
                end
                return
            end

            -- ขั้นที่ 2: จุดไฟด้วย Torch ที่ทะเลทราย
            local desert = Workspace.Map:FindFirstChild("Desert")
            local burn = desert and desert:FindFirstChild("Burn")
            if burn and burn:FindFirstChild("Part") and burn.Part.Transparency == 0 then
                if equipByName("Torch") then
                    setState("ขั้น 2: ถือ Torch → บินไปจุดไฟที่ทะเลทราย", "B")
                    tp(SABER_CF.torchBurn)
                else
                    setState("ขั้น 2: บินไปเก็บ Torch ที่ Jungle", "B")
                    tp(SABER_CF.torchPickup)
                end
                return
            end

            -- ขั้นที่ 3: เชนเควสคนป่วย (SickMan → GetCup → FillCup)
            local sick = CommF:InvokeServer("ProQuestProgress", "SickMan")
            if sick ~= 0 then
                setState("ขั้น 3: เควสคนป่วย (รับถ้วย → ตักน้ำ → ส่ง)", "Y")
                CommF:InvokeServer("ProQuestProgress", "GetCup")
                task.wait(0.5)
                equipByName("Cup")
                task.wait(0.5)
                local cup = char:FindFirstChild("Cup")
                CommF:InvokeServer("ProQuestProgress", "FillCup", cup)
                task.wait(0.3)
                CommF:InvokeServer("ProQuestProgress", "SickMan")
                return
            end

            -- ขั้นที่ 4: เควสลูกเศรษฐี (ฆ่า Mob Leader → เอา Relic ไปวาง)
            local rich = CommF:InvokeServer("ProQuestProgress", "RichSon")
            if rich == nil then
                setState("ขั้น 4: รับเควสลูกเศรษฐี", "Y")
                CommF:InvokeServer("ProQuestProgress", "RichSon")
                return
            elseif rich == 0 then
                setState("ขั้น 4: บินไปฆ่า Mob Leader (Lv.120 Boss)", "B")
                tp(SABER_CF.mobLeader)
                killNamedBoss("Mob Leader")
                return
            elseif rich == 1 then
                -- ส่งเควสครั้งเดียวพอ ถ้าเรียกซ้ำทุกรอบเควสอาจรีเซ็ต
                if not RELIC_SENT then
                    RELIC_SENT = true
                    setState("ขั้น 4: ส่งเควสลูกเศรษฐี → รับ Relic", "Y")
                    CommF:InvokeServer("ProQuestProgress", "RichSon")
                    task.wait(0.5)
                end
                equipByName("Relic")
                local d = (SABER_CF.relicPlace.Position - hrp.Position).Magnitude
                -- อยู่แถวนั้นแล้วไม่ต้องดึงตัว (เดิมห่างเกิน 20 ก็ดึงกลับประตูทุกครั้ง น่ารำคาญ)
                -- ดึงเฉพาะกรณีอยู่คนละโซนจริงๆ
                if d > 150 then
                    setState("ขั้น 4: ถือ Relic บินกลับ Jungle (ห่าง " .. math.floor(d) .. ")", "B")
                    tp(SABER_CF.relicPlace)
                else
                    RELIC_TRIES = (RELIC_TRIES or 0) + 1

                    -- ลองยิง touch อัตโนมัติ 5 รอบ ถ้าไม่ผ่านให้คืนการควบคุมแก่ผู้เล่น
                    -- (ยิง touch ซ้ำไปเรื่อยๆ แต่ล็อกตำแหน่งไว้ = ผู้เล่นขยับเองไม่ได้ด้วย)
                    if RELIC_TRIES > 5 then
                        if not RELIC_MANUAL then
                            RELIC_MANUAL = true
                            releaseControl()
                            log("⚠️ วาง Relic อัตโนมัติไม่ผ่าน — คืนการควบคุมให้คุณแล้ว", "R")
                            log("   เดินด้วย WASD เข้าไปแตะแท่นเอง (สคริปต์จะไม่ดึงตัวคุณ)", "Y")
                            log("   วางสำเร็จแล้วมันจะไปสู้ Saber Expert ต่อเอง", "Y")
                        end
                        setState("ขั้น 4: รอคุณเดินไปแตะแท่นเอง (โหมดคุมมือ)", "Y")
                        task.wait(1)
                        return
                    end

                    -- วาร์ปเข้ามาเลยทำให้ไม่เกิดเหตุการณ์ "แตะ" แท่นวาง
                    -- ต้องยิง firetouchinterest ระหว่าง Handle ของ Relic กับชิ้นส่วนแท่น
                    setState("ขั้น 4: วาง Relic (ยิง touch ใส่แท่น " .. RELIC_TRIES .. "/5)", "Y")
                    local relic = char:FindFirstChild("Relic")
                    local handle = relic and relic:FindFirstChild("Handle")
                    local final = jungle and jungle:FindFirstChild("Final")
                    if handle and final and firetouchinterest then
                        for _, part in ipairs(final:GetDescendants()) do
                            if part:IsA("BasePart") then
                                pcall(function()
                                    firetouchinterest(handle, part, 0)
                                    firetouchinterest(handle, part, 1)
                                end)
                            end
                        end
                    end
                    if relic then pcall(function() relic:Activate() end) end
                    pcall(function() CommF:InvokeServer("ProQuestProgress", "PlaceRelic") end)
                    for i = 1, 4 do
                        local _, h = getChar()
                        if h then
                            h.CFrame = SABER_CF.relicPlace * CFrame.new(math.random(-3,3), 0, math.random(-3,3))
                        end
                        task.wait(0.4)
                    end
                end
                return
            end

            -- ขั้นที่ 5: บอส Saber Expert
            local enemies = Workspace:FindFirstChild("Enemies")
            if enemies and enemies:FindFirstChild("Saber Expert") then
                if killNamedBoss("Saber Expert") then
                    setState("ขั้น 5: ฆ่า Saber Expert สำเร็จ → PlaceRelic", "G")
                    CommF:InvokeServer("ProQuestProgress", "PlaceRelic")
                end
            else
                setState("ขั้น 5: บินไปจุดเรียก Saber Expert", "B")
                tp(SABER_CF.saberExpert)
            end
        end)
    end
    stopTween()
    log("=== STOP AUTO SABER ===", "R")
end
