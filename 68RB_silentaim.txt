-- หลักการ: เกมยิง = server raycast จากทิศ UpdateCameraRotation (ไม่ใช่ Camera.CFrame)
--   → Silent: ดักแก้ UpdateCameraRotation ให้ชี้ head ศัตรู → server hit
--   → auto-calibrate: เรียนรู้ mapping (มุมกล้อง ↔ byte) ตอน Silent OFF
--
-- ⚠ ทดลอง: ถ้า aim ผิดทิศ → ปรับ sign/offset (ปุ่มในจอ)

S.yawOffset = S.yawOffset or 0
S.yawSign   = S.yawSign   or 1            -- ทิศ (ถ้า aim สวน → -1)
local WatchOn  = true
local SilentOn = false
local SHOOT_MAX_SQ = 500 * 500

-- ==================== Char + target ====================
local LPChar, LPRoot
local function cacheLP(c) LPChar = c; LPRoot = c and c:FindFirstChild("HumanoidRootPart") end
cacheLP(LP.Character)
bind(LP.CharacterAdded, function(c) task.wait(0.3) cacheLP(c) end)

local rayP = RaycastParams.new()
rayP.FilterType = Enum.RaycastFilterType.Exclude

local function getTeamID(p)
    local id = p:GetAttribute("TeamID")
    if id and id ~= "" then return id end
    if p.Team then return "T_"..p.Team.Name end
    return nil
end

local curTarget = nil
local function findTarget()
    if not LPRoot then return nil end
    local from = LPRoot.Position
    local myID = getTeamID(LP)
    rayP.FilterDescendantsInstances = {LPChar}
    local best, bestSq = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hum  = p.Character:FindFirstChildOfClass("Humanoid")
            local head = p.Character:FindFirstChild("Head")
            if hum and head and hum.Health > 0 and not (myID and getTeamID(p) == myID) then
                local d = head.Position - from
                local sq = d.X*d.X + d.Y*d.Y + d.Z*d.Z
                if sq < bestSq and sq <= SHOOT_MAX_SQ then bestSq, best = sq, head end
            end
        end
    end
    return best
end

-- ==================== yaw helpers ====================
-- มุม yaw ของกล้องจริง (จาก Camera.CFrame) — องศา -180..180
local function camYawDeg()
    local lv = Camera.CFrame.LookVector
    return math.deg(math.atan2(lv.X, lv.Z))
end
-- มุม yaw จาก LPRoot ไป head เป้า
local function targetYawDeg(head)
    local d = head.Position - Camera.CFrame.Position
    return math.deg(math.atan2(d.X, d.Z))
end
-- แปลงองศา → yaw byte (ใช้ mapping ที่เรียนรู้)
local function yawToByte(deg)
    return math.floor((deg * S.yawScale * S.yawSign + S.yawOffset) % 256 + 0.5) % 256
end

-- ==================== HOOK UpdateCameraRotation ====================
if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local ok, method = pcall(getnc)
        if ok and (method == "FireServer" or method == "InvokeServer") then
            local name = ""
            pcall(function() name = self:GetFullName() end)
            if name:find("UpdateCameraRotation") then
                local s = (...)
                if type(s) == "string" then
                    local cps = {}
                    pcall(function() for _, c in utf8.codes(s) do cps[#cps+1] = c end end)
                    local pitch, yaw = cps[1], cps[2]
                    if pitch and yaw then
                        S.lastPitch = pitch; S.lastYaw = yaw
                        if not S.silent then
                            -- LEARN mapping: byte yaw = camYaw*scale*sign + offset
                            local cd = camYawDeg()
                            S.yawOffset = (yaw - cd * S.yawScale * S.yawSign) % 256
                            S.learnCam = math.floor(cd)
                        elseif S.tgtHead then
                            -- SILENT: แทน yaw byte ให้ชี้ head เป้า
                            local want = yawToByte(targetYawDeg(S.tgtHead))
                            local okE, ns = pcall(utf8.char, pitch, want)
                            if okE then
                                S.aimByte = want
                                return oldNc(self, ns, select(2, ...))
                            end
                        end
                    end
                end
            end
        end
        return oldNc(self, ...)
    end))
    S.restore = oldNc
end

-- ==================== Watch loop (E) ====================
bind(RunService.RenderStepped, function()
    curTarget = (WatchOn or SilentOn) and findTarget() or nil
    S.tgtHead = SilentOn and curTarget or nil
    S.silent  = SilentOn
    if WatchOn and curTarget then
        Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, curTarget.Position)
    end
end)

-- ==================== Logic ====================
local function updW()
    watchB.Text = "E: Watch "..(WatchOn and "ON" or "OFF")
    watchB.BackgroundColor3 = WatchOn and Color3.fromRGB(30,120,30) or Color3.fromRGB(50,50,50)
end
local function updS()
    silentB.Text = "Q: Silent "..(SilentOn and "ON" or "OFF")
    silentB.BackgroundColor3 = SilentOn and Color3.fromRGB(200,50,40) or Color3.fromRGB(60,40,60)
end
local function updSign() signB.Text = "ทิศ: "..(S.yawSign > 0 and "+" or "-") end

watchB.MouseButton1Click:Connect(function() WatchOn = not WatchOn updW() end)
silentB.MouseButton1Click:Connect(function() SilentOn = not SilentOn updS() end)
signB.MouseButton1Click:Connect(function() S.yawSign = -S.yawSign updSign() end)
offM.MouseButton1Click:Connect(function() S.yawOffset = (S.yawOffset - 8) % 256 end)
offP.MouseButton1Click:Connect(function() S.yawOffset = (S.yawOffset + 8) % 256 end)
off0.MouseButton1Click:Connect(function() S.yawOffset = 0 end)

bind(UIS.InputBegan, function(i, gp)
    if gp then return end
    if i.KeyCode == Enum.KeyCode.E then WatchOn = not WatchOn updW()
    elseif i.KeyCode == Enum.KeyCode.Q then SilentOn = not SilentOn updS() end
end)

            tgt, tostring(S.learnCam or "?"), tostring(S.lastYaw or "?"),
            S.yawOffset, S.yawSign, tyaw, tostring(S.aimByte or "?"))
    end
end)
