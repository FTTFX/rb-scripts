-- ── Helper ────────────────────────────────────────────────────────
local function fmt(v, full)
    local t = typeof(v)
    if t == "Instance" then return v.ClassName..":"..v.Name
    elseif t == "Vector3" then return string.format("V3(%.0f,%.0f,%.0f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then local p=v.Position return string.format("CF(%.0f,%.0f,%.0f)", p.X,p.Y,p.Z)
    elseif t == "string" then
        local mx = full and 300 or 40
        if #v > mx then return '"'..v:sub(1,mx)..'..."' end
        return '"'..v..'"'
    elseif t == "number" then return string.format("%.4g", v)
    elseif t == "boolean" then return tostring(v)
    else return t end
end
local function fmtArgs(...)
    local n = select("#", ...)
    if n == 0 then return "(ไม่มี args)" end
    local p = {}
    for i = 1, math.min(n, 6) do p[i] = fmt(select(i, ...), true) end
    return table.concat(p, " | ")
end
local function findPath(root, path)
    local cur = root
    for part in path:gmatch("[^.]+") do
        cur = cur and cur:FindFirstChild(part)
        if not cur then return nil end
    end
    return cur
end

-- ── Hook remotes แบบไม่กรอง (dedup ตาม path) ───────────────────────
S.remoteSeen = S.remoteSeen or {}
S.remoteCount = S.remoteCount or 0
local remoteLabels = {}   -- [path] = TextLabel

if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local m = getnc()
        if (m == "FireServer" or m == "InvokeServer") and S.enabled ~= false then
            local ok, nm = pcall(function() return self:GetFullName() end)
            nm = ok and nm or "?"
            local rec = S.remoteSeen[nm]
            if not rec then
                if S.remoteCount >= 200 then return oldNc(self, ...) end
                S.remoteCount += 1
                rec = { count = 0, first = fmtArgs(...), last = "" }
                S.remoteSeen[nm] = rec
            end
            rec.count += 1
            rec.last = fmtArgs(...)
        end
        return oldNc(self, ...)
    end))
end

-- ── Render hook summary (Heartbeat — อัปเดต label ที่มีอยู่) ────────
RunService.Heartbeat:Connect(function()
    if S.enabled == false then return end
    for nm, rec in pairs(S.remoteSeen) do
        local lbl = remoteLabels[nm]
        local short = nm:gsub("^.-Remotes%.", "")
        local txt = ("  ▶ %s  ×%d"):format(short, rec.count)
        if rec.count == 1 then
            txt = txt .. "  args=" .. rec.first
        else
            txt = txt .. "  last=" .. rec.last
        end
        if not lbl then
            lbl = Instance.new("TextLabel")
            lbl.Size = UDim2.new(1,-12,0,15); lbl.BackgroundTransparency = 1; lbl.Font = Enum.Font.Code
            lbl.TextSize = 11; lbl.TextXAlignment = Enum.TextXAlignment.Left; lbl.TextTruncate = Enum.TextTruncate.AtEnd
            lbl.TextColor3 = C.R; lbl.Parent = scroll
            order += 1; lbl.LayoutOrder = order
            remoteLabels[nm] = lbl
            rec.copyIdx = #copyLines + 1
            table.insert(copyLines, txt)
            scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
        end
        lbl.Text = txt
        if rec.copyIdx then copyLines[rec.copyIdx] = txt end
    end
end)

-- ── Dump table (module) แบบจำกัดความลึก ───────────────────────────
local function dumpTable(name, t, depth, budget)
    depth = depth or 0
    budget = budget or { n = 0 }
    if depth > 3 or budget.n > 400 then return end
    if type(t) ~= "table" then
        addLine(("%s = %s"):format(name, tostring(t)), C.G)
        return
    end
    local pad = string.rep("  ", depth)
    local cnt = 0
    for k, v in pairs(t) do
        cnt += 1
        budget.n += 1
        if budget.n > 400 then addLine(pad .. "...(ตัดท้าย)", C.S) return end
        local kn = tostring(k)
        local vt = type(v)
        if vt == "table" then
            local n = 0
            pcall(function() for _ in pairs(v) do n += 1 end end)
            addLine(("%s%s = <table %d keys>"):format(pad, kn, n), C.B)
            if depth < 2 and n <= 12 then
                dumpTable(name.."."..kn, v, depth + 1, budget)
            end
        elseif vt == "function" then
            addLine(("%s%s = [function]"):format(pad, kn), C.M)
        elseif vt == "userdata" then
            addLine(("%s%s = <userdata>"):format(pad, kn), C.S)
        else
            addLine(("%s%s = %s"):format(pad, kn, tostring(v)), C.G)
        end
    end
    if cnt == 0 then addLine(pad .. "(ว่าง)", C.S) end
end

local function tryRequire(path)
    local mod = findPath(RS, path)
    if not mod or not mod:IsA("ModuleScript") then
        addLine(("  ✗ %s (ไม่เจอ/ไม่ใช่ ModuleScript)"):format(path), C.S)
        return
    end
    local ok, result = pcall(function()
        local req = (getrenv and getrenv().require) or require
        return req(mod)
    end)
    if ok and result ~= nil then
        addLine(("  ✓ %s → %s"):format(path, type(result)), C.G)
        dumpTable("  "..path, result, 1)
    else
        addLine(("  ✗ %s require ไม่ได้: %s"):format(path, tostring(result)), C.O)
    end
end
