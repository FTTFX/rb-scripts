-- 66RB_camremote.txt v1.0 — Camera Rotation Remote Catcher + Replay
-- ดัก UpdateCameraRotation (และ Fighter.* remote) → ดูว่าส่ง args อะไร
-- แล้วลอง FireServer ส่งเอง (REPLAY) → ถ้า server รับ = เราคุมการหันตัวได้!
--
-- วิธีใช้:
--   1. รันตัวนี้ → หมุนกล้อง/เดิน → ดู args ที่ remote ส่ง (ละเอียด type+value)
--   2. กด REPLAY → ส่ง args เดิมซ้ำ → ดูว่าตัวหันตามไหม (= server รับ synthetic)
--   3. ถ้ารับ → ขั้นต่อไปทำ FireServer ด้วยทิศที่เราอยาก (BackFace แท้)

-- ==================== Arg formatter (ละเอียด — type + value เต็ม) ====================
local function detailVal(v)
    local t = typeof(v)
    if t == "string" then
        -- แสดง string + ความยาว + เผื่อ encoded (hex 8 ตัวแรก)
        local hex = ""
        for i = 1, math.min(#v, 8) do hex = hex..string.format("%02X ", v:byte(i)) end
        return string.format('str[%d] "%s"  hex:%s', #v, v:sub(1,40), hex)
    elseif t == "number" then return "num "..tostring(v)
    elseif t == "boolean" then return "bool "..tostring(v)
    elseif t == "Vector3" then return string.format("V3(%.2f,%.2f,%.2f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CFrame pos(%.1f,%.1f,%.1f)", p.X, p.Y, p.Z)
    elseif t == "table" then
        local parts = {}
        for k, val in pairs(v) do parts[#parts+1] = tostring(k).."="..typeof(val) end
        return "{"..table.concat(parts, ",").."}"
    else return t.." "..tostring(v) end
end

-- ==================== HOOK (ครั้งเดียว) ====================
if okHooks and not S.hooked then
    S.hooked = true
    local oldNc
    oldNc = hookmeta(game, "__namecall", newcc(function(self, ...)
        local ok, method = pcall(getnc)
        if ok and (method == "FireServer" or method == "InvokeServer") then
            local name = ""
            pcall(function() name = self:GetFullName() end)
            for _, kw in ipairs(WATCH) do
                if name:find(kw) then
                    -- เก็บ remote + args ล่าสุด (สำหรับ replay)
                    S.lastRemote = self
                    S.lastArgs   = table.pack(...)
                    S.lastName   = name
                    -- แยกเก็บตามชนิด (สำหรับ loop ทดสอบ)
                    if name:find("UpdateState") then
                        S.stateRemote = self; S.stateArgs = table.pack(...)
                    elseif name:find("Camera") or name:find("Rotation") then
                        S.camRemote = self; S.camArgs = table.pack(...)
                        -- decode pitch/yaw จาก arg แรก (UTF-8 codepoints = [pitch, yaw])
                        local s = (...)
                        if type(s) == "string" then
                            local ok2, cps = pcall(function()
                                local t = {}
                                for _, c in utf8.codes(s) do t[#t+1] = c end
                                return t
                            end)
                            if ok2 and cps[1] then
                                S.camPitch = cps[1]; S.camYaw = cps[2] or 0
                            end
                        end
                    end
                    -- ทำ text ละเอียด
                    local parts = {}
                    for i = 1, select("#", ...) do parts[i] = detailVal(select(i, ...)) end
                    local short = name:gsub("^.-Fighter%.", ""):gsub("^.-Remotes%.", "")
                    S.queue[#S.queue+1] = short.." | "..table.concat(parts, "  ||  ")
                    if #S.queue > 60 then table.remove(S.queue, 1) end
                    break
                end
            end
        end
        return oldNc(self, ...)
    end))
    S.restore = oldNc
end

local order = 0
local copyLines = {}
local MAXSHOW = 80
local lastText, lastLbl, lastN = nil, nil, 0
local function addLine(txt, col)
    if txt == lastText and lastLbl and lastLbl.Parent then
        lastN += 1; lastLbl.Text = txt.."   ×"..lastN
        copyLines[#copyLines] = lastLbl.Text
        return
    end
    order += 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1,-10,0,14); l.AutomaticSize = Enum.AutomaticSize.Y
    l.BackgroundTransparency = 1; l.Font = Enum.Font.Code; l.TextSize = 11
    l.TextXAlignment = Enum.TextXAlignment.Left; l.TextWrapped = true
    l.TextColor3 = col or Color3.fromRGB(200,220,255)
    l.Text = txt; l.LayoutOrder = order; l.Parent = scroll
    lastText = txt; lastLbl = l; lastN = 1
    copyLines[#copyLines+1] = txt
    local cnt = 0
    for _, c in ipairs(scroll:GetChildren()) do if c:IsA("TextLabel") then cnt += 1 end end
    if cnt > MAXSHOW then
        for _, c in ipairs(scroll:GetChildren()) do
            if c:IsA("TextLabel") then c:Destroy() break end
        end
    end
end

-- ==================== State LOOP (ทดสอบ God Mode) ====================
-- ส่ง UpdateState ค่าที่ตั้ง (true/false) ต่อเนื่องทุกเฟรม → ดูว่ายิงไม่เข้าไหม
local LOOP = nil  -- nil | "off" | "on"
local function setLoop(mode)
    LOOP = (LOOP == mode) and nil or mode
    stateOffB.BackgroundColor3 = LOOP=="off" and Color3.fromRGB(20,200,120) or Color3.fromRGB(40,140,90)
    stateOnB.BackgroundColor3  = LOOP=="on"  and Color3.fromRGB(220,140,40) or Color3.fromRGB(150,90,30)
    stateOffB.Text = LOOP=="off" and "🛡 OFF กำลังส่ง..." or "🛡 LOOP State=OFF"
    stateOnB.Text  = LOOP=="on"  and "ON กำลังส่ง..." or "LOOP State=ON"
end
stateOffB.MouseButton1Click:Connect(function() setLoop("off") end)
stateOnB.MouseButton1Click:Connect(function() setLoop("on") end)

-- loop ส่ง UpdateState ทุก Heartbeat
bind(RunService.Heartbeat, function()
    if not LOOP or not S.stateRemote then return end
    local val = (LOOP == "on")
    -- ใช้ arg แรกที่ดักได้ ("3") + bool ที่เราเลือก
    local first = S.stateArgs and S.stateArgs[1] or "3"
    pcall(function() S.stateRemote:FireServer(first, val) end)
end)

-- ==================== หันหลัง: ส่ง UpdateCameraRotation yaw+128 (180°) ====================
local CAM_BACK = false
local YAW_OFFSET = 128   -- 128 = +180° (หันหลัง) — ปรับได้ถ้าทิศไม่ตรง
backB.MouseButton1Click:Connect(function()
    CAM_BACK = not CAM_BACK
    backB.BackgroundColor3 = CAM_BACK and Color3.fromRGB(220,40,160) or Color3.fromRGB(140,40,120)
    backB.Text = CAM_BACK and "🔄 หันหลัง ON" or "🔄 หันหลัง LOOP"
    if CAM_BACK and not S.camRemote then
        addLine(">> ยังไม่ดัก Camera remote — หมุนกล้องก่อน", Color3.fromRGB(255,120,120))
    end
end)

-- loop ส่ง yaw+offset ทุก Heartbeat (หลังเกมส่ง yaw จริง → เราทับ)
bind(RunService.Heartbeat, function()
    if not CAM_BACK or not S.camRemote or not S.camPitch then return end
    local newYaw = (S.camYaw + YAW_OFFSET) % 256
    local okC, newStr = pcall(utf8.char, S.camPitch, newYaw)
    if not okC then return end
    local a = S.camArgs
    pcall(function()
        if a then a[1] = newStr; S.camRemote:FireServer(table.unpack(a, 1, a.n))
        else S.camRemote:FireServer(newStr) end
    end)
end)
clrB.MouseButton1Click:Connect(function()
    for _, c in pairs(scroll:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end
    order = 0; lastText = nil; table.clear(copyLines)
end)
copyB.MouseButton1Click:Connect(function()
    local clip = setclipboard or toclipboard or write_clipboard
    local ok = clip and pcall(clip, table.concat(copyLines, "\n"))
    copyB.Text = ok and "Copied!" or "ไม่รองรับ"
    task.delay(2, function() if copyB.Parent then copyB.Text = "COPY" end end)
end)

-- ==================== Drain queue → render + แสดง yaw ====================
bind(RunService.Heartbeat, function()
    -- live yaw readout
    if S.camYaw then
        local deg = math.floor(S.camYaw / 256 * 360)
        statL.Text = string.format("yaw:%d (%d°) pitch:%d %s",
            S.camYaw, deg, S.camPitch or 0, CAM_BACK and "| หันหลัง ON" or "")
    end
    local q = S.queue
    if #q == 0 then return end
    for i = 1, #q do addLine(q[i]) end
    table.clear(q)
    task.defer(function()
        scroll.CanvasSize = UDim2.new(0,0,0,layout.AbsoluteContentSize.Y + 6)
        scroll.CanvasPosition = Vector2.new(0, layout.AbsoluteContentSize.Y)
    end)
end)
