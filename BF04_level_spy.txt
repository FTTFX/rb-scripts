local function getLevel()
    local ok, lv = pcall(function() return LP.Data.Level.Value end)
    return (ok and lv) or 0
end

local function getSea()
    local ids = {
        [2753915549] = 1,
        [4442272183] = 2,
        [7449423635] = 3,           -- Sea3 เก่า
        [100117331123089] = 3,      -- Sea3 ใหม่ (ยืนยัน 2026-09-05)
    }
    return ids[game.PlaceId] or "?"
end

local function getChar()
    local c = LP.Character
    local h = c and c:FindFirstChild("HumanoidRootPart")
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    if c and h and hum then return c, h, hum end
end

local function getCommF()
    local ok, r = pcall(function()
        return RS:WaitForChild("Remotes", 5):WaitForChild("CommF_", 5)
    end)
    return ok and r or nil
end

local function fmtCF(cf)
    local p = cf.Position
    return string.format("%.1f, %.1f, %.1f", p.X, p.Y, p.Z)
end

local function fmtPos(v)
    return string.format("%.1f, %.1f, %.1f", v.X, v.Y, v.Z)
end

-- noclip เบาๆ ตอนบิน
local function setNoclip()
    local c = LP.Character
    if not c then return end
    for _, p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") then p.CanCollide = false end
    end
end

local function flyTo(cf, label)
    local _, hrp = getChar()
    if not hrp then log("ไม่มีตัวละคร", "R") return false end
    setNoclip()
    local dist = (hrp.Position - cf.Position).Magnitude
    log(string.format("บิน→%s ระยะ %.0f | เป้า %s", label, dist, fmtCF(cf)), "B")
    if dist < 20 then
        hrp.CFrame = cf
        log("ถึงแล้ว (ใกล้)", "G")
        refreshStatus()
        return true
    end
    local spd = 200
    if dist > 1000 then spd = 160 end
    local tw = TweenService:Create(hrp, TweenInfo.new(dist / spd, Enum.EasingStyle.Linear), { CFrame = cf })
    tw:Play()
    tw.Completed:Wait()
    local _, h2 = getChar()
    local left = h2 and (h2.Position - cf.Position).Magnitude or 9999
    if left <= 35 then
        log(string.format("ถึงแล้ว (เหลือ %.0f)", left), "G")
        if h2 then h2.CFrame = cf end
    else
        log(string.format("ยังไม่ถึง (เหลือ %.0f) — อาจโดนเด้ง/ติดแมพ", left), "R")
    end
    refreshStatus()
    return left <= 35
end

local function countMobs(filterName, nearPos, radius)
    local enemies = Workspace:FindFirstChild("Enemies")
    local counts, samples = {}, {}
    if not enemies then return counts, samples, 0 end
    local total = 0
    for _, e in ipairs(enemies:GetChildren()) do
        local hum = e:FindFirstChildOfClass("Humanoid")
        local hrp = e:FindFirstChild("HumanoidRootPart")
        if hum and hrp and hum.Health > 0 then
            if nearPos and (hrp.Position - nearPos).Magnitude > (radius or 250) then
                -- skip far
            else
                total += 1
                counts[e.Name] = (counts[e.Name] or 0) + 1
                if #samples < 8 then
                    table.insert(samples, {
                        name = e.Name,
                        hp = math.floor(hum.Health),
                        pos = fmtPos(hrp.Position),
                        d = nearPos and math.floor((hrp.Position - nearPos).Magnitude) or 0,
                    })
                end
            end
        end
    end
    if filterName then
        -- also report exact match count
    end
    return counts, samples, total
end

local function scanAround(center, radius, tag)
    local _, hrp = getChar()
    center = center or (hrp and hrp.Position)
    if not center then log("ไม่มีตำแหน่ง", "R") return end
    log("── สแกน " .. tag .. " รัศมี " .. radius .. " ──", "Y")
    log("ศูนย์กลาง: " .. fmtPos(center), "S")
    local counts, samples, total = countMobs(nil, center, radius)
    log("ศัตรูทั้งหมดในรัศมี: " .. total, total > 0 and "G" or "R")
    local names = {}
    for n, c in pairs(counts) do
        table.insert(names, n .. "(" .. c .. ")")
    end
    table.sort(names)
    if #names == 0 then
        log("ไม่พบมอนในรัศมีนี้", "R")
    else
        log("รายชื่อ: " .. table.concat(names, ", "), "B")
    end
    for _, s in ipairs(samples) do
        log(string.format("  • %s HP%d d=%d @ %s", s.name, s.hp, s.d, s.pos), "S")
    end
    if selected then
        local expect = selected.mob
        local found = 0
        for n, c in pairs(counts) do
            if n:find(expect, 1, true) then found += c end
        end
        if found > 0 then
            log("✓ เจอมอนที่คาดหวัง '" .. expect .. "' = " .. found .. " ตัว", "G")
        else
            log("✗ ไม่เจอ '" .. expect .. "' ในรัศมี — พิกัดเก่าอาจผิด/มอนย้าย", "R")
        end
    end
end

local function shortVal(v, depth)
    depth = depth or 0
    if depth > 3 then return "..." end
    local t = typeof(v)
    if t == "string" then
        if #v > 80 then return '"' .. v:sub(1, 80) .. '..."' end
        return '"' .. v .. '"'
    elseif t == "number" or t == "boolean" then
        return tostring(v)
    elseif t == "Vector3" then
        return string.format("V3(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CF(%.1f,%.1f,%.1f)", p.X, p.Y, p.Z)
    elseif t == "table" then
        local parts = {}
        local n = 0
        for k, vv in pairs(v) do
            n += 1
            if n > 12 then parts[#parts+1] = "..." break end
            parts[#parts+1] = tostring(k) .. "=" .. shortVal(vv, depth + 1)
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    elseif t == "Instance" then
        return v.ClassName .. ":" .. v.Name
    end
    return t
end

local function dumpQuestModule()
    log("── ReplicatedStorage.Quests (require) ──", "Y")
    local mod = RS:FindFirstChild("Quests")
    if not mod then
        -- หาชื่อคล้าย
        for _, c in ipairs(RS:GetChildren()) do
            if c.Name:lower():find("quest") then
                log("  เจอใกล้เคียง: " .. c.ClassName .. " " .. c.Name, "O")
            end
        end
        log("  ✗ ไม่มี RS.Quests", "R")
        return
    end
    log("  ✓ มี " .. mod.ClassName .. ":Quests", "G")
    local ok, data = pcall(require, mod)
    if not ok then
        log("  require ล้ม: " .. tostring(data), "R")
        return
    end
    if type(data) ~= "table" then
        log("  require ได้ แต่ไม่ใช่ table: " .. typeof(data), "O")
        return
    end
    -- ลิสต์ key ทั้งหมด (สั้น)
    local keys = {}
    for k in pairs(data) do keys[#keys+1] = tostring(k) end
    table.sort(keys)
    log("  keys ทั้งหมด (" .. #keys .. "): " .. table.concat(keys, ", "), "B")
    -- โฟกัส Amazon / Hydra / selected
    local want = {}
    if selected then want[selected.q] = true end
    want["AmazonQuest"] = true
    want["AmazonQuest2"] = true
    for _, k in ipairs(keys) do
        local kl = k:lower()
        if want[k] or kl:find("amazon") or kl:find("hydra") or kl:find("dragon") then
            log("  ★ [" .. k .. "] = " .. shortVal(data[k]), "G")
        end
    end
end

local function listAllNpcs(filterOnly)
    foundNpcs = {}
    log("── workspace.NPCs ──", "Y")
    local folder = Workspace:FindFirstChild("NPCs")
    if not folder then
        log("  ✗ ไม่มี workspace.NPCs", "R")
        -- ลองหา folder อื่น
        for _, name in ipairs({"NPC", "Npcs", "QuestNPCs", "Characters"}) do
            if Workspace:FindFirstChild(name) then
                log("  มี Workspace." .. name .. " แทน", "O")
            end
        end
        return
    end
    local kids = folder:GetChildren()
    log("  จำนวน NPC: " .. #kids, "B")
    local keywords = { "amazon", "hydra", "dragon", "quest", "island", "crew", "warrior", "archer" }
    local shown = 0
    for _, n in ipairs(kids) do
        local part = n:FindFirstChild("HumanoidRootPart")
            or n:FindFirstChildWhichIsA("BasePart", true)
        local pos = part and part.Position
        local nl = n.Name:lower()
        local hit = not filterOnly
        if filterOnly then
            for _, kw in ipairs(keywords) do
                if nl:find(kw, 1, true) then hit = true break end
            end
            if selected and nl:find(selected.mob:lower():sub(1, 6), 1, true) then hit = true end
        end
        if hit then
            shown += 1
            if pos then
                foundNpcs[#foundNpcs+1] = { name = n.Name, pos = pos }
                log(string.format("  • %s @ %.1f, %.1f, %.1f", n.Name, pos.X, pos.Y, pos.Z), "G")
                log("    CFrame.new(" .. string.format("%.2f, %.2f, %.2f", pos.X, pos.Y, pos.Z) .. ")", "S")
            else
                log("  • " .. n.Name .. " (ไม่มี BasePart)", "O")
            end
        end
    end
    if filterOnly and shown == 0 then
        log("  ไม่เจอชื่อที่เกี่ยวกับ Amazon/Hydra/Dragon — กด 'ลิสต์ NPCs ทั้งหมด'", "R")
    elseif not filterOnly then
        log("  แสดงครบ " .. shown .. " ตัว", "B")
    else
        log("  ตรงคีย์เวิร์ด " .. shown .. " ตัว (จาก " .. #kids .. ")", "B")
    end
end

local function listAllMobs()
    log("── workspace.Enemies ทั้งแมพ ──", "Y")
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then
        log("  ✗ ไม่มี Enemies", "R")
        return
    end
    local counts = {}
    local samples = {}
    for _, e in ipairs(enemies:GetChildren()) do
        local hum = e:FindFirstChildOfClass("Humanoid")
        local hrp = e:FindFirstChild("HumanoidRootPart")
        if hum and hum.Health > 0 then
            counts[e.Name] = (counts[e.Name] or 0) + 1
            if hrp and not samples[e.Name] then
                samples[e.Name] = hrp.Position
            end
        end
    end
    local names = {}
    for n, c in pairs(counts) do names[#names+1] = n end
    table.sort(names)
    if #names == 0 then
        log("  ไม่มีมอนใน Enemies ตอนนี้เลย (streaming/ไกลเกาะ?)", "R")
        return
    end
    log("  ชนิดมอน: " .. #names, "B")
    for _, n in ipairs(names) do
        local p = samples[n]
        if p then
            log(string.format("  %s x%d @ %.0f, %.0f, %.0f", n, counts[n], p.X, p.Y, p.Z), "G")
            log("    CFrame.new(" .. string.format("%.2f, %.2f, %.2f", p.X, p.Y, p.Z) .. ")", "S")
        else
            log(string.format("  %s x%d", n, counts[n]), "G")
        end
    end
    if selected then
        local found = false
        for _, n in ipairs(names) do
            if n:find(selected.mob, 1, true) then found = true break end
        end
        if found then
            log("✓ มี '" .. selected.mob .. "' ในแมพ — พิกัดเก่าผิด/ไกล", "G")
        else
            log("✗ ไม่มี '" .. selected.mob .. "' ใน Enemies ตอนนี้", "R")
        end
    end
end

local function dumpGameData()
    log("════ หาใน Data เกม ════", "Y")
    log("PlaceId=" .. tostring(game.PlaceId) .. " Sea=" .. tostring(getSea()) .. " Lv=" .. getLevel(), "B")
    -- 1 Quests module
    pcall(dumpQuestModule)
    -- 2 NPCs related
    pcall(function() listAllNpcs(true) end)
    -- 3 All mobs
    pcall(listAllMobs)
    -- 4 LP.Data quest-ish
    log("── LP.Data (คีย์ที่เกี่ยวกับเควส) ──", "Y")
    pcall(function()
        local data = LP:FindFirstChild("Data")
        if not data then log("  ไม่มี LP.Data", "R") return end
        for _, c in ipairs(data:GetChildren()) do
            local nl = c.Name:lower()
            if nl:find("quest") or nl:find("spawn") or nl:find("level") or nl:find("sea") then
                local extra = c:IsA("ValueBase") and (" = " .. tostring(c.Value)) or ""
                log("  " .. c.Name .. " [" .. c.ClassName .. "]" .. extra, "G")
            end
        end
    end)
    -- 5 RS children with quest in name
    log("── RS ที่มีชื่อ quest ──", "Y")
    pcall(function()
        for _, c in ipairs(RS:GetDescendants()) do
            if c.Name:lower():find("quest") and (c:IsA("ModuleScript") or c:IsA("Folder") or c:IsA("RemoteEvent") or c:IsA("RemoteFunction")) then
                log("  " .. c.ClassName .. " " .. c:GetFullName():gsub("ReplicatedStorage%.", "RS."), "S")
            end
        end
    end)
    log("════ จบ Data dump — Copy ส่งมาได้ ════", "Y")
end
