local currentTween
local function flyTo(posOrCF, label)
    local _, hrp = getChar()
    if not hrp then return false end
    setNoclip()
    local dest = typeof(posOrCF) == "CFrame" and posOrCF
        or CFrame.new(posOrCF + Vector3.new(0, 5, 0))
    local dist = (hrp.Position - dest.Position).Magnitude
    log(string.format("บิน→%s %.0f studs", label or "เป้า", dist), "B")
    if dist < 18 then
        hrp.CFrame = dest
        return true
    end
    local spd = FLY_SPEED
    if dist > 1200 then spd = 150 end
    if currentTween then pcall(function() currentTween:Cancel() end) end
    local tw = TweenService:Create(hrp,
        TweenInfo.new(dist / spd, Enum.EasingStyle.Linear), { CFrame = dest })
    currentTween = tw
    tw:Play()
    tw.Completed:Wait()
    currentTween = nil
    local _, h2 = getChar()
    if not h2 then return false end
    local left = (h2.Position - dest.Position).Magnitude
    if left <= 50 then
        h2.CFrame = dest
        return true
    end
    log(string.format("ยังไม่ถึง (เหลือ %.0f)", left), "Y")
    return false
end

-- คืน { qName, qNum, mob, need, levelReq, displayName }
local function pickQuestForLevel(lv)
    local data = getQuests()
    if not data then return nil end
    local best = nil
    for qName, entry in pairs(data) do
        if type(entry) == "table" then
            for num, sub in pairs(entry) do
                if type(sub) == "table" and type(sub.LevelReq) == "number"
                   and sub.LevelReq <= lv and type(sub.Task) == "table" then
                    local mob, need = next(sub.Task)
                    if mob and type(need) == "number" then
                        local cand = {
                            qName = qName,
                            qNum = tonumber(num) or num,
                            mob = mob,
                            need = need,
                            levelReq = sub.LevelReq,
                            displayName = sub.Name or mob,
                        }
                        if not best or cand.levelReq > best.levelReq
                           or (cand.levelReq == best.levelReq and cand.qNum > best.qNum) then
                            best = cand
                        end
                    end
                end
            end
        end
    end
    return best
end

-- ── หาเป้าเข็มทิศ / NPC ────────────────────────────────────────────
local function tryGetBestQuestNPC(timeoutSec)
    timeoutSec = timeoutSec or 3
    local rem = RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("GetBestQuestNPC")
    if not rem then return nil, "ไม่มี GetBestQuestNPC" end
    local done, res, err = false, nil, nil
    task.spawn(function()
        local ok, r = pcall(function() return rem:InvokeServer() end)
        done = true
        if ok then res = r else err = tostring(r) end
    end)
    local t0 = os.clock()
    while not done and os.clock() - t0 < timeoutSec do
        task.wait(0.05)
    end
    if not done then return nil, "timeout " .. timeoutSec .. "s (ค้าง?)" end
    if err then return nil, err end
    return res, nil
end

local function posFromAny(v)
    if typeof(v) == "Vector3" then return v end
    if typeof(v) == "CFrame" then return v.Position end
    if typeof(v) == "Instance" then
        if v:IsA("Model") then
            local h = v:FindFirstChild("HumanoidRootPart") or v:FindFirstChildWhichIsA("BasePart")
            return h and h.Position
        elseif v:IsA("BasePart") then
            return v.Position
        end
    end
    if type(v) == "table" then
        if typeof(v.Position) == "Vector3" then return v.Position end
        if typeof(v.CFrame) == "CFrame" then return v.CFrame.Position end
        if v[1] and typeof(v[1]) == "number" and v[3] then
            return Vector3.new(v[1], v[2], v[3])
        end
    end
end

local function dumpValue(v, depth)
    depth = depth or 0
    if depth > 2 then return "..." end
    local t = typeof(v)
    if t == "Instance" then
        local p = posFromAny(v)
        local extra = p and string.format(" @%.0f,%.0f,%.0f", p.X, p.Y, p.Z) or ""
        return v.ClassName .. ":" .. v.Name .. extra
    elseif t == "Vector3" then
        return string.format("V3(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CF(%.1f,%.1f,%.1f)", p.X, p.Y, p.Z)
    elseif t == "table" then
        local parts = {}
        local n = 0
        for k, vv in pairs(v) do
            n += 1
            if n > 10 then parts[#parts+1] = "..." break end
            parts[#parts+1] = tostring(k) .. "=" .. dumpValue(vv, depth + 1)
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    return tostring(v)
end

-- อ่าน UI เข็มทิศ / เควสแนะนำ (กรอง noise จาก settings)
local function readCompassUI()
    local info = { texts = {}, distance = nil, title = nil }
    local main = pg:FindFirstChild("Main")
    if not main then return info end
    for _, d in ipairs(main:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            local t = tostring(d.Text or "")
            if t ~= "" and #t < 100 then
                local tl = t:lower()
                if tl:find("mobile") or tl:find("robux") or tl:find("lag")
                   or tl:find("fury") or tl:find("restore data") or tl:find("material") then
                    -- skip
                elseif tl:find("find the") or tl:find("quest giver")
                    or t == "Recommended Quest" or t == "Track"
                    or tl:find("เควสแนะนำ") or tl:find("เมตร") then
                    info.texts[#info.texts+1] = t
                    local meters = t:match("(%d+)%s*เมตร") or t:match("(%d+)%s*[Mm]eter")
                    if meters then info.distance = tonumber(meters) end
                    if tl:find("find the") or tl:find("quest giver") then
                        info.title = t
                    end
                end
            end
        end
    end
    return info
end

-- หา Quest Giver ใน workspace.NPCs ที่ชื่อเกี่ยวกับเควส/มอน
local function findQuestGiver(quest)
    local folder = Workspace:FindFirstChild("NPCs")
    if not folder then return nil end
    local _, myHrp = getChar()
    local best, bestD = nil, math.huge
    local keys = {}
    if quest then
        -- "Dragon Crew Warrior" → ลอง "Dragon Crew"
        local mob = quest.mob or ""
        local words = {}
        for w in mob:gmatch("%S+") do words[#words+1] = w end
        if #words >= 2 then
            keys[#keys+1] = words[1] .. " " .. words[2]
        end
        keys[#keys+1] = mob
        keys[#keys+1] = quest.displayName or ""
        keys[#keys+1] = quest.qName or ""
    end
    keys[#keys+1] = "Quest Giver"
    for _, npc in ipairs(folder:GetChildren()) do
        local nl = npc.Name:lower()
        local hit = nl:find("quest giver", 1, true) or nl:find("questgiver", 1, true)
        if not hit and quest then
            for _, k in ipairs(keys) do
                if k ~= "" and nl:find(k:lower(), 1, true) then hit = true break end
            end
        end
        if hit then
            local part = npc:FindFirstChild("HumanoidRootPart")
                or npc:FindFirstChildWhichIsA("BasePart")
            if part then
                local d = myHrp and (part.Position - myHrp.Position).Magnitude or 0
                if d < bestD then bestD = d; best = part end
            end
        end
    end
    return best, bestD
end

-- รวมแหล่งหาเป้า NPC (NPCs ก่อน — GetBestQuestNPC ค้างบ่อย)
local function resolveNpcTarget(quest)
    local part, dist = findQuestGiver(quest)
    if part then
        log(string.format("เจอ NPC: %s (%.0f studs)", part.Parent and part.Parent.Name or part.Name, dist or 0), "G")
        return part.Position, "workspace.NPCs"
    end

    local ui = readCompassUI()
    if ui.title then log("เข็มทิศ UI: " .. ui.title, "O") end

    local res, err = tryGetBestQuestNPC(1.5)
    if res ~= nil then
        log("GetBestQuestNPC → " .. dumpValue(res), "G")
        local p = posFromAny(res)
        if not p and type(res) == "table" then
            p = posFromAny(res.NPC or res.npc or res.Object or res.Position or res.CFrame)
        end
        if p then return p, "GetBestQuestNPC" end
    elseif err then
        log("GetBestQuestNPC ข้าม: " .. tostring(err), "S")
    end

    return nil, "NPC ยังไม่โหลด — จะบินใกล้มอนเพื่อสตรีมเกาะ"
end

local function findMob(mobName)
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return nil end
    local _, myHrp = getChar()
    local best, bestD = nil, math.huge
    for _, e in ipairs(enemies:GetChildren()) do
        if e.Name:find(mobName, 1, true) then
            local hum = e:FindFirstChildOfClass("Humanoid")
            local hrp = e:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local d = myHrp and (hrp.Position - myHrp.Position).Magnitude or 0
                if d < bestD then bestD = d; best = e end
            end
        end
    end
    return best, bestD
end

local function getToolByTip(tip)
    local char = LP.Character
    local bp = LP:FindFirstChild("Backpack")
    for _, src in ipairs({ char, bp }) do
        if src then
            for _, t in ipairs(src:GetChildren()) do
                if t:IsA("Tool") and tostring(t.ToolTip or "") == tip then return t end
            end
        end
    end
end

local function equipSword()
    local char, _, hum = getChar()
    if not char or not hum then return false end
    local held = char:FindFirstChildOfClass("Tool")
    if held and tostring(held.ToolTip or "") == "Sword" then return true end
    local sword = getToolByTip("Sword")
    if sword then
        pcall(function() hum:EquipTool(sword) end)
        task.wait(0.15)
        return true
    end
    -- fallback กด 3 ตามที่ใช้อยู่
    pressKey(Enum.KeyCode.Three, 0.08)
    task.wait(0.2)
    held = char:FindFirstChildOfClass("Tool")
    return held ~= nil
end

local function isBigForm()
    local _, hrp = getChar()
    if not hrp then return false end
    return hrp.Size.Magnitude >= 5 or hrp.Size.Y >= 3.5
end

-- คอมโบผู้ใช้: กด 2 (ผล) → Z (ขยาย) → กด 3 (ดาบ) ร่างค้าง
local lastBuddhaT = 0
local function ensureBuddhaSword(force)
    if isBigForm() then
        return equipSword()
    end
    if not force and os.clock() - lastBuddhaT < 10 then
        return equipSword()
    end
    lastBuddhaT = os.clock()
    local char, _, hum = getChar()
    if not char or not hum then return false end

    log("คอมโบร่างพระ: 2 → Z → 3", "Y")
    local fruit = getToolByTip("Blox Fruit")
    if fruit then
        pcall(function() hum:EquipTool(fruit) end)
        task.wait(0.35)
    else
        pressKey(Enum.KeyCode.Two, 0.08)
        task.wait(0.35)
    end

    if not pressKey(Enum.KeyCode.Z, 0.12) then
        log("กด Z ไม่ได้ (ไม่มี VIM) → กด Z เองครั้งเดียว", "O")
    end
    task.wait(0.9)

    if not isBigForm() then
        pressKey(Enum.KeyCode.Z, 0.12)
        task.wait(0.7)
    end

    equipSword()
    if isBigForm() then
        log("ร่างพระพร้อม + ถือดาบ ✓", "G")
        return true
    end
    log("ยังไม่ใหญ่ — ถือดาบไว้ก่อน (กด Z เองได้)", "O")
    return equipSword()
end

            end
        else
            log("ไม่เจอ " .. quest.mob .. " ใน Enemies ตอนนี้", "Y")
        end
    end

    pcall(function()
        local qtl = RS:FindFirstChild("Effect") and RS.Effect:FindFirstChild("Container")
            and RS.Effect.Container:FindFirstChild("QuestTracingLine")
        if qtl then
            log("มี Effect.QuestTracingLine [" .. qtl.ClassName .. "]", "B")
        end
    end)
    log("════ จบ Dump — Copy ส่งมาได้ ════", "Y")
end

-- ── Main loop ──────────────────────────────────────────────────────
local function mainLoop()
    log("=== START Compass Auto ===", "Y")
    local cf = getCommF()
    if not cf then log("ไม่มี CommF_", "R") return end
    getCombatRemotes()
    pcall(function() ensureBuddhaSword(true) end)

    while _G.BF04C_ON do
        local ok, err = pcall(function()
            local lv = getLevel()
            local quest = pickQuestForLevel(lv)
            if not quest then
                log("ไม่มีเควสที่ LevelReq <= " .. lv, "R")
                task.wait(3)
                return
            end
            refreshStatus(quest)

            -- ── ยังไม่มีเควส → หา NPC ──
            if not questUIVisible() then
                local pos, src = resolveNpcTarget(quest)
                if not pos then
                    log("รอเป้า NPC... (" .. tostring(src) .. ")", "Y")
                    local mob, md = findMob(quest.mob)
                    if mob and md and md > 80 then
                        local mhrp = mob:FindFirstChild("HumanoidRootPart")
                        if mhrp then flyTo(mhrp.CFrame * CFrame.new(0, 20, 0), "ใกล้เกาะมอน") end
                    end
                    task.wait(1.5)
                    return
                end
                local _, hrp = getChar()
                local dist = hrp and (hrp.Position - pos).Magnitude or 9999
                local nearDist = isBigForm() and 55 or 28
                if dist > nearDist then
                    flyTo(CFrame.new(pos + Vector3.new(0, 6, 0)), "NPC/" .. src)
                    return
                end
                equipSword() -- ถือดาบ ไม่ถือผล ตอนรับเควส
                for _ = 1, 4 do
                    local _, h = getChar()
                    if h then h.CFrame = CFrame.new(pos + Vector3.new(0, 4, 0)) end
                    task.wait(0.12)
                end
                local _, hNow = getChar()
                local dNow = hNow and (hNow.Position - pos).Magnitude or -1
                log(string.format("StartQuest(%s, %s) ร่างใหญ่=%s ระยะ=%.0f",
                    quest.qName, tostring(quest.qNum), tostring(isBigForm()), dNow), "Y")
                local iok, ires = pcall(function()
                    return cf:InvokeServer("StartQuest", quest.qName, quest.qNum)
                end)
                log("ผล: ok=" .. tostring(iok) .. " res=" .. tostring(ires), "B")
                local w = 0
                while not questUIVisible() and w < 25 and _G.BF04C_ON do
                    local _, h = getChar()
                    if h then h.CFrame = CFrame.new(pos + Vector3.new(0, 4, 0)) end
                    task.wait(0.1); w += 1
                end
                if questUIVisible() then
                    log("รับเควสสำเร็จ: " .. questTitle(), "G")
                else
                    log("ยังไม่เปิด UI (ลองใหม่)", "R")
                    task.wait(1.2)
                end
                return
            end

            local titleTxt = questTitle()
            if titleTxt ~= "" and not titleTxt:lower():find(quest.mob:lower():sub(1, 8), 1, true)
               and not titleTxt:find(quest.displayName or "", 1, true) then
                log("เควสไม่ตรง (" .. titleTxt .. ") → Abandon", "Y")
                pcall(function() cf:InvokeServer("AbandonQuest") end)
                task.wait(0.8)
                return
            end

            -- ก่อนตี: ให้มีร่างพระ + ดาบ
            ensureBuddhaSword(false)

            local mob, md = findMob(quest.mob)
            if not mob then
                log("ไม่เจอ " .. quest.mob .. " ใน Enemies → รอโหลด", "Y")
                task.wait(1.2)
                return
            end
            local mhrp = mob:FindFirstChild("HumanoidRootPart")
            local mhum = mob:FindFirstChildOfClass("Humanoid")
            if not mhrp or not mhum then return end

            if (md or 999) > SAFE_HIT_RANGE then
                flyTo(mhrp.CFrame * CFrame.new(0, attackHeight(), 0), "มอน")
                return
            end

            log(string.format("สู้ %s HP%.0f (d=%.0f big=%s)",
                mob.Name, mhum.Health, md or 0, tostring(isBigForm())), "Y")
            local lastHp = mhum.Health
            local lastCheck = os.clock()
            local noDmg = 0
            while mob.Parent and mhum.Parent and mhum.Health > 0
                  and questUIVisible() and _G.BF04C_ON do
                local _, myHrp = getChar()
                if not myHrp then break end
                mhrp = mob:FindFirstChild("HumanoidRootPart")
                if not mhrp then break end
                setNoclip()
                equipSword()
                -- ล็อกเหนือหัว → ระยะจริง ≈ attackHeight (≤34 จาก Spy)
                myHrp.CFrame = mhrp.CFrame * CFrame.new(0, attackHeight(), 0)
                local nowD = (myHrp.Position - mhrp.Position).Magnitude
                if nowD > SAFE_HIT_RANGE then
                    -- เด้งออกนอกระยะ → วาร์ปกลับทันที
                    myHrp.CFrame = mhrp.CFrame * CFrame.new(0, attackHeight(), 0)
                end
                pcall(function()
                    -- hitbox เสริมเล็กน้อย (ไม่พึ่งเป็นหลัก — Spy วัดโดยไม่ขยาย)
                    mhrp.Size = Vector3.new(25, 25, 25)
                    mhrp.CanCollide = false
                    mhum.PlatformStand = true
                    mhum.Sit = true
                end)
                attackMob(mob)
                if os.clock() - lastCheck >= 1.2 then
                    lastCheck = os.clock()
                    local hp = mhum.Health
                    if hp >= lastHp then
                        noDmg += 1
                        if noDmg >= 3 then
                            log("ตีไม่เข้า → ข้ามตัวนี้", "Y")
                            break
                        end
                    else
                        noDmg = 0
                        lastHp = hp
                    end
                end
                task.wait()
            end
            log(mob.Name .. " จบรอบ", "G")
            task.wait(0.3)
        end)
        if not ok then
            log("Loop error: " .. tostring(err), "R")
            task.wait(1)
        else
            task.wait(0.15)
        end
    end
    log("=== STOP ===", "R")
end

-- ── Spy ระยะตี: จำตำแหน่งมอน → เริ่มจากไกลสุด → ค่อยเข้าใกล้ → โดนครั้งแรก = ระยะสูงสุด ──
local spyBusy = false
local function spyAttackRange()
    if spyBusy then log("กำลังวัดอยู่ รอจบก่อน", "Y") return end
    spyBusy = true
    local ok, err = pcall(function()
        log("════ Spy ระยะตี v2 (ไกล→ใกล้) ════", "Y")
        _G.BF04C_ON = false
        autoBtn.Text = "AUTO: OFF"
        autoBtn.BackgroundColor3 = Color3.fromRGB(30, 120, 50)
        task.wait(0.4) -- ให้ AUTO หยุดบินก่อน

        ensureBuddhaSword(true)
        local _, hrp0 = getChar()
        if hrp0 then
            log(string.format("HRP Size=%.1f,%.1f,%.1f | big=%s",
                hrp0.Size.X, hrp0.Size.Y, hrp0.Size.Z, tostring(isBigForm())), "B")
        end
        if not isBigForm() then
            log("⚠ ยังไม่ใช่ร่างใหญ่ — กด Z เองแล้ว Spy ใหม่ได้", "O")
        end

        local quest = pickQuestForLevel(getLevel())
        local mobName = quest and quest.mob or nil
        local mob = mobName and select(1, findMob(mobName)) or nil
        if not mob then
            local enemies = Workspace:FindFirstChild("Enemies")
            local bestD = math.huge
            local _, my = getChar()
            if enemies and my then
                for _, e in ipairs(enemies:GetChildren()) do
                    local hum = e:FindFirstChildOfClass("Humanoid")
                    local eh = e:FindFirstChild("HumanoidRootPart")
                    if hum and eh and hum.Health > 0 then
                        local d = (eh.Position - my.Position).Magnitude
                        if d < bestD then bestD = d; mob = e end
                    end
                end
            end
        end
        if not mob then
            log("ไม่เจอมอน — ยืนใกล้มอนก่อน (อย่าเปิด AUTO)", "R")
            return
        end

        local mhrp = mob:FindFirstChild("HumanoidRootPart")
        local mhum = mob:FindFirstChildOfClass("Humanoid")
        if not mhrp or not mhum then log("มอนไม่มี HRP", "R") return end

        -- จำพิกัดมอนตอนเริ่ม (ล็อกมอนนิ่งระหว่างวัด — กันเดินหนีทำให้ระยะเพี้ยน)
        local anchorPos = mhrp.Position
        log(string.format("เป้า: %s HP%.0f @ %.0f,%.0f,%.0f",
            mob.Name, mhum.Health, anchorPos.X, anchorPos.Y, anchorPos.Z), "G")
        log("วิธี: วางจากไกล 150→5 | รอดาเมจเข้าก่อนค่อยนับระยะจริง", "S")

        local height = isBigForm() and 18 or 8
        local firstHitD = nil -- ระยะจริงครั้งแรกที่โดน (จากไกลเข้ามา)
        local confirmed = {}

        -- วางตัว + รอให้ตำแหน่งนิ่ง (กันวาปเด้ง)
        local function settleAt(horizDist)
            mhrp = mob:FindFirstChild("HumanoidRootPart")
            mhum = mob:FindFirstChildOfClass("Humanoid")
            if not mhrp or not mhum or mhum.Health <= 0 then return nil end
            -- ตรึงมอนที่จุดเดิม
            pcall(function()
                mhrp.Anchored = true
                mhrp.CFrame = CFrame.new(anchorPos)
                mhum.PlatformStand = true
            end)
            local want = math.sqrt(height * height + horizDist * horizDist)
            for try = 1, 6 do
                local _, my = getChar()
                if not my then return nil end
                setNoclip()
                equipSword()
                my.CFrame = CFrame.new(anchorPos) * CFrame.new(0, height, horizDist)
                task.wait(0.25)
                _, my = getChar()
                mhrp = mob:FindFirstChild("HumanoidRootPart")
                if not my or not mhrp then return nil end
                -- ตรึงมอนอีกครั้งหลัง wait
                pcall(function()
                    mhrp.CFrame = CFrame.new(anchorPos)
                end)
                local real = (my.Position - mhrp.Position).Magnitude
                if math.abs(real - want) <= 18 then
                    return real, my
                end
                -- ยังเด้ง → วาร์ปซ้ำ
            end
            local _, my = getChar()
            mhrp = mob:FindFirstChild("HumanoidRootPart")
            if my and mhrp then
                return (my.Position - mhrp.Position).Magnitude, my
            end
            return nil
        end

        -- ไกล → ใกล้
        for horiz = 150, 5, -5 do
            if not mob.Parent or not mhum or mhum.Health <= 0 then
                log("มอนตายกลางทาง หยุดวัด", "Y")
                break
            end
            local realD = settleAt(horiz)
            if not realD then
                log(string.format("ข้ามตั้ง%d (วางตัวไม่ได้)", horiz), "O")
            else
                local want = math.sqrt(height * height + horiz * horiz)
                if math.abs(realD - want) > 25 then
                    log(string.format("~ ไม่นิ่ง ตั้ง%d อยาก=%.0f จริง=%.0f (ข้าม)", horiz, want, realD), "O")
                else
                    local hp0 = mhum.Health
                    for _ = 1, 10 do
                        -- ตรึงตำแหน่งระหว่างยิง
                        local _, my = getChar()
                        if my and mhrp then
                            my.CFrame = CFrame.new(anchorPos) * CFrame.new(0, height, horiz)
                            pcall(function() mhrp.CFrame = CFrame.new(anchorPos) end)
                        end
                        attackMob(mob)
                        task.wait(0.05)
                    end
                    task.wait(0.3)
                    mhum = mob:FindFirstChildOfClass("Humanoid")
                    if not mhum then break end
                    local dmg = hp0 - mhum.Health
                    -- วัดระยะจริงอีกครั้งหลังยิง
                    local _, my2 = getChar()
                    mhrp = mob:FindFirstChild("HumanoidRootPart")
                    local realAfter = (my2 and mhrp) and (my2.Position - mhrp.Position).Magnitude or realD
                    if dmg > 0 then
                        if not firstHitD then
                            firstHitD = realAfter
                            log(string.format("★ โดนครั้งแรก @ จริง=%.0f (ตั้งแนวนอน%d) ดาเมจ=-%.0f",
                                realAfter, horiz, dmg), "G")
                        else
                            log(string.format("✓ ยังโดน @ จริง=%.0f ดาเมจ=-%.0f", realAfter, dmg), "G")
                        end
                        confirmed[#confirmed + 1] = realAfter
                        -- โดนแล้วเก็บอีก 2 ขั้นใกล้เพื่อยืนยัน แล้วจบ
                        if #confirmed >= 3 then
                            log("ยืนยันโดน 3 ระยะแล้ว → จบ", "Y")
                            break
                        end
                    else
                        log(string.format("✗ ไม่โดน @ ตั้ง%d จริง=%.0f", horiz, realAfter), "S")
                        if firstHitD then
                            -- เคยโดนแล้วแต่ระยะนี้ไม่โดน = แปลก ข้าม
                        end
                    end
                end
            end
        end

        -- ปลดมอน
        pcall(function()
            if mhrp then mhrp.Anchored = false end
        end)

        log("──────── สรุป ────────", "Y")
        log(string.format("ร่างใหญ่=%s | สูงทดสอบ=%.0f | ไล่จากไกล→ใกล้", tostring(isBigForm()), height), "B")
        if firstHitD then
            local safe = math.floor(firstHitD * 0.85)
            log(string.format("★ ระยะตีโดนสูงสุด (ครั้งแรกจากไกล) ≈ %.0f studs", firstHitD), "G")
            log(string.format("  แนะนำล็อกตีที่ระยะ ≤ %.0f studs", safe), "G")
        else
            log("ไม่โดนเลย — เช็คดาบ/ร่างพระ หรือมอนตายก่อนวัด", "R")
        end
        log("════ จบ Spy ระยะตี v2 — Copy ส่งมาได้ ════", "Y")
    end)
    if not ok then log("Spy error: " .. tostring(err), "R") end
    spyBusy = false
end
