local noclipConn
local function setNoclip(on)
    if on == false then
        if noclipConn then noclipConn:Disconnect(); noclipConn = nil end
        return
    end
    local c = LP.Character
    if not c then return end
    for _, p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") then p.CanCollide = false end
    end
    if not noclipConn then
        noclipConn = RunService.Stepped:Connect(function()
            if not _G.BF04C_ON then return end
            local ch = LP.Character
            if not ch then return end
            for _, p in ipairs(ch:GetDescendants()) do
                if p:IsA("BasePart") then p.CanCollide = false end
            end
        end)
    end
end

local currentTween
local function flyTo(posOrCF, label)
    local _, hrp = getChar()
    if not hrp then return false end
    setNoclip(true)
    local dest = typeof(posOrCF) == "CFrame" and posOrCF
        or CFrame.new(posOrCF + Vector3.new(0, 5, 0))
    local dist = (hrp.Position - dest.Position).Magnitude
    log(string.format("บิน→%s %.0f studs", label or "เป้า", dist), "B")
    if dist < 18 then
        hrp.CFrame = dest
        return true
    end
    local spd = FLY_SPEED
    if dist > 1200 then spd = 180 end
    if currentTween then pcall(function() currentTween:Cancel() end) end
    local tw = TweenService:Create(hrp,
        TweenInfo.new(dist / spd, Enum.EasingStyle.Linear), { CFrame = dest })
    currentTween = tw
    tw:Play()
    tw.Completed:Wait()
    currentTween = nil
    local _, h2 = getChar()
    if not h2 then return false end
    local left = (h2.Position - dest.Position).Magnitude
    if left <= 50 then
        h2.CFrame = dest
        return true
    end
    log(string.format("ยังไม่ถึง (เหลือ %.0f)", left), "Y")
    return false
end

-- คืน { qName, qNum, mob, need, levelReq, displayName }
local function pickQuestForLevel(lv)
    local data = getQuests()
    if not data then return nil end
    local best = nil
    for qName, entry in pairs(data) do
        if type(entry) == "table" then
            for num, sub in pairs(entry) do
                if type(sub) == "table" and type(sub.LevelReq) == "number"
                   and sub.LevelReq <= lv and type(sub.Task) == "table" then
                    local mob, need = next(sub.Task)
                    if mob and type(need) == "number" then
                        local cand = {
                            qName = qName,
                            qNum = tonumber(num) or num,
                            mob = mob,
                            need = need,
                            levelReq = sub.LevelReq,
                            displayName = sub.Name or mob,
                        }
                        if not best or cand.levelReq > best.levelReq
                           or (cand.levelReq == best.levelReq and cand.qNum > best.qNum) then
                            best = cand
                        end
                    end
                end
            end
        end
    end
    return best
end

local function tryGetBestQuestNPC(timeoutSec)
    timeoutSec = timeoutSec or 3
    local rem = RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("GetBestQuestNPC")
    if not rem then return nil, "ไม่มี GetBestQuestNPC" end
    local done, res, err = false, nil, nil
    task.spawn(function()
        local ok, r = pcall(function() return rem:InvokeServer() end)
        done = true
        if ok then res = r else err = tostring(r) end
    end)
    local t0 = os.clock()
    while not done and os.clock() - t0 < timeoutSec do
        task.wait(0.05)
    end
    if not done then return nil, "timeout " .. timeoutSec .. "s (ค้าง?)" end
    if err then return nil, err end
    return res, nil
end

local function posFromAny(v)
    if typeof(v) == "Vector3" then return v end
    if typeof(v) == "CFrame" then return v.Position end
    if typeof(v) == "Instance" then
        if v:IsA("Model") then
            local h = v:FindFirstChild("HumanoidRootPart") or v:FindFirstChildWhichIsA("BasePart")
            return h and h.Position
        elseif v:IsA("BasePart") then
            return v.Position
        end
    end
    if type(v) == "table" then
        if typeof(v.Position) == "Vector3" then return v.Position end
        if typeof(v.CFrame) == "CFrame" then return v.CFrame.Position end
        if v[1] and typeof(v[1]) == "number" and v[3] then
            return Vector3.new(v[1], v[2], v[3])
        end
    end
end

local function dumpValue(v, depth)
    depth = depth or 0
    if depth > 2 then return "..." end
    local t = typeof(v)
    if t == "Instance" then
        local p = posFromAny(v)
        local extra = p and string.format(" @%.0f,%.0f,%.0f", p.X, p.Y, p.Z) or ""
        return v.ClassName .. ":" .. v.Name .. extra
    elseif t == "Vector3" then
        return string.format("V3(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CF(%.1f,%.1f,%.1f)", p.X, p.Y, p.Z)
    elseif t == "table" then
        local parts = {}
        local n = 0
        for k, vv in pairs(v) do
            n += 1
            if n > 10 then parts[#parts+1] = "..." break end
            parts[#parts+1] = tostring(k) .. "=" .. dumpValue(vv, depth + 1)
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    return tostring(v)
end

-- อ่าน UI เข็มทิศ — path จริง: Main.Guide (Spy ยืนยัน)
-- ระยะ = TopFrame.Subtitle2 เช่น "232m away"
-- ลูกศร = LeftFrame.IconFrame.GuideIcon
local function readCompassUI()
    local info = {
        texts = {}, distance = nil, title = nil,
        arrowRot = nil, arrowAbs = nil, arrowName = nil,
        navigateBtn = nil, frame = nil,
    }
    local main = pg:FindFirstChild("Main")
    if not main then return info end
    local guide = main:FindFirstChild("Guide")
    if guide then
        info.frame = guide
        local top = guide:FindFirstChild("TopFrame")
        if top then
            local sub1 = top:FindFirstChild("Subtitle1")
            local sub2 = top:FindFirstChild("Subtitle2")
            if sub1 and sub1:IsA("TextLabel") and tostring(sub1.Text) ~= "" then
                info.title = tostring(sub1.Text)
                info.texts[#info.texts+1] = info.title
            end
            if sub2 and sub2:IsA("TextLabel") then
                local t = tostring(sub2.Text)
                info.texts[#info.texts+1] = t
                -- "232m away" | "ห่างออกไป 232เมตร" | "232 m"
                local meters = t:match("(%d+)%s*m%s*away")
                    or t:match("(%d+)%s*เมตร")
                    or t:match("ห่างออกไป%s*(%d+)")
                    or t:match("^(%d+)%s*m")
                if meters then info.distance = tonumber(meters) end
            end
        end
        local track = guide:FindFirstChild("Track")
        if track then info.navigateBtn = track end
        local iconFrame = guide:FindFirstChild("LeftFrame")
            and guide.LeftFrame:FindFirstChild("IconFrame")
        if iconFrame then
            local needle = iconFrame:FindFirstChild("GuideIcon")
            if not needle then
                local bg = iconFrame:FindFirstChild("Background")
                needle = bg and bg:FindFirstChild("Arrow")
            end
            if needle then
                info.arrowRot = needle.Rotation
                info.arrowAbs = needle.AbsoluteRotation
                info.arrowName = needle:GetFullName():gsub("^.-PlayerGui%.", "")
            end
        end
        return info
    end
    return info
end

local function compassWorldDir(arrowRot)
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    local look = cam.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude < 0.05 then flat = Vector3.new(0, 0, -1) else flat = flat.Unit end
    local ang = math.rad(arrowRot or 0)
    local cf = CFrame.lookAt(Vector3.zero, flat)
    local dir = (cf * CFrame.Angles(0, -ang, 0)).LookVector
    return Vector3.new(dir.X, 0, dir.Z).Unit
end

-- "Find the Dragon Crew Quest Giver." → "Dragon Crew Quest Giver"
local function parseTargetNameFromTitle(title)
    if not title or title == "" then return nil end
    local t = tostring(title)
    local who = t:match("^[Ff]ind the%s+(.+)")
        or t:match("หา%s*(.+)")
        or t
    who = who:gsub("%.%s*$", ""):gsub("%s+$", ""):gsub("^%s+", "")
    if who == "" then return nil end
    return who
end

local function rememberNpc(part)
    if not part then return end
    local name = (part.Parent and part.Parent.Name) or part.Name
    NPC_CACHE[name:lower()] = part.Position
end

-- หาโมเดล/พาร์ทในโลกจากชื่อ (ตีกลับจากชื่อที่เข็มทิศบอก)
local function findNpcByExactName(targetName)
    if not targetName or targetName == "" then return nil end
    local want = targetName:lower()
    local _, myHrp = getChar()
    local best, bestD, bestName = nil, math.huge, nil

    local function consider(model)
        if not model then return end
        local nl = model.Name:lower()
        if nl ~= want and not nl:find(want, 1, true) and not want:find(nl, 1, true) then
            return
        end
        local part = model:FindFirstChild("HumanoidRootPart")
            or model.PrimaryPart
            or model:FindFirstChildWhichIsA("BasePart")
        if not part then return end
        local d = myHrp and (part.Position - myHrp.Position).Magnitude or 0
        if d < bestD then
            bestD = d; best = part; bestName = model.Name
        end
    end

    local folder = Workspace:FindFirstChild("NPCs")
    if folder then
        for _, npc in ipairs(folder:GetChildren()) do consider(npc) end
    end
    -- เผื่ออยู่นอกโฟลเดอร์ NPCs
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Model") then consider(obj) end
    end

    if best then
        rememberNpc(best)
        return best, bestD, bestName
    end
    return nil
end

local function npcFromTitle(title)
    local who = parseTargetNameFromTitle(title)
    if not who then return nil end
    local key = who:lower()
    if NPC_CACHE[key] then return NPC_CACHE[key], "cache:" .. who end
    if KNOWN_NPC[key] then return KNOWN_NPC[key], "known:" .. who end
    for k, v in pairs(NPC_CACHE) do
        if key:find(k, 1, true) or k:find(key, 1, true) then return v, "cache:" .. k end
    end
    for k, v in pairs(KNOWN_NPC) do
        if key:find(k, 1, true) or k:find(key, 1, true) then return v, "known:" .. k end
    end
    return nil
end

-- เปิดดู Guide ว่ามีการอ้างอิง Instance/Attribute ไป NPC ไหม
local function spyGuideLinks(guide)
    local hits = 0
    if not guide then return 0 end
    log("── ตีกลับ: เปิดไส้ Guide หาลิงก์ NPC ──", "Y")
    for _, d in ipairs(guide:GetDescendants()) do
        if d:IsA("ObjectValue") or d:IsA("NumberValue") or d:IsA("StringValue")
           or d:IsA("CFrameValue") or d:IsA("Vector3Value") then
            hits += 1
            local extra = ""
            pcall(function()
                if d:IsA("ObjectValue") and d.Value then
                    local v = d.Value
                    extra = " → " .. v:GetFullName()
                    local p = posFromAny(v)
                    if p then
                        extra ..= string.format(" @%.1f,%.1f,%.1f", p.X, p.Y, p.Z)
                        local nm = v.Name
                        if v:IsA("BasePart") and v.Parent then nm = v.Parent.Name end
                        NPC_CACHE[nm:lower()] = p
                        log("  ★ ObjectValue." .. d.Name .. extra, "G")
                    else
                        log("  ObjectValue." .. d.Name .. extra, "B")
                    end
                elseif d:IsA("CFrameValue") then
                    local cf = d.Value
                    extra = string.format(" CF=%.1f,%.1f,%.1f", cf.X, cf.Y, cf.Z)
                    log("  CFrameValue." .. d.Name .. extra, "G")
                elseif d:IsA("Vector3Value") then
                    local v = d.Value
                    extra = string.format(" V3=%.1f,%.1f,%.1f", v.X, v.Y, v.Z)
                    log("  Vector3Value." .. d.Name .. extra, "G")
                else
                    log("  " .. d.ClassName .. "." .. d.Name .. "=" .. tostring(d.Value):sub(1, 60), "S")
                end
            end)
        end
    end
    for name, val in pairs(guide:GetAttributes()) do
        hits += 1
        log(string.format("  attr Guide.%s=%s", name, tostring(val)), "B")
    end
    -- ViewportFrame อาจมีโมเดลเป้า
    local vp = guide:FindFirstChild("LeftFrame")
        and guide.LeftFrame:FindFirstChild("IconFrame")
        and guide.LeftFrame.IconFrame:FindFirstChild("ViewportFrame")
    if vp then
        for _, c in ipairs(vp:GetDescendants()) do
            if c:IsA("Model") or c:IsA("WorldModel") then
                hits += 1
                log("  Viewport มี " .. c.ClassName .. ":" .. c.Name, "B")
            end
            if c:IsA("ObjectValue") and c.Value then
                hits += 1
                log("  Viewport ObjectValue → " .. c.Value:GetFullName(), "G")
            end
        end
    end
    if hits == 0 then log("  ไม่เจอ ObjectValue/Attribute ใน Guide (เกมซ่อนในสคริปต์)", "O") end
    return hits
end

-- หัวใจ: ชื่อจากเข็มทิศ → พิกัด
local function reverseCompassToPos(quest)
    local ui = readCompassUI()
    local targetName = parseTargetNameFromTitle(ui.title)
    if ui.title then log("เข็มทิศข้อความ: " .. ui.title, "O") end
    if targetName then
        log("→ ตีกลับชื่อเป้า: [" .. targetName .. "]", "Y")
    else
        log("→ ยังอ่านชื่อจาก Guide ไม่ได้", "R")
    end

    -- 1) หาในแมพตามชื่อที่เข็มทิศบอก
    if targetName then
        local part, dist, realName = findNpcByExactName(targetName)
        if part then
            log(string.format("→ เจอในแมพ: %s (%.0f studs) @ %.1f,%.1f,%.1f",
                realName or targetName, dist or 0, part.Position.X, part.Position.Y, part.Position.Z), "G")
            return part.Position, "name→NPCs:" .. (realName or targetName)
        end
        log("→ ยังไม่โหลดในแมพ: " .. targetName, "O")
    end

    -- 2) cache / known จากชื่อเดิม
    local cached, csrc = npcFromTitle(ui.title)
    if cached then
        log(string.format("→ ใช้พิกัดจำไว้ (%s) @ %.0f,%.0f,%.0f",
            csrc, cached.X, cached.Y, cached.Z), "G")
        return cached, csrc
    end

    -- 3) สำรองจากเควส (Dragon Crew ฯลฯ)
    if quest then

-- บินทีละก้าวตามเข็มทิศ — ใช้เมื่อยังไม่มีชื่อ/พิกัดจำ
local function flyByCompassStep()
    local ui = readCompassUI()
    local d0 = ui.distance
    local rot = ui.arrowAbs or ui.arrowRot
    if not d0 or rot == nil then return false, "ไม่มีระยะ/ลูกศรจาก Guide UI" end

    local cached, csrc = npcFromTitle(ui.title)
    if cached then
        log(string.format("มีพิกัดแล้ว (%s) — ไม่เดาลูกศร", csrc), "G")
        flyTo(CFrame.new(cached + Vector3.new(0, 6, 0)), "NPC/" .. csrc)
        compassStuck = 0
        return true, 0
    end
    if d0 <= 25 then
        return false, "ใกล้แล้ว (" .. d0 .. "ม.) รอ NPC โหลด"
    end

    if ui.navigateBtn then
        pcall(function() firesignal(ui.navigateBtn.MouseButton1Click) end)
    end

    local dir = compassWorldDir(rot)
    local _, hrp = getChar()
    if not dir or not hrp then return false, "ไม่มีทิศ/ตัวละคร" end

    local hopStuds
    if d0 > 150 then
        hopStuds = math.clamp(d0 * 3 * 0.32, 80, 260)
    elseif d0 > 100 then
        hopStuds = math.clamp(d0 * 3 * 0.22, 45, 120)
    else
        hopStuds = math.clamp(d0 * 3 * 0.15, 28, 70)
    end

    local dest = hrp.Position + dir * hopStuds + Vector3.new(0, 35, 0)
    log(string.format("ตามเข็มทิศ: %dม. → ก้าว %.0f studs (rot=%.0f)", d0, hopStuds, rot), "B")
    flyTo(CFrame.new(dest), "เข็มทิศ")
    task.wait(0.4)
    local ui2 = readCompassUI()
    local d1 = ui2.distance
    if d1 and d1 < d0 - 1 then
        log(string.format("เมตรลด %d → %d ✓", d0, d1), "G")
        compassStuck = 0
        lastMeter = d1
        -- ใกล้แล้วลองหาชื่ออีกครั้ง
        local tn = parseTargetNameFromTitle(ui2.title)
        if tn then findNpcByExactName(tn) end
        return true, d1
    end
    if d1 and d1 >= d0 then
        log(string.format("เมตรไม่ลด %d → %d ลองทิศกลับ", d0, d1), "O")
        local _, h2 = getChar()
        if h2 then
            local back = h2.Position - dir * hopStuds + Vector3.new(0, 35, 0)
            flyTo(CFrame.new(back), "เข็มทิศกลับ")
            task.wait(0.3)
            local ui3 = readCompassUI()
            if ui3.distance and ui3.distance < d0 - 1 then
                log(string.format("ทิศกลับแล้วเมตร=%d ✓", ui3.distance), "G")
                compassStuck = 0
                lastMeter = ui3.distance
                return true, ui3.distance
            end
        end
        if lastMeter and d1 and math.abs(d1 - lastMeter) <= 4 then
            compassStuck += 1
            log("วนเมตร≈" .. tostring(d1) .. " stuck=" .. compassStuck, "Y")
        end
        lastMeter = d1 or d0
    end
    return true, d1 or d0
end

-- หา Quest Giver จากเควส (สำรอง ถ้า Guide ไม่มีชื่อ)
local function findQuestGiver(quest)
    local folder = Workspace:FindFirstChild("NPCs")
    if not folder then return nil end
    local _, myHrp = getChar()
    local best, bestD = nil, math.huge
    local keys = {}
    if quest then
        local mob = quest.mob or ""
        local words = {}
        for w in mob:gmatch("%S+") do words[#words+1] = w end
        if #words >= 2 then
            keys[#keys+1] = words[1] .. " " .. words[2]
        end
        keys[#keys+1] = mob
        keys[#keys+1] = quest.displayName or ""
        keys[#keys+1] = quest.qName or ""
    end
    keys[#keys+1] = "Quest Giver"
    for _, npc in ipairs(folder:GetChildren()) do
        local nl = npc.Name:lower()
        local hit = nl:find("quest giver", 1, true) or nl:find("questgiver", 1, true)
        if not hit and quest then
            for _, k in ipairs(keys) do
                if k ~= "" and nl:find(k:lower(), 1, true) then hit = true break end
            end
        end
        if hit then
            local part = npc:FindFirstChild("HumanoidRootPart")
                or npc:FindFirstChildWhichIsA("BasePart")
            if part then
                local d = myHrp and (part.Position - myHrp.Position).Magnitude or 0
                if d < bestD then bestD = d; best = part end
            end
        end
    end
    return best, bestD
end

-- รวมแหล่งหาเป้า: ตีกลับชื่อจากเข็มทิศเป็นหลัก
local function resolveNpcTarget(quest)
    local pos, srcOrUi = reverseCompassToPos(quest)
    if pos then return pos, srcOrUi end

    local ui = type(srcOrUi) == "table" and srcOrUi or readCompassUI()
    local rot = ui.arrowAbs or ui.arrowRot
    if ui.distance and rot ~= nil and ui.distance > 5 then
        return "COMPASS_STEP", "compassUI(ยังไม่รู้พิกัด)"
    end
    return nil, "ยังไม่มีชื่อ/พิกัดจากเข็มทิศ"
end

local function findMob(mobName)
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return nil end
    local _, myHrp = getChar()
    local best, bestD = nil, math.huge
    for _, e in ipairs(enemies:GetChildren()) do
        if e.Name:find(mobName, 1, true) then
            local hum = e:FindFirstChildOfClass("Humanoid")
            local hrp = e:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local d = myHrp and (hrp.Position - myHrp.Position).Magnitude or 0
                if d < bestD then
                    bestD = d; best = e
                    MOB_CACHE[mobName:lower()] = hrp.Position
                end
            end
        end
    end
    return best, bestD
end

local function mobSpawnHint(mobName)
    local key = (mobName or ""):lower()
    if MOB_CACHE[key] then return MOB_CACHE[key], "cache" end
    if KNOWN_MOB[key] then return KNOWN_MOB[key], "known" end
    return nil
end

local function getToolByTip(tip)
    local char = LP.Character
    local bp = LP:FindFirstChild("Backpack")
    for _, src in ipairs({ char, bp }) do
        if src then
            for _, t in ipairs(src:GetChildren()) do
                if t:IsA("Tool") and tostring(t.ToolTip or "") == tip then return t end
            end
        end
    end
end

local function equipSword()
    local char, _, hum = getChar()
    if not char or not hum then return false end
    local held = char:FindFirstChildOfClass("Tool")
    if held and tostring(held.ToolTip or "") == "Sword" then return true end
    local sword = getToolByTip("Sword")
    if sword then
        pcall(function() hum:EquipTool(sword) end)
        task.wait(0.15)
        return true
    end
    -- fallback กด 3 ตามที่ใช้อยู่
    pressKey(Enum.KeyCode.Three, 0.08)
    task.wait(0.2)
    held = char:FindFirstChildOfClass("Tool")
    return held ~= nil
end

local function isBigForm()
    local _, hrp = getChar()
    if not hrp then return false end
    return hrp.Size.Magnitude >= 5 or hrp.Size.Y >= 3.5
end

                log(string.format("  attr %s=%s", name, tostring(val)), "S")
            end
            for _, c in ipairs(qtl:GetDescendants()) do
                if c:IsA("Attachment") or c:IsA("Beam") or c:IsA("BasePart") then
                    local p = posFromAny(c)
                    log(string.format("  %s:%s%s", c.ClassName, c.Name,
                        p and string.format(" @%.0f,%.0f,%.0f", p.X, p.Y, p.Z) or ""), "B")
                end
            end
        else
            log("ไม่มี QuestTracingLine ใน Effect.Container", "O")
        end
    end)
    local ui2 = readCompassUI()
    local knownP, knownS = npcFromTitle(ui2.title)
    if knownP then
        log(string.format("พิกัดจาก cache/known (%s): %.1f, %.1f, %.1f",
            knownS, knownP.X, knownP.Y, knownP.Z), "G")
    else
        log("ยังไม่มีพิกัด cache สำหรับหัวข้อนี้ (จะจำตอนเจอใน NPCs)", "O")
    end
    log("════ จบ Dump — Copy ส่งมาได้ ════", "Y")
end

-- ── Main loop ──────────────────────────────────────────────────────
local function mainLoop()
    log("=== START Compass Auto ===", "Y")
    local cf = getCommF()
    if not cf then log("ไม่มี CommF_", "R") return end
    getCombatRemotes()
    pcall(function() ensureBuddhaSword(true) end)

    while _G.BF04C_ON do
        local ok, err = pcall(function()
            local lv = getLevel()
            local quest = pickQuestForLevel(lv)
            if not quest then
                log("ไม่มีเควสที่ LevelReq <= " .. lv, "R")
                task.wait(3)
                return
            end
            refreshStatus(quest)

            -- ── ยังไม่มีเควส → หา NPC / ตามเข็มทิศ ──
            if not questUIVisible() then
                local pos, src = resolveNpcTarget(quest)
                if pos == "COMPASS_STEP" then
                    setNoclip(true)
                    flyByCompassStep()
                    -- หลังบิน ลองหา NPC อีกรอบ
                    local part2 = findQuestGiver(quest)
                    if part2 then
                        log("โหลด NPC แล้วหลังตามเข็มทิศ", "G")
                    end
                    task.wait(0.2)
                    return
                end
                if not pos then
                    log("รอเป้า NPC... (" .. tostring(src) .. ")", "Y")
                    local mob, md = findMob(quest.mob)
                    if mob and md and md > 80 then
                        local mhrp = mob:FindFirstChild("HumanoidRootPart")
                        if mhrp then flyTo(mhrp.CFrame * CFrame.new(0, 20, 0), "ใกล้เกาะมอน") end
                    end
                    task.wait(1.5)
                    return
                end
                local _, hrp = getChar()
                local dist = hrp and (hrp.Position - pos).Magnitude or 9999
                local nearDist = isBigForm() and 55 or 28
                if dist > nearDist then
                    flyTo(CFrame.new(pos + Vector3.new(0, 6, 0)), "NPC/" .. src)
                    return
                end
                equipSword() -- ถือดาบ ไม่ถือผล ตอนรับเควส
                for _ = 1, 4 do
                    local _, h = getChar()
                    if h then h.CFrame = CFrame.new(pos + Vector3.new(0, 4, 0)) end
                    task.wait(0.12)
                end
                local _, hNow = getChar()
                local dNow = hNow and (hNow.Position - pos).Magnitude or -1
                log(string.format("StartQuest(%s, %s) ร่างใหญ่=%s ระยะ=%.0f",
                    quest.qName, tostring(quest.qNum), tostring(isBigForm()), dNow), "Y")
                local iok, ires = pcall(function()
                    return cf:InvokeServer("StartQuest", quest.qName, quest.qNum)
                end)
                log("ผล: ok=" .. tostring(iok) .. " res=" .. tostring(ires), "B")
                local w = 0
                while not questUIVisible() and w < 25 and _G.BF04C_ON do
                    local _, h = getChar()
                    if h then h.CFrame = CFrame.new(pos + Vector3.new(0, 4, 0)) end
                    task.wait(0.1); w += 1
                end
                if questUIVisible() then
                    log("รับเควสสำเร็จ: " .. questTitle(), "G")
                else
                    log("ยังไม่เปิด UI (ลองใหม่)", "R")
                    task.wait(1.2)
                end
                return
            end

            local titleTxt = questTitle()
            if titleTxt ~= "" and not titleTxt:lower():find(quest.mob:lower():sub(1, 8), 1, true)
               and not titleTxt:find(quest.displayName or "", 1, true) then
                log("เควสไม่ตรง (" .. titleTxt .. ") → Abandon", "Y")
                pcall(function() cf:InvokeServer("AbandonQuest") end)
                task.wait(0.8)
                return
            end

            -- ก่อนตี: ให้มีร่างพระ + ดาบ
            ensureBuddhaSword(false)

            local mob, md = findMob(quest.mob)
            if not mob then
                log("ไม่เจอ " .. quest.mob .. " ใน Enemies → รอโหลด", "Y")
                local hint, src = mobSpawnHint(quest.mob)
                if hint then
                    log("บินไปจุดมอน (" .. src .. ") บังคับสตรีม", "B")
                    flyTo(CFrame.new(hint + Vector3.new(0, 40, 0)), "จุดมอน/" .. src)
                else
                    local npcP = select(1, npcFromTitle("Find the Dragon Crew Quest Giver"))
                        or KNOWN_NPC["dragon crew quest giver"]
                    if npcP then
                        flyTo(CFrame.new(npcP + Vector3.new(80, 40, -80)), "ใกล้เกาะNPC")
                    end
                end
                task.wait(1.0)
                return
            end
            local mhrp = mob:FindFirstChild("HumanoidRootPart")
            local mhum = mob:FindFirstChildOfClass("Humanoid")
            if not mhrp or not mhum then return end

            if (md or 999) > SAFE_HIT_RANGE then
                flyTo(mhrp.CFrame * CFrame.new(0, attackHeight(), 0), "มอน")
                return
            end

            log(string.format("สู้ %s HP%.0f (d=%.0f big=%s)",
                mob.Name, mhum.Health, md or 0, tostring(isBigForm())), "Y")
            local lastHp = mhum.Health
            local lastCheck = os.clock()
            local noDmg = 0
            while mob.Parent and mhum.Parent and mhum.Health > 0
                  and questUIVisible() and _G.BF04C_ON do
                local _, myHrp = getChar()
                if not myHrp then break end
                mhrp = mob:FindFirstChild("HumanoidRootPart")
                if not mhrp then break end
                setNoclip()
                equipSword()
                -- ล็อกเหนือหัว → ระยะจริง ≈ attackHeight (≤34 จาก Spy)
                myHrp.CFrame = mhrp.CFrame * CFrame.new(0, attackHeight(), 0)
                local nowD = (myHrp.Position - mhrp.Position).Magnitude
                if nowD > SAFE_HIT_RANGE then
                    -- เด้งออกนอกระยะ → วาร์ปกลับทันที
                    myHrp.CFrame = mhrp.CFrame * CFrame.new(0, attackHeight(), 0)
                end
                pcall(function()
                    -- hitbox เสริมเล็กน้อย (ไม่พึ่งเป็นหลัก — Spy วัดโดยไม่ขยาย)
                    mhrp.Size = Vector3.new(25, 25, 25)
                    mhrp.CanCollide = false
                    mhum.PlatformStand = true
                    mhum.Sit = true
                end)
                attackMob(mob)
                if os.clock() - lastCheck >= 1.2 then
                    lastCheck = os.clock()
                    local hp = mhum.Health
                    if hp >= lastHp then
                        noDmg += 1
                        if noDmg >= 3 then
                            log("ตีไม่เข้า → ข้ามตัวนี้", "Y")
                            break
                        end
                    else
                        noDmg = 0
                        lastHp = hp
                    end
                end
                task.wait()
            end
            log(mob.Name .. " จบรอบ", "G")
            task.wait(0.3)
        end)
        if not ok then
            log("Loop error: " .. tostring(err), "R")
            task.wait(1)
        else
            task.wait(0.15)
        end
    end
    log("=== STOP ===", "R")
end

-- ── Spy ระยะตี: จำตำแหน่งมอน → เริ่มจากไกลสุด → ค่อยเข้าใกล้ → โดนครั้งแรก = ระยะสูงสุด ──
local spyBusy = false
local function spyAttackRange()
    if spyBusy then log("กำลังวัดอยู่ รอจบก่อน", "Y") return end
    spyBusy = true
    local ok, err = pcall(function()
        log("════ Spy ระยะตี v2 (ไกล→ใกล้) ════", "Y")
        _G.BF04C_ON = false
        autoBtn.Text = "AUTO: OFF"
        autoBtn.BackgroundColor3 = Color3.fromRGB(30, 120, 50)
        task.wait(0.4) -- ให้ AUTO หยุดบินก่อน

        ensureBuddhaSword(true)
        local _, hrp0 = getChar()
        if hrp0 then
            log(string.format("HRP Size=%.1f,%.1f,%.1f | big=%s",
                hrp0.Size.X, hrp0.Size.Y, hrp0.Size.Z, tostring(isBigForm())), "B")
        end
        if not isBigForm() then
            log("⚠ ยังไม่ใช่ร่างใหญ่ — กด Z เองแล้ว Spy ใหม่ได้", "O")
        end

        local quest = pickQuestForLevel(getLevel())
        local mobName = quest and quest.mob or nil
        local mob = mobName and select(1, findMob(mobName)) or nil
        if not mob then
            local enemies = Workspace:FindFirstChild("Enemies")
            local bestD = math.huge
            local _, my = getChar()
            if enemies and my then
                for _, e in ipairs(enemies:GetChildren()) do
                    local hum = e:FindFirstChildOfClass("Humanoid")
                    local eh = e:FindFirstChild("HumanoidRootPart")
                    if hum and eh and hum.Health > 0 then
                        local d = (eh.Position - my.Position).Magnitude
                        if d < bestD then bestD = d; mob = e end
                    end
                end
            end
        end
        if not mob then
            log("ไม่เจอมอน — ยืนใกล้มอนก่อน (อย่าเปิด AUTO)", "R")
            return
        end

        local mhrp = mob:FindFirstChild("HumanoidRootPart")
        local mhum = mob:FindFirstChildOfClass("Humanoid")
        if not mhrp or not mhum then log("มอนไม่มี HRP", "R") return end

        -- จำพิกัดมอนตอนเริ่ม (ล็อกมอนนิ่งระหว่างวัด — กันเดินหนีทำให้ระยะเพี้ยน)
        local anchorPos = mhrp.Position
        log(string.format("เป้า: %s HP%.0f @ %.0f,%.0f,%.0f",
            mob.Name, mhum.Health, anchorPos.X, anchorPos.Y, anchorPos.Z), "G")
        log("วิธี: วางจากไกล 150→5 | รอดาเมจเข้าก่อนค่อยนับระยะจริง", "S")

        local height = isBigForm() and 18 or 8
        local firstHitD = nil -- ระยะจริงครั้งแรกที่โดน (จากไกลเข้ามา)
        local confirmed = {}

        -- วางตัว + รอให้ตำแหน่งนิ่ง (กันวาปเด้ง)
        local function settleAt(horizDist)
            mhrp = mob:FindFirstChild("HumanoidRootPart")
            mhum = mob:FindFirstChildOfClass("Humanoid")
            if not mhrp or not mhum or mhum.Health <= 0 then return nil end
            -- ตรึงมอนที่จุดเดิม
            pcall(function()
                mhrp.Anchored = true
                mhrp.CFrame = CFrame.new(anchorPos)
                mhum.PlatformStand = true
            end)
            local want = math.sqrt(height * height + horizDist * horizDist)
            for try = 1, 6 do
                local _, my = getChar()
                if not my then return nil end
                setNoclip()
                equipSword()
                my.CFrame = CFrame.new(anchorPos) * CFrame.new(0, height, horizDist)
                task.wait(0.25)
                _, my = getChar()
                mhrp = mob:FindFirstChild("HumanoidRootPart")
                if not my or not mhrp then return nil end
                -- ตรึงมอนอีกครั้งหลัง wait
                pcall(function()
                    mhrp.CFrame = CFrame.new(anchorPos)
                end)
                local real = (my.Position - mhrp.Position).Magnitude
                if math.abs(real - want) <= 18 then
                    return real, my
                end
                -- ยังเด้ง → วาร์ปซ้ำ
            end
            local _, my = getChar()
            mhrp = mob:FindFirstChild("HumanoidRootPart")
            if my and mhrp then
                return (my.Position - mhrp.Position).Magnitude, my
            end
            return nil
        end

        -- ไกล → ใกล้
        for horiz = 150, 5, -5 do
            if not mob.Parent or not mhum or mhum.Health <= 0 then
                log("มอนตายกลางทาง หยุดวัด", "Y")
                break
            end
            local realD = settleAt(horiz)
            if not realD then
                log(string.format("ข้ามตั้ง%d (วางตัวไม่ได้)", horiz), "O")
            else
                local want = math.sqrt(height * height + horiz * horiz)
                if math.abs(realD - want) > 25 then
                    log(string.format("~ ไม่นิ่ง ตั้ง%d อยาก=%.0f จริง=%.0f (ข้าม)", horiz, want, realD), "O")
                else
                    local hp0 = mhum.Health
                    for _ = 1, 10 do
                        -- ตรึงตำแหน่งระหว่างยิง
                        local _, my = getChar()
                        if my and mhrp then
                            my.CFrame = CFrame.new(anchorPos) * CFrame.new(0, height, horiz)
                            pcall(function() mhrp.CFrame = CFrame.new(anchorPos) end)
                        end
                        attackMob(mob)
                        task.wait(0.05)
                    end
                    task.wait(0.3)
                    mhum = mob:FindFirstChildOfClass("Humanoid")
                    if not mhum then break end
                    local dmg = hp0 - mhum.Health
                    -- วัดระยะจริงอีกครั้งหลังยิง
                    local _, my2 = getChar()
                    mhrp = mob:FindFirstChild("HumanoidRootPart")
                    local realAfter = (my2 and mhrp) and (my2.Position - mhrp.Position).Magnitude or realD
                    if dmg > 0 then
                        if not firstHitD then
                            firstHitD = realAfter
                            log(string.format("★ โดนครั้งแรก @ จริง=%.0f (ตั้งแนวนอน%d) ดาเมจ=-%.0f",
                                realAfter, horiz, dmg), "G")
                        else
                            log(string.format("✓ ยังโดน @ จริง=%.0f ดาเมจ=-%.0f", realAfter, dmg), "G")
                        end
                        confirmed[#confirmed + 1] = realAfter
                        -- โดนแล้วเก็บอีก 2 ขั้นใกล้เพื่อยืนยัน แล้วจบ
                        if #confirmed >= 3 then
                            log("ยืนยันโดน 3 ระยะแล้ว → จบ", "Y")
                            break
                        end
                    else
                        log(string.format("✗ ไม่โดน @ ตั้ง%d จริง=%.0f", horiz, realAfter), "S")
                        if firstHitD then
                            -- เคยโดนแล้วแต่ระยะนี้ไม่โดน = แปลก ข้าม
                        end
                    end
                end
            end
        end

        -- ปลดมอน
        pcall(function()
            if mhrp then mhrp.Anchored = false end
        end)

        log("──────── สรุป ────────", "Y")
        log(string.format("ร่างใหญ่=%s | สูงทดสอบ=%.0f | ไล่จากไกล→ใกล้", tostring(isBigForm()), height), "B")
        if firstHitD then
            local safe = math.floor(firstHitD * 0.85)
            log(string.format("★ ระยะตีโดนสูงสุด (ครั้งแรกจากไกล) ≈ %.0f studs", firstHitD), "G")
            log(string.format("  แนะนำล็อกตีที่ระยะ ≤ %.0f studs", safe), "G")
        else
            log("ไม่โดนเลย — เช็คดาบ/ร่างพระ หรือมอนตายก่อนวัด", "R")
        end
        log("════ จบ Spy ระยะตี v2 — Copy ส่งมาได้ ════", "Y")
    end)
    if not ok then log("Spy error: " .. tostring(err), "R") end
    spyBusy = false
end
