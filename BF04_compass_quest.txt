local noclipConn
local function setNoclip(on)
    if on == false then
        if noclipConn then noclipConn:Disconnect(); noclipConn = nil end
        return
    end
    local c = LP.Character
    if not c then return end
    for _, p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") then p.CanCollide = false end
    end
    if not noclipConn then
        noclipConn = RunService.Stepped:Connect(function()
            if not _G.BF04C_ON then return end
            local ch = LP.Character
            if not ch then return end
            for _, p in ipairs(ch:GetDescendants()) do
                if p:IsA("BasePart") then p.CanCollide = false end
            end
        end)
    end
end

local currentTween
local flyLock = false

local function stopFlyTween()
    if currentTween then
        pcall(function() currentTween:Cancel() end)
        currentTween = nil
    end
end

local function hardStopFly()
    stopFlyTween()
    flyLock = false
end

local function flyTo(posOrCF, label)
    local _, hrp = getChar()
    if not hrp then return false end
    if flyLock then
        local t0 = os.clock()
        while flyLock and os.clock() - t0 < 8 and _G.BF04C_ON and _G.BF04C_GEN == MY_GEN do
            task.wait(0.1)
        end
        if flyLock then hardStopFly() end
    end
    flyLock = true
    setNoclip(true)
    local dest = typeof(posOrCF) == "CFrame" and posOrCF
        or CFrame.new(posOrCF + Vector3.new(0, 5, 0))
    local dist = (hrp.Position - dest.Position).Magnitude
    log(string.format("บิน→%s %.0f studs", label or "เป้า", dist), "B")

    local ok = false
    local okRun, err = pcall(function()
        if not _G.BF04C_ON or _G.BF04C_GEN ~= MY_GEN then return end
        if dist < 25 then
            hrp.CFrame = dest
            hrp.AssemblyLinearVelocity = Vector3.zero
            task.wait(0.2)
            ok = true
            return
        end
        -- ไกล = ช้า + แบ่งท่อน กัน rubberband ดีดกลับ
        local spd = FLY_SPEED
        if label == "มอน" or (label and label:find("มอน", 1, true)) then
            spd = FLY_SPEED_COMBAT
        elseif dist >= 3000 then spd = FLY_SPEED_LONG
        elseif dist >= 1500 then spd = 110
        elseif dist >= 800 then spd = 125
        end
        local chunk = (dist >= 3000) and 450 or ((dist >= 1500) and 550 or 700)
        local guard = 0
        while _G.BF04C_ON and _G.BF04C_GEN == MY_GEN and guard < 60 do
            guard = guard + 1
            local _, h = getChar()
            if not h then break end
            hrp = h
            setNoclip(true)
            local left = (hrp.Position - dest.Position).Magnitude
            if left <= 35 then
                hrp.CFrame = dest
                hrp.AssemblyLinearVelocity = Vector3.zero
                break
            end
            local step = math.min(chunk, left)
            local dir = (dest.Position - hrp.Position)
            if dir.Magnitude < 1 then break end
            local nxt = hrp.Position + dir.Unit * step
            local dur = math.clamp(step / spd, 0.3, 8)
            stopFlyTween()
            local look = Vector3.new(dir.X, 0, dir.Z)
            local lookCF = look.Magnitude > 1 and CFrame.new(nxt, nxt + look) or CFrame.new(nxt)
            local tw = TweenService:Create(hrp,
                TweenInfo.new(dur, Enum.EasingStyle.Linear),
                { CFrame = lookCF })
            currentTween = tw
            tw:Play()
            local tEnd = os.clock() + dur + 0.4
            while currentTween == tw and os.clock() < tEnd
                  and _G.BF04C_ON and _G.BF04C_GEN == MY_GEN do
                setNoclip(true)
                local _, h2 = getChar()
                if h2 then
                    h2.AssemblyLinearVelocity = Vector3.zero
                    hrp = h2
                end
                task.wait(0.05)
            end
            stopFlyTween()
            task.wait(0.12)
        end
        for _ = 1, 4 do
            local _, h2 = getChar()
            if not h2 then break end
            setNoclip(true)
            h2.CFrame = dest
            h2.AssemblyLinearVelocity = Vector3.zero
            task.wait(0.25)
            if (h2.Position - dest.Position).Magnitude <= 90 then
                ok = true
                break
            end
        end
        local _, h3 = getChar()
        if h3 then
            local left = (h3.Position - dest.Position).Magnitude
            ok = left <= 120
            if not ok then
                log(string.format("บินไม่ติด (เหลือ %.0f) — จะลองใหม่", left), "Y")
            end
        end
    end)
    flyLock = false
    if not okRun then log("fly error: " .. tostring(err), "R") end
    return ok
end

-- InvokeServer มี timeout กันค้างเงียบ
local function invokeCF(cf, ...)
    local args = { ... }
    local done, res, err = false, nil, nil
    task.spawn(function()
        local ok, a, b = pcall(function()
            return cf:InvokeServer(table.unpack(args))
        end)
        if ok then res = a else err = a end
        done = true
    end)
    local t0 = os.clock()
    while not done and os.clock() - t0 < 6 and _G.BF04C_ON and _G.BF04C_GEN == MY_GEN do
        task.wait(0.1)
    end
    if not done then
        return false, "timeout"
    end
    if err then return false, err end
    return true, res
end

-- คืน { qName, qNum, mob, need, levelReq, displayName }
local bossSkipUntil = 0
local bossMissCount = 0

local function isBossQuest(q)
    if not q then return false end
    if type(q.need) == "number" and q.need == 1 then return true end
    local m = (q.mob or ""):lower()
    return m:find("leader", 1, true) ~= nil
        or m:find("boss", 1, true) ~= nil
        or m:find("king", 1, true) ~= nil
        or m:find("admiral", 1, true) ~= nil
end

local function pickQuestForLevel(lv, opts)
    opts = opts or {}
    local skipBoss = opts.skipBoss
    if skipBoss == nil then skipBoss = os.clock() < bossSkipUntil end
    local preferName = opts.qName
    local maxQNum = opts.maxQNum
    local data = getQuests()
    if not data then return nil end
    local best = nil
    for qName, entry in pairs(data) do
        if type(entry) == "table" and (not preferName or qName == preferName) then
            for num, sub in pairs(entry) do
                if type(sub) == "table" and type(sub.LevelReq) == "number"
                   and sub.LevelReq <= lv and type(sub.Task) == "table" then
                    local mob, need = next(sub.Task)
                    local qn = tonumber(num) or num
                    if mob and type(need) == "number"
                       and not (maxQNum and type(qn) == "number" and qn > maxQNum) then
                        local cand = {
                            qName = qName,
                            qNum = qn,
                            mob = mob,
                            need = need,
                            levelReq = sub.LevelReq,
                            displayName = sub.Name or mob,
                        }
                        if not (skipBoss and isBossQuest(cand)) then
                            if not best or cand.levelReq > best.levelReq
                               or (cand.levelReq == best.levelReq and cand.qNum > best.qNum) then
                                best = cand
                            end
                        end
                    end
                end
            end
        end
    end
    if not best and preferName then
        return pickQuestForLevel(lv, { skipBoss = skipBoss, maxQNum = maxQNum })
    end
    return best
end

-- เควสก่อนหน้า (บอสยังไม่เกิด → ฟาร์มธรรมดา)
local function pickPreviousFarmQuest(lv, current)
    if current then
        local qn = tonumber(current.qNum)
        if qn and qn > 1 then
            local prev = pickQuestForLevel(lv, {
                skipBoss = true,
                qName = current.qName,
                maxQNum = qn - 1,
            })
            if prev then return prev end
        end
    end
    return pickQuestForLevel(lv, { skipBoss = true })
end

-- ── หาเป้าเข็มทิศ / NPC ────────────────────────────────────────────
local function tryGetBestQuestNPC(timeoutSec)
    timeoutSec = timeoutSec or 3
    local rem = RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("GetBestQuestNPC")
    if not rem then return nil, "ไม่มี GetBestQuestNPC" end
    local done, res, err = false, nil, nil
    task.spawn(function()
        local ok, r = pcall(function() return rem:InvokeServer() end)
        done = true
        if ok then res = r else err = tostring(r) end
    end)
    local t0 = os.clock()
    while not done and os.clock() - t0 < timeoutSec do
        task.wait(0.05)
    end
    if not done then return nil, "timeout " .. timeoutSec .. "s (ค้าง?)" end
    if err then return nil, err end
    return res, nil
end

local function posFromAny(v)
    if typeof(v) == "Vector3" then return v end
    if typeof(v) == "CFrame" then return v.Position end
    if typeof(v) == "Instance" then
        if v:IsA("Model") then
            local h = v:FindFirstChild("HumanoidRootPart") or v:FindFirstChildWhichIsA("BasePart")
            return h and h.Position
        elseif v:IsA("BasePart") then
            return v.Position
        end
    end
    if type(v) == "table" then
        if typeof(v.Position) == "Vector3" then return v.Position end
        if typeof(v.CFrame) == "CFrame" then return v.CFrame.Position end
        if v[1] and typeof(v[1]) == "number" and v[3] then
            return Vector3.new(v[1], v[2], v[3])
        end
    end
end

local function dumpValue(v, depth)
    depth = depth or 0
    if depth > 2 then return "..." end
    local t = typeof(v)
    if t == "Instance" then
        local p = posFromAny(v)
        local extra = p and string.format(" @%.0f,%.0f,%.0f", p.X, p.Y, p.Z) or ""
        return v.ClassName .. ":" .. v.Name .. extra
    elseif t == "Vector3" then
        return string.format("V3(%.1f,%.1f,%.1f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        local p = v.Position
        return string.format("CF(%.1f,%.1f,%.1f)", p.X, p.Y, p.Z)
    elseif t == "table" then
        local parts = {}
        local n = 0
        for k, vv in pairs(v) do
            n += 1
            if n > 10 then parts[#parts+1] = "..." break end
            parts[#parts+1] = tostring(k) .. "=" .. dumpValue(vv, depth + 1)
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    return tostring(v)
end

-- อ่าน UI เข็มทิศ — path จริง: Main.Guide (Spy ยืนยัน)
-- ระยะ = TopFrame.Subtitle2 เช่น "232m away"
-- ลูกศร = LeftFrame.IconFrame.GuideIcon
local function readCompassUI()
    local info = {
        texts = {}, distance = nil, title = nil,
        arrowRot = nil, arrowAbs = nil, arrowName = nil,
        navigateBtn = nil, frame = nil,
    }
    local main = pg:FindFirstChild("Main")
    if not main then return info end
    local guide = main:FindFirstChild("Guide")
    if guide then
        info.frame = guide
        local top = guide:FindFirstChild("TopFrame")
        if top then
            local sub1 = top:FindFirstChild("Subtitle1")
            local sub2 = top:FindFirstChild("Subtitle2")
            if sub1 and sub1:IsA("TextLabel") and tostring(sub1.Text) ~= "" then
                info.title = tostring(sub1.Text)
                info.texts[#info.texts+1] = info.title
            end
            if sub2 and sub2:IsA("TextLabel") then
                local t = tostring(sub2.Text)
                info.texts[#info.texts+1] = t
                -- "232m away" | "ห่างออกไป 232เมตร" | "232 m"
                local meters = t:match("(%d+)%s*m%s*away")
                    or t:match("(%d+)%s*เมตร")
                    or t:match("ห่างออกไป%s*(%d+)")
                    or t:match("^(%d+)%s*m")
                if meters then info.distance = tonumber(meters) end
            end
        end
        local track = guide:FindFirstChild("Track")
        if track then info.navigateBtn = track end
        local iconFrame = guide:FindFirstChild("LeftFrame")
            and guide.LeftFrame:FindFirstChild("IconFrame")
        if iconFrame then
            local needle = iconFrame:FindFirstChild("GuideIcon")
            if not needle then
                local bg = iconFrame:FindFirstChild("Background")
                needle = bg and bg:FindFirstChild("Arrow")
            end
            if needle then
                info.arrowRot = needle.Rotation
                info.arrowAbs = needle.AbsoluteRotation
                info.arrowName = needle:GetFullName():gsub("^.-PlayerGui%.", "")
            end
        end
        return info
    end
    return info
end

local function compassWorldDir(arrowRot)
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    local look = cam.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude < 0.05 then flat = Vector3.new(0, 0, -1) else flat = flat.Unit end
    local ang = math.rad(arrowRot or 0)
    local cf = CFrame.lookAt(Vector3.zero, flat)
    local dir = (cf * CFrame.Angles(0, -ang, 0)).LookVector
    return Vector3.new(dir.X, 0, dir.Z).Unit
end

-- "Find the Dragon Crew Quest Giver." → "Dragon Crew Quest Giver"
-- "Find the NPC at Great Tree." → "NPC at Great Tree" (+ place ในค่าที่ 2)
local function parseTargetNameFromTitle(title)
    if not title or title == "" then return nil end
    local t = tostring(title)
    local place = t:match("^[Ff]ind the NPC at%s+(.+)")
    if place then
        place = place:gsub("%.%s*$", ""):gsub("%s+$", ""):gsub("^%s+", "")
        if place ~= "" then return "NPC at " .. place, place end
    end
    local who = t:match("^[Ff]ind the%s+(.+)")
        or t:match("หา%s*(.+)")
        or t
    who = who:gsub("%.%s*$", ""):gsub("%s+$", ""):gsub("^%s+", "")
    if who == "" then return nil end
    return who, nil
end

local function rememberNpc(part, nameOverride)
    if not part then return end
    local name = nameOverride or (part.Parent and part.Parent.Name) or part.Name
    NPC_CACHE[name:lower()] = part.Position
    -- จำสั้นด้วย เช่น "hydra town quest giver"
    local short = name:lower():gsub("%s+", " ")
    NPC_CACHE[short] = part.Position
end

-- สแกน NPC ทั้งแมพ (ไม่จำกัดแค่โฟลเดอร์ NPCs)
local function iterNpcModels(cb)
    local seen = {}
    local function add(model)
        if not model or seen[model] then return end
        if not model:IsA("Model") then return end
        seen[model] = true
        cb(model)
    end
    local folder = Workspace:FindFirstChild("NPCs") or Workspace:FindFirstChild("NPC")
    if folder then
        for _, n in ipairs(folder:GetChildren()) do add(n) end
    end
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Model") then add(obj) end
        if obj:IsA("Folder") or obj:IsA("Model") then
            local nl = obj.Name:lower()
            if nl:find("npc", 1, true) or nl:find("character", 1, true)
               or nl:find("quest", 1, true) or nl:find("town", 1, true) then
                for _, c in ipairs(obj:GetChildren()) do add(c) end
            end
        end
    end
    -- ลึก: ชื่อมี Quest Giver / Hydra / Dragon
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("Model") then
            local nl = d.Name:lower()
            if nl:find("quest giver", 1, true) or nl:find("questgiver", 1, true)
               or nl:find("hydra", 1, true) or nl:find("dragon crew", 1, true)
               or nl:find("great tree", 1, true) or nl:find("marine", 1, true) then
                add(d)
            end
        end
    end
end

local function modelPart(model)
    return model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart")
end

-- หาโมเดลจากชื่อ — ต้องครบคำสำคัญ (ห้ามจับ "Quest Giver" แทน "Hydra Town Quest Giver")
local function findNpcByExactName(targetName)
    if not targetName or targetName == "" then return nil end
    local want = targetName:lower()
    local wantTok = {}
    for w in want:gmatch("%S+") do
        if w ~= "the" and w ~= "a" and w ~= "quest" and w ~= "giver" then
            wantTok[#wantTok+1] = w -- hydra, town, dragon, crew ...
        end
    end
    local _, myHrp = getChar()
    local best, bestScore, bestD, bestName = nil, -1, math.huge, nil

    local function consider(model)
        if not model then return end
        local nl = model.Name:lower()
        local score = -1
        if nl == want then
            score = 100
        elseif nl:find(want, 1, true) then
            score = 90
        else
            -- ทุกคำสำคัญต้องอยู่ในชื่อ NPC
            if #wantTok == 0 then return end
            local ok = true
            for _, t in ipairs(wantTok) do
                if not nl:find(t, 1, true) then ok = false break end
            end
            if not ok then return end
            -- ห้ามจับแค่ "quest giver" เปล่าๆ ถ้าเป้ามีคำเฉพาะ
            if #wantTok >= 1 and (nl == "quest giver" or nl == "questgiver") then return end
            score = 50 + #wantTok
        end
        local part = modelPart(model)
        if not part then return end
        local d = myHrp and (part.Position - myHrp.Position).Magnitude or 0
        if score > bestScore or (score == bestScore and d < bestD) then
            bestScore = score; bestD = d; best = part; bestName = model.Name
        end
    end

    iterNpcModels(consider)

    if best then
        rememberNpc(best, bestName)
        return best, bestD, bestName
    end
    return nil
end

local function npcFromTitle(title)
    local who, place = parseTargetNameFromTitle(title)
    if not who and not place then return nil end
    if place then
        local pk = place:lower()
        if NPC_CACHE[pk] then return NPC_CACHE[pk], "cache:" .. place end
        if KNOWN_NPC[pk] then return KNOWN_NPC[pk], "known:" .. place end
        if NPC_CACHE["npc at " .. pk] then return NPC_CACHE["npc at " .. pk], "cache:npc at " .. place end
        if KNOWN_NPC["npc at " .. pk] then return KNOWN_NPC["npc at " .. pk], "known:npc at " .. place end
    end
    if not who then return nil end
    local key = who:lower()
    if NPC_CACHE[key] then return NPC_CACHE[key], "cache:" .. who end
    if KNOWN_NPC[key] then return KNOWN_NPC[key], "known:" .. who end
    for k, v in pairs(NPC_CACHE) do
        if key:find(k, 1, true) or k:find(key, 1, true) then return v, "cache:" .. k end
    end
    for k, v in pairs(KNOWN_NPC) do
        if key:find(k, 1, true) or k:find(key, 1, true) then return v, "known:" .. k end
    end
    return nil
end

-- เปิดดู Guide ว่ามีการอ้างอิง Instance/Attribute ไป NPC ไหม
local function spyGuideLinks(guide)
    local hits = 0
    if not guide then return 0 end
    log("── ตีกลับ: เปิดไส้ Guide หาลิงก์ NPC ──", "Y")
    for _, d in ipairs(guide:GetDescendants()) do
        if d:IsA("ObjectValue") or d:IsA("NumberValue") or d:IsA("StringValue")
           or d:IsA("CFrameValue") or d:IsA("Vector3Value") then
            hits += 1
            local extra = ""
            pcall(function()
                if d:IsA("ObjectValue") and d.Value then
                    local v = d.Value
                    extra = " → " .. v:GetFullName()
                    local p = posFromAny(v)
                    if p then
                        extra ..= string.format(" @%.1f,%.1f,%.1f", p.X, p.Y, p.Z)
                        local nm = v.Name
                        if v:IsA("BasePart") and v.Parent then nm = v.Parent.Name end
                        NPC_CACHE[nm:lower()] = p
                        log("  ★ ObjectValue." .. d.Name .. extra, "G")
                    else
                        log("  ObjectValue." .. d.Name .. extra, "B")
                    end
                elseif d:IsA("CFrameValue") then
                    local cf = d.Value
                    extra = string.format(" CF=%.1f,%.1f,%.1f", cf.X, cf.Y, cf.Z)
                    log("  CFrameValue." .. d.Name .. extra, "G")
                elseif d:IsA("Vector3Value") then
                    local v = d.Value
                    extra = string.format(" V3=%.1f,%.1f,%.1f", v.X, v.Y, v.Z)
                    log("  Vector3Value." .. d.Name .. extra, "G")
                else
                    log("  " .. d.ClassName .. "." .. d.Name .. "=" .. tostring(d.Value):sub(1, 60), "S")
                end
            end)
        end
    end
    for name, val in pairs(guide:GetAttributes()) do
        hits += 1
        log(string.format("  attr Guide.%s=%s", name, tostring(val)), "B")
    end
    -- ViewportFrame อาจมีโมเดลเป้า
    local vp = guide:FindFirstChild("LeftFrame")
        and guide.LeftFrame:FindFirstChild("IconFrame")
        and guide.LeftFrame.IconFrame:FindFirstChild("ViewportFrame")
    if vp then
        for _, c in ipairs(vp:GetDescendants()) do
            if c:IsA("Model") or c:IsA("WorldModel") then
                hits += 1
                log("  Viewport มี " .. c.ClassName .. ":" .. c.Name, "B")
            end
            if c:IsA("ObjectValue") and c.Value then
                hits += 1
                log("  Viewport ObjectValue → " .. c.Value:GetFullName(), "G")
            end
        end
    end
    if hits == 0 then log("  ไม่เจอ ObjectValue/Attribute ใน Guide (เกมซ่อนในสคริปต์)", "O") end
    return hits
end

    if targetName then

                realName or targetName, dist or 0, part.Position.X, part.Position.Y, part.Position.Z), "G")

        end
        log("→ ยังไม่โหลดในแมพ: " .. targetName, "O")
    end

    -- 2) cache / known จากชื่อเดิม
    local cached, csrc = npcFromTitle(ui.title)

        return cached, csrc

    if findQuestGiver then
        local part2 = findQuestGiver(quest, targetName)
        if part2 then
            rememberNpc(part2)
            return part2.Position, "quest→NPCs"
        end
    end

    -- 4) ไม่มีพิกัด → ให้ resolve ใช้ COMPASS_STEP
    return nil, ui
end

-- probing: 8 ทิศราบ + ขึ้น/ลง/เฉียง (หลังมี findNpc แล้ว)
local function probe8Phase(targetName, placeName, meters)
    local _, hrp = getChar()
    if not hrp then return false end
    local angles = { 0, 45, 90, 135, 180, 225, 270, 315 }
    local yOffs = { 0, 120, 280, 450, 700, 1000, -60 }
    local n = #angles * #yOffs
    local idx = probeStep % n
    probeStep += 1
    local ai = (idx % #angles) + 1
    local yi = math.floor(idx / #angles) % #yOffs + 1
    local ang = math.rad(angles[ai])
    local hop = math.clamp((meters or 200) * 3 * 0.28, 100, 550)
    local dir = Vector3.new(math.cos(ang), 0, math.sin(ang))
    local dest = hrp.Position + dir * hop + Vector3.new(0, yOffs[yi], 0)
    if dest.Y < 5 then dest = Vector3.new(dest.X, 5, dest.Z) end
    log(string.format("Probe 8ทิศ #%d ang=%d Y%+d hop=%.0f",
        idx + 1, angles[ai], yOffs[yi], hop), "Y")
    flyTo(CFrame.new(dest), "probe")
    task.wait(0.7)
    local part
    if targetName then part = select(1, findNpcByExactName(targetName)) end
    if not part and placeName then
        part = select(1, findNpcByExactName(placeName))
    end
    if not part then
        local _, myHrp = getChar()
        local best, bestD = nil, math.huge
        iterNpcModels(function(model)
            local nl = model.Name:lower()
            if not (nl:find("quest giver", 1, true) or nl:find("questgiver", 1, true)) then return end
            local p = modelPart(model)
            if not p or not myHrp then return end
            local d = (p.Position - myHrp.Position).Magnitude
            if d < bestD and d < 700 then bestD = d; best = p end
        end)
        part = best
    end
    if part then
        rememberNpc(part)
        log(string.format("★ Probe เจอ NPC @ %.0f,%.0f,%.0f",
            part.Position.X, part.Position.Y, part.Position.Z), "G")
        flyTo(CFrame.new(part.Position + Vector3.new(0, 6, 0)), "NPC/probe")
        compassStuck = 0
        return true
    end
    local ui = readCompassUI()
    if ui.distance and meters and ui.distance < meters - 5 then
        log(string.format("Probe เมตรลด %s → %s", tostring(meters), tostring(ui.distance)), "G")
        compassStuck = 0
        return true
    end
    return false
end

-- บินตามเข็มทิศ / จุด known / Probe 8ทิศ+ขึ้นลง
local function flyByCompassStep(opts)
    opts = opts or {}
    local ui = readCompassUI()
    local d0 = ui.distance
    local rot = ui.arrowAbs or ui.arrowRot
    local targetName, placeName = parseTargetNameFromTitle(ui.title)
    local isPlaceGuide = placeName ~= nil and placeName ~= ""
    local tallIsland = isPlaceGuide and (
        placeName:lower():find("tree", 1, true)
        or placeName:lower():find("great", 1, true)
    )

    local function scanAfterMove()
        local part
        if targetName then part = select(1, findNpcByExactName(targetName)) end
        if not part and placeName then
            part = select(1, findNpcByExactName(placeName))
        end
        -- เกาะ: รับ Quest Giver ตัวใดก็ได้ที่ใกล้จุด known เกาะ
        if not part and (isPlaceGuide or tallIsland) then
            local _, myHrp = getChar()
            local best, bestD = nil, math.huge
            iterNpcModels(function(model)
                local nl = model.Name:lower()
                if not (nl:find("quest giver", 1, true) or nl:find("questgiver", 1, true)) then return end
                local p = modelPart(model)
                if not p or not myHrp then return end
                local d = (p.Position - myHrp.Position).Magnitude
                if d < bestD and d < 900 then bestD = d; best = p end
            end)
            part = best
        end
        if not part and findQuestGiver then
            part = findQuestGiver(nil, placeName and (placeName .. " Quest Giver") or targetName)
        end
        if part then
            rememberNpc(part)
            if placeName then
                NPC_CACHE[("npc at " .. placeName):lower()] = part.Position
                NPC_CACHE[placeName:lower()] = part.Position
            end
            flyTo(CFrame.new(part.Position + Vector3.new(0, 6, 0)), "NPC")
            compassStuck = 0
            return true
        end
        return false
    end

    if not opts.skipNpcCache and scanAfterMove() then
        return true, 0
    end

    if not d0 or rot == nil then
        -- ไม่มีระยะ → probe เลยถ้าเป็นเกาะ
        if isPlaceGuide then
            probe8Phase(targetName, placeName, 300)
            scanAfterMove()
            return true, 0
        end
        return false, "ไม่มีระยะ/ลูกศรจาก Guide UI"
    end

    if not opts.skipNpcCache then
        local cached, csrc = npcFromTitle(ui.title)
        if cached then
            log(string.format("มีพิกัดแล้ว (%s) — ไม่เดาลูกศร", csrc), "G")
            flyTo(CFrame.new(cached + Vector3.new(0, 8, 0)), "NPC/" .. csrc)
            task.wait(0.9)
            if scanAfterMove() then return true, 0 end
        end
        -- จุด known เกาะ (Great Tree หลายชั้น Y)
        local key = targetName and targetName:lower() or ""
        local key2 = placeName and placeName:lower() or ""
        if (key ~= "" or key2 ~= "") and not triedNpcKnown[key .. key2] then
            triedNpcKnown[key .. key2] = true
            local pts = KNOWN_NPC_ALT[key] or KNOWN_NPC_ALT[key2]
                or (KNOWN_NPC[key] and { KNOWN_NPC[key] })
                or (KNOWN_NPC[key2] and { KNOWN_NPC[key2] })
            if pts then
                log("ไปจุด known เกาะก่อนเดาเข็ม (" .. #pts .. " จุด)", "B")
                for i, pt in ipairs(pts) do
                    log(string.format("ลองจุด NPC known#%d @ %.0f,%.0f,%.0f", i, pt.X, pt.Y, pt.Z), "B")
                    flyTo(CFrame.new(pt + Vector3.new(0, 10, 0)), "NPC known")
                    task.wait(1.0)
                    if scanAfterMove() then return true, 0 end
                    local uiK = readCompassUI()
                    if uiK.distance and d0 and uiK.distance < d0 - 20 then
                        d0 = uiK.distance
                        log("ใกล้ขึ้นหลัง known → " .. d0 .. "ม.", "G")
                        break
                    end
                end
            end
        end
    end

    if d0 <= 12 then
        if scanAfterMove() then return true, 0 end
        -- ใกล้แล้วแต่ยังไม่เจอ NPC → probe ขึ้นลงรอบตัว
        if probe8Phase(targetName, placeName, d0) and scanAfterMove() then return true, 0 end
        return false, "ใกล้แล้ว (" .. d0 .. "ม.)"
    end

    -- stuck / เกาะสูง → Probe 8 ทิศ + Y
    if compassStuck >= 1 or (tallIsland and compassStuck >= 0 and probeStep > 0 and compassStuck >= 1) then
        if compassStuck >= 1 then
            if probe8Phase(targetName, placeName, d0) then
                return true, readCompassUI().distance or d0
            end
            if scanAfterMove() then return true, 0 end
        end
    end
    if tallIsland and compassStuck >= 1 then
        -- บังคับ probe แทนบินราบ Y=26
        probe8Phase(targetName, placeName, d0)
        scanAfterMove()
        local uiP = readCompassUI()
        local dP = uiP.distance
        if dP and dP < d0 - 3 then
            compassStuck = 0
            return true, dP
        end
        compassStuck += 1
        return true, dP or d0
    end

    if ui.navigateBtn then
        pcall(function() firesignal(ui.navigateBtn.MouseButton1Click) end)
    end

    local dir = compassWorldDir(rot)
    local _, hrp = getChar()
    if not dir or not hrp then return false, "ไม่มีทิศ/ตัวละคร" end

    local fullStuds = d0 * 3
    -- อย่ากระโดด 2000 ทุกครั้ง — ครึ่งระยะพอประมาณ
    local hopStuds = math.clamp(fullStuds * 0.55, 80, 1400)
    if compassStuck >= 2 then
        hopStuds = math.clamp(fullStuds * 0.4, 100, 800)
        local off = (compassStuck % 2 == 0) and 40 or -40
        local ang = math.rad(off)
        dir = Vector3.new(
            dir.X * math.cos(ang) - dir.Z * math.sin(ang),
            0,
            dir.X * math.sin(ang) + dir.Z * math.cos(ang)
        ).Unit
    end

    local yBoost = tallIsland and 180 or (opts.highY and 60 or 35)
    local dest = hrp.Position + dir * hopStuds + Vector3.new(0, yBoost, 0)
    if tallIsland then
        dest = Vector3.new(dest.X, math.max(dest.Y, hrp.Position.Y + 80, 120), dest.Z)
    end
    hrp.CFrame = CFrame.new(hrp.Position, Vector3.new(dest.X, hrp.Position.Y, dest.Z))
    log(string.format("บินตรงระยะ UI: %dม. ≈ %.0f studs Y=%.0f (rot=%.0f)",
        d0, hopStuds, dest.Y, rot), "B")
    flyTo(CFrame.new(dest), "เข็มทิศ")
    task.wait(0.7)
    if scanAfterMove() then return true, 0 end

    local ui2 = readCompassUI()
    local d1 = ui2.distance
    if d1 and d1 < d0 - 3 then
        log(string.format("เมตรลด %d → %d ✓", d0, d1), "G")
        compassStuck = 0
        lastMeter = d1
        return true, d1
    end
    if d1 and d1 >= d0 - 3 then
        compassStuck += 1
        log(string.format("เมตรไม่ลด %d → %s stuck=%d → เข้า Probe", d0, tostring(d1), compassStuck), "O")
        lastMeter = d1 or d0
        if compassStuck >= 1 then
            probe8Phase(targetName, placeName, d1 or d0)
            scanAfterMove()
        end
        if compassStuck >= 3 and compassStuck % 3 == 0 then
            dumpNpcList()
        end
    end
    return true, d1 or d0
end

-- หา Quest Giver จากเควส (สำรอง ถ้า Guide ไม่มีชื่อ)
findQuestGiver = function(quest, preferName)
    local folder = Workspace:FindFirstChild("NPCs")
    if not folder then return nil end
    local _, myHrp = getChar()
    local best, bestD = nil, math.huge
    local preferL = preferName and preferName:lower() or nil

    -- ชื่อจากเข็มทิศตรงๆ ก่อน
    if preferL then
        for _, npc in ipairs(folder:GetChildren()) do
            if npc.Name:lower() == preferL or npc.Name:lower():find(preferL, 1, true) then
                local part = npc:FindFirstChild("HumanoidRootPart")
                    or npc:FindFirstChildWhichIsA("BasePart")
                if part then return part, 0 end
            end
        end
    end

    local keys = {}
    if quest then
        local mob = quest.mob or ""
        local words = {}
        for w in mob:gmatch("%S+") do words[#words+1] = w end
        if #words >= 2 then
            keys[#keys+1] = words[1] .. " " .. words[2]
        end
        if #words >= 1 then keys[#keys+1] = words[1] end -- "Hydra" จาก Hydra Enforcer
        keys[#keys+1] = mob
        keys[#keys+1] = quest.displayName or ""
        keys[#keys+1] = quest.qName or ""
    end
    if preferName then
        for w in preferName:gmatch("%S+") do
            if w:lower() ~= "quest" and w:lower() ~= "giver" and w:lower() ~= "the" then
                keys[#keys+1] = w
            end
        end
    end
    keys[#keys+1] = "Quest Giver"
    for _, npc in ipairs(folder:GetChildren()) do
        local nl = npc.Name:lower()
        local hit = false
        if preferL and (nl == preferL or nl:find(preferL, 1, true)) then
            hit = true
        end
        if not hit then
            hit = nl:find("quest giver", 1, true) or nl:find("questgiver", 1, true)
        end
        if not hit and quest then
            for _, k in ipairs(keys) do
                if k ~= "" and nl:find(k:lower(), 1, true) then hit = true break end
            end
        end
        -- อย่าเลือก Dragon Crew ถ้าเป้าเป็น Hydra Town (และกลับกัน)
        if hit and preferL then
            if preferL:find("hydra", 1, true) and nl:find("dragon", 1, true) then hit = false end
            if preferL:find("dragon", 1, true) and nl:find("hydra", 1, true) then hit = false end
        end
        if hit then
            local part = npc:FindFirstChild("HumanoidRootPart")
                or npc:FindFirstChildWhichIsA("BasePart")
            if part then
                local d = myHrp and (part.Position - myHrp.Position).Magnitude or 0
                if d < bestD then bestD = d; best = part end
            end
        end
    end
    return best, bestD
end

-- รวมแหล่งหาเป้า: ตีกลับชื่อจากเข็มทิศเป็นหลัก
local function resolveNpcTarget(quest)
    local pos, srcOrUi = reverseCompassToPos(quest)
    if pos then return pos, srcOrUi end

    local ui = type(srcOrUi) == "table" and srcOrUi or readCompassUI()
    local rot = ui.arrowAbs or ui.arrowRot
    if ui.distance and rot ~= nil and ui.distance > 5 then
        return "COMPASS_STEP", "compassUI(ยังไม่รู้พิกัด)"
    end
    return nil, "ยังไม่มีชื่อ/พิกัดจากเข็มทิศ"
end

local function findMob(mobName)
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return nil end
    local _, myHrp = getChar()
    local best, bestD = nil, math.huge
    local keys = { mobName }
    if mobName and mobName:find(" ", 1, true) then
        local last = mobName:match("(%S+)$")
        if last and #last >= 4 then keys[#keys+1] = last end
    end
    for _, e in ipairs(enemies:GetChildren()) do
        local match = false
        for _, k in ipairs(keys) do
            if k and e.Name:find(k, 1, true) then match = true break end
        end
        if match then
            local hum = e:FindFirstChildOfClass("Humanoid")
            local hrp = e:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local d = myHrp and (hrp.Position - myHrp.Position).Magnitude or 0
                if d < bestD then
                    bestD = d; best = e
                    if mobName then MOB_CACHE[mobName:lower()] = hrp.Position end
                end
            end
        end
    end
    return best, bestD
end

-- บอสจริงใน Enemies มักมีแท็ก [Boss] หรือชื่อในรายการ — ห้ามจับ Staff/เมนูมั่ว
local KNOWN_BOSS_NAMES = {
    "hydra leader", "stone", "island empress", "kilo admiral",
    "captain elephant", "beautiful pirate", "cake queen", "longma",
    "soul reaper", "rip_indra", "dough king", "cake prince",
    "darkbeard", "tide keeper", "cursed captain",
}

local function isBossMobName(name)
    local raw = tostring(name or "")
    local n = raw:lower()
    if n == "" then return false end
    if n:find("enforcer", 1, true) or n:find("assailant", 1, true) then return false end
    if n:find("staff", 1, true) or n:find("crew leader", 1, true) then return false end
    -- แท็กเกม
    if raw:find("%[Boss%]") or raw:find("%[Raid Boss%]") then return true end
    for _, k in ipairs(KNOWN_BOSS_NAMES) do
        if n == k or n:find(k, 1, true) then return true end
    end
    return false
end

-- สแกนบอสที่เกิดอยู่ตอนนี้ใน Enemies
local function scanAliveBosses()
    local list = {}
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return list end
    local _, myHrp = getChar()
    for _, e in ipairs(enemies:GetChildren()) do
        if isBossMobName(e.Name) then
            local hum = e:FindFirstChildOfClass("Humanoid")
            local hrp = e:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local d = myHrp and (hrp.Position - myHrp.Position).Magnitude or 0
                list[#list+1] = {
                    model = e, name = e.Name, hp = hum.Health,
                    p = hrp.Position, d = d,
                }
                MOB_CACHE[e.Name:lower()] = hrp.Position
            end
        end
    end
    table.sort(list, function(a, b) return a.d < b.d end)
    return list
end

local function findAliveBoss(nameHint)
    local want = (nameHint or ""):lower()
    for _, it in ipairs(scanAliveBosses()) do
        if want == "" or it.name:lower():find(want, 1, true) or want:find(it.name:lower(), 1, true) then
            return it.model, it.d, it
        end
    end
    if want ~= "" then
        return findMob(nameHint)
    end
    return nil
end

local function pickBossQuestForLevel(lv)
    local data = getQuests()
    if not data then return nil end
    local best = nil
    for qName, entry in pairs(data) do
        if type(entry) == "table" then
            for num, sub in pairs(entry) do
                if type(sub) == "table" and type(sub.LevelReq) == "number"
                   and sub.LevelReq <= lv and type(sub.Task) == "table" then
                    local mob, need = next(sub.Task)
                    if mob and type(need) == "number" then
                        local cand = {
                            qName = qName, qNum = tonumber(num) or num,
                            mob = mob, need = need, levelReq = sub.LevelReq,
                            displayName = sub.Name or mob,
                        }
                        if isBossQuest(cand) then
                            if not best or cand.levelReq > best.levelReq
                               or (cand.levelReq == best.levelReq and cand.qNum > best.qNum) then
                                best = cand
                            end
                        end
                    end
                end
            end
        end
    end
    return best
end

local function mobSpawnHint(mobName)
    local key = (mobName or ""):lower()
    if MOB_CACHE[key] then return MOB_CACHE[key], "cache" end
    if KNOWN_MOB[key] then return KNOWN_MOB[key], "known" end
    return nil
end

local spiralR = 150
local spiralStep = 0
local spiralOrigin = nil
local triedKnownHigh = false

local function resetSpiral(origin, keepTried)
    spiralR = 150
    spiralStep = 0
    spiralOrigin = origin
    if not keepTried then
        triedKnownHigh = false
    end
end

local function nextSpiralPoint(mobName)
    local _, hrp = getChar()
    if not hrp then return nil end
    if not spiralOrigin then
        -- ศูนย์กลาง = จุดมอนเควสเท่านั้น (ห้าม Warrior / ตำแหน่งตัวเองต่ำ)
        local hint = mobSpawnHint(mobName)
            or mobSpawnHint((pickQuestForLevel(getLevel()) or {}).mob or "")
        if hint then
            local minY = (tostring(mobName or ""):lower():find("hydra", 1, true)) and 550 or 380
            spiralOrigin = Vector3.new(hint.X, math.max(hint.Y, minY), hint.Z)
        else
            spiralOrigin = hrp.Position + Vector3.new(0, 250, 0)
            if spiralOrigin.Y < 350 then
                spiralOrigin = Vector3.new(spiralOrigin.X, 380, spiralOrigin.Z)
            end
        end
        log(string.format("ก้นหอยเริ่มที่ %.0f,%.0f,%.0f รัศมี %d",
            spiralOrigin.X, spiralOrigin.Y, spiralOrigin.Z, spiralR), "Y")
    end
    local ang = spiralStep * (math.pi / 4)
    spiralStep += 1
    if spiralStep >= 8 then
        spiralStep = 0
        spiralR += 150
        if spiralR > 1200 then
            spiralR = 150
            -- รีเซ็ตกลับจุด known ของมอนเควส ไม่ใช่ตำแหน่งตัว (กันหลงเกาะล่าง)
            local hint = mobSpawnHint(mobName)
            if hint then
                spiralOrigin = Vector3.new(hint.X, math.max(hint.Y, 380), hint.Z)
                log("ก้นหอยครบ → กลับจุด known สูง", "O")
            else
                spiralOrigin = Vector3.new(hrp.Position.X, 400, hrp.Position.Z)
                log("ก้นหอยครบ → เริ่มใหม่รอบตัว (สูง)", "O")
            end
        else
            log("ก้นหอยขยายรัศมี → " .. spiralR .. " studs", "Y")
        end
    end
    local baseY = math.max(spiralOrigin.Y, 200)
    -- ห้ามบังคับ Y≤520 — Hydra Town อยู่ ~1000
    local y = baseY + (spiralR > 400 and 40 or 0)
    return Vector3.new(
        spiralOrigin.X + math.cos(ang) * spiralR,
        y,
        spiralOrigin.Z + math.sin(ang) * spiralR
    )
end

local function logNearbyEnemies(radius)
    local enemies = Workspace:FindFirstChild("Enemies")
    local _, hrp = getChar()
    if not enemies or not hrp then return end
    local list = {}
    for _, e in ipairs(enemies:GetChildren()) do
        local eh = e:FindFirstChild("HumanoidRootPart")
        local hum = e:FindFirstChildOfClass("Humanoid")
        if eh and hum and hum.Health > 0 then
            local d = (eh.Position - hrp.Position).Magnitude
            if d <= radius then
                list[#list+1] = { name = e.Name, d = d, p = eh.Position }
            end
        end
    end
    table.sort(list, function(a, b) return a.d < b.d end)
    if #list == 0 then
        log(string.format("  ในรัศมี %d: ไม่มีมอนโหลด", radius), "O")
        return
    end
    log(string.format("  ในรัศมี %d: %d ตัว", radius, #list), "B")
    for i = 1, math.min(8, #list) do
        local it = list[i]
        log(string.format("  • %s d=%.0f @ %.0f,%.0f,%.0f",
            it.name, it.d, it.p.X, it.p.Y, it.p.Z), "S")
    end
end

local function huntMobOrSweep(quest)
    local mob, md = findMob(quest.mob)
    if mob then return mob, md end

    -- เปลี่ยนมอนเควส → รีเซ็ตก้นหอย
    if spiralOrigin and quest and quest.mob then
        local hint = mobSpawnHint(quest.mob)
        if hint and (spiralOrigin - hint).Magnitude > 800 then
            resetSpiral(Vector3.new(hint.X, hint.Y, hint.Z))
        end
    end

    -- หลุดเกาะ/ดีดกลับไกลจากจุดมอน → บินช้ากลับ known ก่อนก้นหอย
    do
        local hint = mobSpawnHint(quest.mob)
        local _, hrp0 = getChar()
        if hint and hrp0 and (hrp0.Position - hint).Magnitude > 2500 then
            log(string.format("อยู่ไกลจุดมอน %.0f → บินกลับ known ก่อน",
                (hrp0.Position - hint).Magnitude), "O")
            flyTo(CFrame.new(hint + Vector3.new(0, 40, 0)), "กลับเกาะมอน")
            task.wait(0.5)
            mob, md = findMob(quest.mob)
            if mob then return mob, md end
            resetSpiral(Vector3.new(hint.X, math.max(hint.Y, 200), hint.Z))
        end
    end

    -- มีจุดที่เคยเจอ Archer แล้ว → บินกลับที่นั่นก่อน (ห้ามไป NPC)
    local cached = MOB_CACHE[(quest.mob or ""):lower()]
    if cached then
        log(string.format("กลับจุดมอนที่จำไว้ @ %.0f,%.0f,%.0f", cached.X, cached.Y, cached.Z), "G")
        flyTo(CFrame.new(cached + Vector3.new(0, 30, 0)), "cacheมอน")

    local ui = readCompassUI()

        and not titleL:find("quest giver", 1, true)

            task.wait(isBossQuest(quest) and 0.8 or 1.4)
            mob, md = findMob(quest.mob)
            if mob then return mob, md end
            logNearbyEnemies(1500)
        end
        if isBossQuest(quest) then
            -- นับหนักหลังกวาดจุด known แล้วไม่เจอ
            bossMissCount += 2
            log(string.format("บอสไม่เจอหลังจุด known (สะสม %d/6)", bossMissCount), "Y")
        end
    end

    local pt = nextSpiralPoint(quest.mob)
    if not pt then return nil end
    log(string.format("ก้นหอย r=%d @ %.0f,%.0f,%.0f", spiralR, pt.X, pt.Y, pt.Z), "B")
    flyTo(CFrame.new(pt), "ก้นหอย")
    task.wait(1.4)
    mob, md = findMob(quest.mob)
    if not mob then logNearbyEnemies(1500) end

            for _, t in ipairs(src:GetChildren()) do

local function equipSword()
    local char, _, hum = getChar()
    if not char or not hum then return false end
    local held = char:FindFirstChildOfClass("Tool")
    if held and tostring(held.ToolTip or "") == "Sword" then return true end
    local sword = getToolByTip("Sword")
    if sword then
        pcall(function() hum:EquipTool(sword) end)
        task.wait(0.15)
        return true
    end
    -- fallback กด 3 ตามที่ใช้อยู่
    pressKey(Enum.KeyCode.Three, 0.08)
    task.wait(0.2)
    held = char:FindFirstChildOfClass("Tool")
    return held ~= nil
end

end

local function guiReallyVisible(gui)
    if not gui or not gui:IsA("GuiObject") then return false end
    local a = gui
    while a do
        if a:IsA("GuiObject") and not a.Visible then return false end
        if a:IsA("LayerCollector") and a:IsA("ScreenGui") and not a.Enabled then return false end
        if a:IsA("ScreenGui") and not a.Enabled then return false end
        a = a.Parent
    end
    return gui.AbsoluteSize.X > 10 and gui.AbsoluteSize.Y > 8
end

-- อ่าน progress เควส เช่น (5/8) หรือ 5/8 — กันตัวเลขมั่วจาก HP/เงิน
local function parseProgress(tx)
    if not tx then return nil, nil end
    local s = tostring(tx):gsub(",", "")
    -- เฉพาะข้อความสั้นแบบ progress
    local a, b = s:match("%((%d+)%s*/%s*(%d+)%)")
    if not a then
        -- ทั้งสตริงเป็นแค่ "0/1" หรือมีชื่อมอนนำ "Hydra Leader 0/1"
        a, b = s:match("(%d+)%s*/%s*(%d+)%s*$")
    end
    if not a or not b then return nil, nil end
    local cur, need = tonumber(a), tonumber(b)
    if not cur or not need then return nil, nil end
    if need < 1 or need > 30 then return nil, nil end -- เควสปกติไม่เกิน 30
    if cur > need + 2 then return nil, nil end -- กัน 772/1 จาก UI อื่น
    if #s > 40 and not s:lower():find("defeat", 1, true) and not s:find("เอาชนะ", 1, true)
       and not s:lower():find("leader", 1, true) and not s:lower():find("hydra", 1, true) then
        return nil, nil
    end
    return cur, need
end

local function readQuestProgress(quest)
    local title, cur, need, src = "", nil, nil, nil

    local function considerTitle(tx, from)
        if not tx or tx == "" then return end
        local tl = tx:lower()
        if tl:find("bandit", 1, true) then return end
        if not (tx:find("เอาชนะ", 1, true) or tl:find("defeat", 1, true)) then return end
        title = tx
        src = from
        local a, b = parseProgress(tx)
        if a and b then cur, need = a, b end
        if not need then
            need = tonumber(tx:match("[Dd]efeat%s*(%d+)") or tx:match("เอาชนะ%s*(%d+)") or "")
        end
        -- "Defeat Hydra Leader" ไม่มีเลข → ใช้ need จากเควส (บอส = 1)
        if not need and quest and quest.need then need = quest.need end
    end

    local function takeProgress(a, b)
        if not a or not b then return end
        if b < 1 or b > 30 or a > b + 2 then return end
        cur, need = a, b
    end

    local ok, vis = pcall(function() return pg.Main.Quest.Visible end)
    -- อ่านข้อความใน Main.Quest แม้ Visible=false
    do
        local t = ""
        pcall(function() t = pg.Main.Quest.Container.QuestTitle.Title.Text or "" end)
        local tl = t:lower()
        if t ~= "" and not tl:find("bandit", 1, true) then
            considerTitle(t, vis == true and "Main.Quest" or "Main.Quest.hidden")
        end
        pcall(function()
            for _, d in ipairs(pg.Main.Quest:GetDescendants()) do
                if d:IsA("TextLabel") then
                    local tx = tostring(d.Text or "")
                    local a, b = parseProgress(tx)
                    if a and b then takeProgress(a, b) end
                    local tl2 = tx:lower()
                    if title == "" and (tl2:find("defeat", 1, true) or tx:find("เอาชนะ", 1, true))
                       and not tl2:find("bandit", 1, true) then
                        considerTitle(tx, vis == true and "Main.Quest" or "Main.Quest.hidden")
                    end
                end
            end
        end)
    end

    if title == "" then
        for _, d in ipairs(pg:GetDescendants()) do
            if d:IsA("TextLabel") or d:IsA("TextButton") then
                if not (sg and d:IsDescendantOf(sg)) and guiReallyVisible(d) then
                    local tx = tostring(d.Text or "")
                    if #tx > 0 and #tx < 90 and not tx:find("%[%d%d:%d%d:%d%d%]")
                       and not tx:find("มีเควสแล้ว", 1, true) then
                        considerTitle(tx, "onScreen")
                        if title ~= "" then
                            -- หา progress ใกล้ๆ บนจอ เช่น "Hydra Leader 0/1"
                            break
                        end
                    end
                end
            end
        end
    end

    -- progress แยกช่อง เช่น "0/8" หรือ "Hydra Leader 0/1"
    if title ~= "" then
        local expect = (quest and quest.need) or need or 8
        for _, d in ipairs(pg:GetDescendants()) do
            if d:IsA("TextLabel") and not (sg and d:IsDescendantOf(sg)) and guiReallyVisible(d) then
                local tx = tostring(d.Text or "")
                local a, b = parseProgress(tx)
                if a and b and a <= b and b <= 30 then
                    local tl = tx:lower()
                    local last = quest and quest.mob and (quest.mob:match("(%S+)$") or "") or ""
                    local mobOk = last ~= "" and tl:find(last:lower(), 1, true)
                    if b == expect or mobOk or (b >= 2 and b <= 20 and #tx <= 28 and a <= b) then
                        takeProgress(a, b)
                    end
                end
            end
        end
    end

    -- บอส/ไม่มีเลขในชื่อ
    if title ~= "" and not need and quest and quest.need then
        need = quest.need
    end
    if title ~= "" and need and not cur then cur = 0 end
    if cur and need and cur > need then cur = 0 end

    if title == "" then return nil, nil, "", nil end
    return cur, need, title, src
end

local function hasActiveQuest(quest)
    local cur, need, title, src = readQuestProgress(quest)

    -- ครบแล้ว → รับใหม่
    if cur and need and need > 0 and cur >= need then
        forceQuestUntil = 0
        forceQuestLabel = ""
        return false, string.format("ครบ%d/%d", cur, need)
    end

    -- มีข้อความ Defeat/เอาชนะ บนจอ = มีเควส (สำคัญกว่า Guide)
    if title ~= "" and not title:lower():find("bandit", 1, true) then
        local tl = title:lower()
        if tl:find("defeat", 1, true) or title:find("เอาชนะ", 1, true) then
            if cur and need then
                return true, string.format("%s (%d/%d)", title, cur, need)
            end
            return true, title
        end
    end

    -- เพิ่ง StartQuest สำเร็จ — เชื่อ force ก่อน (อย่าให้ Guide ล้าง)
    if os.clock() < forceQuestUntil then
        return true, forceQuestLabel ~= "" and forceQuestLabel or "(รอUIเควส)"
    end

    -- ไม่มี Defeat บนจอ + Guide ชี้ Giver = ยังไม่มีเควส
    if guideWantsQuestGiver() then
        return false, ""
    end

    local okVis, vis = pcall(function() return pg.Main.Quest.Visible end)
    if okVis and vis == true and title ~= "" then
        return true, title
    end

    return false, ""
end

local function dumpQuestUI()
    log("════ Spy เควส UI (รับแล้ว/จบแล้ว?) ════", "Y")
    local quest = pickQuestForLevel(getLevel())
    local cur, need, title, src = readQuestProgress(quest)
    local main = pg:FindFirstChild("Main")
    if main then
        local q = main:FindFirstChild("Quest")
        if q then
            log(string.format("Main.Quest Visible=%s", tostring(q.Visible)), q.Visible and "G" or "O")
            for _, d in ipairs(q:GetDescendants()) do
                if d:IsA("TextLabel") or d:IsA("TextButton") then
                    local tx = tostring(d.Text or "")
                    if tx ~= "" and #tx < 80 then
                        log(string.format("  %s txt=%s onScreen=%s", d.Name, tx, tostring(guiReallyVisible(d))), "S")
                    end
                end
            end
        end
    end
    if title ~= "" then
        log(string.format("ข้อความเควส [%s]: %s", tostring(src), title), "G")
    else
        log("ไม่เจอข้อความเควสบนจอ", "O")
    end
    if cur and need then
        log(string.format("★ Progress: %d / %d %s", cur, need, cur >= need and "← จบแล้ว ต้องรับใหม่" or ""), cur >= need and "Y" or "G")
    else
        log("ยังอ่าน x/y ไม่ได้ (จะใช้ข้อความ Defeat/เอาชนะ แทน)", "O")
    end
    local ui = readCompassUI()
    if ui.title then log("Guide: " .. ui.title, "B") end
    if guideWantsQuestGiver() then
        log("Guide ชี้ Quest Giver — ถ้ามี Defeat บนจอ = ยังมีเควสอยู่", "Y")
    end
    local on, t2 = hasActiveQuest(quest)
    log(string.format("สรุป hasActiveQuest=%s | %s", tostring(on), t2 ~= "" and t2 or "(ว่าง)"), on and "G" or "R")
    if on then
        log("→ AUTO ควรไปตีมอน ไม่วน StartQuest", "G")
    else
        log("→ AUTO จะไปรับเควสที่ NPC", "Y")
    end
    log("════ จบ Spy เควส — Copy ส่งมาได้ ════", "Y")
end

local function refreshStatus(quest)
    local lv = getLevel()
    local q = quest and string.format("%s#%s %s", quest.qName, tostring(quest.qNum), quest.mob) or "-"
    local cur, need, title = readQuestProgress(quest)
    local on, t2 = hasActiveQuest(quest)
    local ui
    if not on then
        ui = "OFF"
        if t2:find("ครบ", 1, true) then ui = "DONE:" .. t2 end
    else
        if cur and need then
            ui = string.format("ON:%d/%d %s", cur, need, (title ~= "" and title or t2):sub(1, 28))
        else
            ui = "ON:" .. (t2 ~= "" and t2:sub(1, 40) or "?")
        end
    end
    statusLbl.Text = string.format("Lv.%d | QuestUI %s\nเป้า: %s", lv, ui, q)
end

                end
            end
        else
            log("ไม่มี QuestTracingLine ใน Effect.Container", "O")
        end
    end)
    local ui2 = readCompassUI()
    local knownP, knownS = npcFromTitle(ui2.title)
    if knownP then
        log(string.format("พิกัดจาก cache/known (%s): %.1f, %.1f, %.1f",
            knownS, knownP.X, knownP.Y, knownP.Z), "G")
    else
        log("ยังไม่มีพิกัด cache สำหรับหัวข้อนี้ (จะจำตอนเจอใน NPCs)", "O")
    end
    log("════ จบ Dump — Copy ส่งมาได้ ════", "Y")
end

-- ── Main loop ──────────────────────────────────────────────────────
local function mainLoop()
    if _G.BF04C_LOOPING then
        -- ลูปเก่าค้าง (โหลดซ้ำ / Invoke ค้าง) → บังคับรีสตาร์ท
        log("AUTO ค้าง → รีสตาร์ทลูป", "O")
        _G.BF04C_GEN = (_G.BF04C_GEN or 0) + 1
        MY_GEN = _G.BF04C_GEN
        hardStopFly()
        _G.BF04C_LOOPING = false
        task.wait(0.3)
    end
    _G.BF04C_LOOPING = true
    local loopGen = MY_GEN
    log("=== START Compass Auto ===", "Y")
    local cf = getCommF()
    if not cf then log("ไม่มี CommF_", "R"); _G.BF04C_LOOPING = false; return end
    getCombatRemotes()
    pcall(function() ensureBuddhaSword(true) end)

    while _G.BF04C_ON and _G.BF04C_GEN == loopGen do
        local ok, err = pcall(function()
            local lv = getLevel()
            local bossQ = pickBossQuestForLevel(lv)
            local bossAlive = bossQ and findAliveBoss(bossQ.mob)

            if bossAlive then
                if os.clock() < bossSkipUntil or bossMissCount > 0 then
                    log("★ Spy บอสเกิดแล้ว → สลับไป " .. bossQ.mob, "G")
                end
                bossSkipUntil = 0
                bossMissCount = 0
            end

            -- เลือกเควส: บอสเกิดถึงรับบอส — ไม่เกิดใช้เควสก่อนหน้า
            local quest = pickQuestForLevel(lv)
            if quest and isBossQuest(quest) and not bossAlive then
                local prev = pickPreviousFarmQuest(lv, quest)
                if prev then
                    log(string.format("บอสยังไม่เกิด → รับเควสก่อนหน้า %s#%s %s",
                        prev.qName, tostring(prev.qNum), prev.mob), "Y")
                    bossSkipUntil = math.max(bossSkipUntil, os.clock() + 180)
                    quest = prev
                end
            elseif quest and isBossQuest(quest) and os.clock() < bossSkipUntil and not bossAlive then
                local prev = pickPreviousFarmQuest(lv, quest)
                if prev then quest = prev end
            end

            -- ฟาร์มอยู่ แต่บอสเพิ่งเกิด → สลับ
            if quest and not isBossQuest(quest) and bossAlive then
                local hasFarm = select(1, hasActiveQuest(quest))
                local hasBoss = select(1, hasActiveQuest(bossQ))
                if hasFarm and not hasBoss then
                    log("บอสเกิด → Abandon เควสฟาร์ม แล้วไปรับเควสบอส", "Y")
                    invokeCF(cf, "AbandonQuest")
                    forceQuestUntil = 0
                    bossSkipUntil = 0
                    task.wait(0.5)
                    return
                end
                quest = bossQ
            end
            if not quest then
                log("ไม่มีเควสที่ LevelReq <= " .. lv, "R")
                task.wait(3)
                return
            end
            refreshStatus(quest)

            -- ── มีเควสแล้ว? ──
            local hasQ, qTxt = hasActiveQuest(quest)

            -- ติดเควสบอสอยู่ แต่บอสไม่เกิด → ทิ้งทันที ไปฟาร์ม
            if hasQ and qTxt then
                local ql = qTxt:lower()
                if (ql:find("leader", 1, true) or ql:find("boss", 1, true))
                   and not findAliveBoss(bossQ and bossQ.mob or "Hydra Leader") then
                    log("มีเควสบอสแต่บอสไม่เกิด → Abandon กลับไปฟาร์ม", "O")
                    invokeCF(cf, "AbandonQuest")
                    forceQuestUntil = 0
                    bossSkipUntil = os.clock() + 180
                    bossMissCount = 0
                    resetSpiral(nil)
                    task.wait(0.6)
                    return
                end
            end

            if not hasQ then
                log("ยังไม่มีเควส (Quest UI ปิด) → ไปรับที่ NPC", "Y")
                local pos, src = resolveNpcTarget(quest)
                if pos == "COMPASS_STEP" then
                    setNoclip(true)
                    flyByCompassStep()
                    local part2 = findQuestGiver(quest)
                    if part2 then
                        log("โหลด NPC แล้วหลังตามเข็มทิศ", "G")
                    end
                    task.wait(0.2)
                    return
                end
                if not pos then
                    log("รอเป้า NPC... (" .. tostring(src) .. ")", "Y")
                    local mob, md = findMob(quest.mob)
                    if mob and md and md > 80 then
                        local mhrp = mob:FindFirstChild("HumanoidRootPart")
                        if mhrp then flyTo(mhrp.CFrame * CFrame.new(0, 20, 0), "ใกล้เกาะมอน") end
                    end
                    task.wait(1.5)
                    return
                end
                local _, hrp = getChar()
                local dist = hrp and (hrp.Position - pos).Magnitude or 9999
                local nearDist = isBigForm() and 55 or 28
                if dist > nearDist then
                    local arrived = flyTo(CFrame.new(pos + Vector3.new(0, 6, 0)), "NPC/" .. src)
                    task.wait(0.35)
                    local _, hCheck = getChar()
                    local left = hCheck and (hCheck.Position - pos).Magnitude or 9999
                    if not arrived or left > 150 then
                        log(string.format("ยังไม่ถึง NPC (เหลือ %.0f) — ไม่ StartQuest", left), "Y")
                        task.wait(0.4)
                        return
                    end
                end
                -- ยืนยันระยะอีกครั้งก่อนรับเควส
                do
                    local _, hNow0 = getChar()
                    local dGate = hNow0 and (hNow0.Position - pos).Magnitude or 9999
                    if dGate > 80 then
                        log(string.format("ระยะยังไกล %.0f — บินช้าเข้าใหม่", dGate), "O")
                        flyTo(CFrame.new(pos + Vector3.new(0, 6, 0)), "NPC/retry")
                        task.wait(0.4)
                        local _, h2 = getChar()
                        dGate = h2 and (h2.Position - pos).Magnitude or 9999
                        if dGate > 100 then
                            log("ยังไกลอยู่ ข้ามรอบนี้", "R")
                            return
                        end
                    end
                end
                equipSword()
                for _ = 1, 4 do
                    local _, h = getChar()
                    if h then h.CFrame = CFrame.new(pos + Vector3.new(0, 4, 0)) end
                    task.wait(0.12)
                end
                local _, hNow = getChar()
                local dNow = hNow and (hNow.Position - pos).Magnitude or -1
                log(string.format("StartQuest(%s, %s) ร่างใหญ่=%s ระยะ=%.0f",
                    quest.qName, tostring(quest.qNum), tostring(isBigForm()), dNow), "Y")
                local iok, ires = invokeCF(cf, "StartQuest", quest.qName, quest.qNum)
                log("ผล: ok=" .. tostring(iok) .. " res=" .. tostring(ires), "B")
                if iok and ires ~= "timeout" then
                    forceQuestUntil = os.clock() + 25
                    forceQuestLabel = quest.mob or "StartQuest"
                    resetSpiral(nil) -- รับเควสใหม่ → อย่าใช้ก้นหอยเกาะเก่า
                    log("ตั้งโหมดมีเควส 25วิ (รอ UI)", "G")
                elseif ires == "timeout" then
                    log("StartQuest timeout — ลองใหม่", "R")
                    task.wait(1)
                    return
                end
                -- คลิกปุ่มเลือกเควสใน UI ถ้ามี (Warrior / Archer)
                pcall(function()
                    local main = pg:FindFirstChild("Main")
                    if not main then return end
                    local want = quest.mob or ""
                    for _, d in ipairs(main:GetDescendants()) do
                        if (d:IsA("TextButton") or d:IsA("ImageButton") or d:IsA("TextLabel")) then
                            local tx = tostring(d.Text or d.Name or "")
                            if want ~= "" and tx:find(want, 1, true) then
                                local btn = d:IsA("GuiButton") and d or d.Parent
                                if btn and btn:IsA("GuiButton") then
                                    log("คลิกเลือกเควส UI: " .. tx, "G")
                                    pcall(function() firesignal(btn.MouseButton1Click) end)
                                    pcall(function()
                                        local vim = game:GetService("VirtualInputManager")
                                        local a, s = btn.AbsolutePosition, btn.AbsoluteSize
                                        vim:SendMouseButtonEvent(a.X+s.X/2, a.Y+s.Y/2, 0, true, game, 1)
                                        task.wait(0.05)
                                        vim:SendMouseButtonEvent(a.X+s.X/2, a.Y+s.Y/2, 0, false, game, 1)
                                    end)
                                    break
                                end
                            end
                        end
                    end
                end)
                task.wait(0.6)
                local w = 0
                while not select(1, hasActiveQuest(quest)) and w < 30
                      and _G.BF04C_ON and _G.BF04C_GEN == loopGen do
                    local _, h = getChar()
                    if h then h.CFrame = CFrame.new(pos + Vector3.new(0, 4, 0)) end
                    task.wait(0.1); w += 1
                end
                hasQ, qTxt = hasActiveQuest(quest)
                if hasQ then
                    log("รับเควสสำเร็จ: " .. tostring(qTxt), "G")
                else
                    log("ยังไม่เจอว่ามีเควส (ลองใหม่ / กด Spy เควส)", "R")
                    task.wait(1.0)
                end
                return
            end

            log("มีเควสแล้ว: " .. tostring(qTxt) .. " → ไปตีมอน", "G")

            local titleTxt = qTxt ~= "" and qTxt or questTitle()
            local mobKey = (quest.mob or ""):lower()
            local titleOk = titleTxt ~= "" and (
                (mobKey ~= "" and titleTxt:lower():find(mobKey:sub(1, math.min(10, #mobKey)), 1, true))
                or (quest.displayName and titleTxt:find(quest.displayName, 1, true))
            )
            if questUIVisible() and titleTxt ~= "" and not titleOk then
                log("เควสไม่ตรง (" .. titleTxt .. ") → Abandon แล้วรอเคลียร์", "Y")
                invokeCF(cf, "AbandonQuest")
                local t0 = os.clock()
                while questUIVisible() and os.clock() - t0 < 3
                      and _G.BF04C_ON and _G.BF04C_GEN == loopGen do
                    task.wait(0.15)
                end
                if questUIVisible() then
                    log("Abandon แล้วยังเปิดอยู่ — รอสักครู่", "O")
                    task.wait(1.5)
                end
                return
            end

            -- ก่อนตี: ให้มีร่างพระ + ดาบ
            ensureBuddhaSword(false)

            local mob, md = huntMobOrSweep(quest)
            if not mob then
                if isBossQuest(quest) then
                    bossMissCount += 1
                    log(string.format("บอสยังไม่เกิด (%d/6) — หาต่อหรือจะสลับเควส", bossMissCount), "Y")
                    if bossMissCount >= 6 then
                        local prev = pickPreviousFarmQuest(lv, quest)
                        log("★ บอสไม่เจอ → Abandon แล้วกลับเควสก่อนหน้า"
                            .. (prev and (": " .. prev.mob) or ""), "O")
                        invokeCF(cf, "AbandonQuest")
                        forceQuestUntil = 0
                        bossMissCount = 0
                        bossSkipUntil = os.clock() + 300
                        resetSpiral(nil)
                        task.wait(0.8)
                        return
                    end
                else
                    bossMissCount = 0
                end
                log("ยังไม่เจอ " .. quest.mob .. " → ไล่จุดต่อไป", "Y")
                task.wait(0.4)
                return
            end
            if isBossQuest(quest) then bossMissCount = 0 end
            local mhrp = mob:FindFirstChild("HumanoidRootPart")
            local mhum = mob:FindFirstChildOfClass("Humanoid")
            if not mhrp or not mhum then return end

            if (md or 999) > SAFE_HIT_RANGE then
                -- เข้าใกล้ช้าๆ แล้ววาร์ปบนหัว (ไม่ Tween ระยะสั้น)
                if (md or 999) > 80 then
                    flyTo(mhrp.CFrame * CFrame.new(0, attackHeight() + 10, 0), "มอน")
                end
                local _, hNear = getChar()
                if hNear then lockOnMobHead(mob, hNear) end
                task.wait(0.15)
                return
            end

            log(string.format("สู้ %s HP%.0f (d=%.0f big=%s) ล็อกบนหัว",
                mob.Name, mhum.Health, md or 0, tostring(isBigForm())), "Y")
            local lastHp = mhum.Health
            local lastCheck = os.clock()
            local noDmg = 0
            local heightTry = { 10, 14, 18, 22, 8, 26 }
            local hi = 1
            while mob.Parent and mhum.Parent and mhum.Health > 0
                  and select(1, hasActiveQuest(quest))
                  and _G.BF04C_ON and _G.BF04C_GEN == loopGen do
                local _, myHrp = getChar()
                if not myHrp then break end
                mhrp = mob:FindFirstChild("HumanoidRootPart")
                if not mhrp then break end
                equipSword()
                lockOnMobHead(mob, myHrp, heightTry[hi] or attackHeight())
                attackMob(mob)
                if os.clock() - lastCheck >= 1.0 then
                    lastCheck = os.clock()
                    local hp = mhum.Health
                    if hp >= lastHp - 1 then
                        noDmg += 1
                        -- ลองความสูงใหม่ก่อนข้ามตัว
                        if noDmg >= 2 then
                            hi = (hi % #heightTry) + 1
                            log("ลองสูงบนหัว=" .. tostring(heightTry[hi]), "O")
                        end
                        if noDmg >= 8 then
                            log("ตีไม่เข้า → ข้ามตัวนี้", "Y")
                            break
                        end
                    else
                        noDmg = 0
                        lastHp = hp
                    end
                end
                task.wait(0.05) -- ไม่กระตุกทุกเฟรม
            end
            pcall(function()
                local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
                if hum then hum.PlatformStand = false; hum.AutoRotate = true end
            end)
            log(mob.Name .. " จบรอบ", "G")
            local pc, pn = readQuestProgress(quest)
            if pc and pn then
                log(string.format("เควส progress %d/%d", pc, pn), "B")
                if pc >= pn then
                    forceQuestUntil = 0
                    log("★ เควสครบแล้ว → กลับไปรับใหม่", "Y")
                    task.wait(0.4)
                    return
                end
            elseif guideWantsQuestGiver() then
                forceQuestUntil = 0
                log("★ Guide ชี้ Quest Giver → กลับไปรับใหม่", "Y")
                task.wait(0.4)
                return
            end
            task.wait(0.3)
        end)
        if _G.BF04C_GEN ~= loopGen then break end
        if not ok then
            log("Loop error: " .. tostring(err), "R")
            hardStopFly()
            task.wait(1)
        else
            task.wait(0.15)
        end
    end
    log("=== STOP ===", "R")
    hardStopFly()
    if _G.BF04C_GEN == loopGen then
        _G.BF04C_LOOPING = false
    end
end

-- ── Spy ระยะตี: จำตำแหน่งมอน → เริ่มจากไกลสุด → ค่อยเข้าใกล้ → โดนครั้งแรก = ระยะสูงสุด ──
local spyBusy = false
local function spyAttackRange()
    if spyBusy then log("กำลังวัดอยู่ รอจบก่อน", "Y") return end
    spyBusy = true
    local ok, err = pcall(function()
        log("════ Spy ระยะตี v2 (ไกล→ใกล้) ════", "Y")
        _G.BF04C_ON = false
        autoBtn.Text = "AUTO: OFF"
        autoBtn.BackgroundColor3 = Color3.fromRGB(30, 120, 50)
        task.wait(0.4) -- ให้ AUTO หยุดบินก่อน

        ensureBuddhaSword(true)
        local _, hrp0 = getChar()
        if hrp0 then
            log(string.format("HRP Size=%.1f,%.1f,%.1f | big=%s",
                hrp0.Size.X, hrp0.Size.Y, hrp0.Size.Z, tostring(isBigForm())), "B")
        end
        if not isBigForm() then
            log("⚠ ยังไม่ใช่ร่างใหญ่ — กด Z เองแล้ว Spy ใหม่ได้", "O")
        end

        local quest = pickQuestForLevel(getLevel())
        local mobName = quest and quest.mob or nil
        local mob = mobName and select(1, findMob(mobName)) or nil
        if not mob then
            local enemies = Workspace:FindFirstChild("Enemies")
            local bestD = math.huge
            local _, my = getChar()
            if enemies and my then
                for _, e in ipairs(enemies:GetChildren()) do
                    local hum = e:FindFirstChildOfClass("Humanoid")
                    local eh = e:FindFirstChild("HumanoidRootPart")
                    if hum and eh and hum.Health > 0 then
                        local d = (eh.Position - my.Position).Magnitude
                        if d < bestD then bestD = d; mob = e end
                    end
                end
            end
        end
        if not mob then
            log("ไม่เจอมอน — ยืนใกล้มอนก่อน (อย่าเปิด AUTO)", "R")
            return
        end

        local mhrp = mob:FindFirstChild("HumanoidRootPart")
        local mhum = mob:FindFirstChildOfClass("Humanoid")
        if not mhrp or not mhum then log("มอนไม่มี HRP", "R") return end

        -- จำพิกัดมอนตอนเริ่ม (ล็อกมอนนิ่งระหว่างวัด — กันเดินหนีทำให้ระยะเพี้ยน)
        local anchorPos = mhrp.Position
        log(string.format("เป้า: %s HP%.0f @ %.0f,%.0f,%.0f",
            mob.Name, mhum.Health, anchorPos.X, anchorPos.Y, anchorPos.Z), "G")
        log("วิธี: วางจากไกล 150→5 | รอดาเมจเข้าก่อนค่อยนับระยะจริง", "S")

        local height = isBigForm() and 18 or 8
        local firstHitD = nil -- ระยะจริงครั้งแรกที่โดน (จากไกลเข้ามา)
        local confirmed = {}

        -- วางตัว + รอให้ตำแหน่งนิ่ง (กันวาปเด้ง)
        local function settleAt(horizDist)
            mhrp = mob:FindFirstChild("HumanoidRootPart")
            mhum = mob:FindFirstChildOfClass("Humanoid")
            if not mhrp or not mhum or mhum.Health <= 0 then return nil end
            -- ตรึงมอนที่จุดเดิม
            pcall(function()
                mhrp.Anchored = true
                mhrp.CFrame = CFrame.new(anchorPos)
                mhum.PlatformStand = true
            end)
            local want = math.sqrt(height * height + horizDist * horizDist)
            for try = 1, 6 do
                local _, my = getChar()
                if not my then return nil end
                setNoclip()
                equipSword()
                my.CFrame = CFrame.new(anchorPos) * CFrame.new(0, height, horizDist)
                task.wait(0.25)
                _, my = getChar()
                mhrp = mob:FindFirstChild("HumanoidRootPart")
                if not my or not mhrp then return nil end
                -- ตรึงมอนอีกครั้งหลัง wait
                pcall(function()
                    mhrp.CFrame = CFrame.new(anchorPos)
                end)
                local real = (my.Position - mhrp.Position).Magnitude
                if math.abs(real - want) <= 18 then
                    return real, my
                end
                -- ยังเด้ง → วาร์ปซ้ำ
            end
            local _, my = getChar()
            mhrp = mob:FindFirstChild("HumanoidRootPart")
            if my and mhrp then
                return (my.Position - mhrp.Position).Magnitude, my
            end
            return nil
        end

        -- ไกล → ใกล้
        for horiz = 150, 5, -5 do
            if not mob.Parent or not mhum or mhum.Health <= 0 then
                log("มอนตายกลางทาง หยุดวัด", "Y")
                break
            end
            local realD = settleAt(horiz)
            if not realD then
                log(string.format("ข้ามตั้ง%d (วางตัวไม่ได้)", horiz), "O")
            else
                local want = math.sqrt(height * height + horiz * horiz)
                if math.abs(realD - want) > 25 then
                    log(string.format("~ ไม่นิ่ง ตั้ง%d อยาก=%.0f จริง=%.0f (ข้าม)", horiz, want, realD), "O")
                else
                    local hp0 = mhum.Health
                    for _ = 1, 10 do
                        -- ตรึงตำแหน่งระหว่างยิง
                        local _, my = getChar()
                        if my and mhrp then
                            my.CFrame = CFrame.new(anchorPos) * CFrame.new(0, height, horiz)
                            pcall(function() mhrp.CFrame = CFrame.new(anchorPos) end)
                        end
                        attackMob(mob)
                        task.wait(0.05)
                    end
                    task.wait(0.3)
                    mhum = mob:FindFirstChildOfClass("Humanoid")
                    if not mhum then break end
                    local dmg = hp0 - mhum.Health
                    -- วัดระยะจริงอีกครั้งหลังยิง
                    local _, my2 = getChar()
                    mhrp = mob:FindFirstChild("HumanoidRootPart")
                    local realAfter = (my2 and mhrp) and (my2.Position - mhrp.Position).Magnitude or realD
                    if dmg > 0 then
                        if not firstHitD then
                            firstHitD = realAfter
                            log(string.format("★ โดนครั้งแรก @ จริง=%.0f (ตั้งแนวนอน%d) ดาเมจ=-%.0f",
                                realAfter, horiz, dmg), "G")
                        else
                            log(string.format("✓ ยังโดน @ จริง=%.0f ดาเมจ=-%.0f", realAfter, dmg), "G")
                        end
                        confirmed[#confirmed + 1] = realAfter
                        -- โดนแล้วเก็บอีก 2 ขั้นใกล้เพื่อยืนยัน แล้วจบ
                        if #confirmed >= 3 then
                            log("ยืนยันโดน 3 ระยะแล้ว → จบ", "Y")
                            break
                        end
                    else
                        log(string.format("✗ ไม่โดน @ ตั้ง%d จริง=%.0f", horiz, realAfter), "S")
                        if firstHitD then
                            -- เคยโดนแล้วแต่ระยะนี้ไม่โดน = แปลก ข้าม
                        end
                    end
                end
            end
        end

        -- ปลดมอน
        pcall(function()
            if mhrp then mhrp.Anchored = false end
        end)

        log("──────── สรุป ────────", "Y")
        log(string.format("ร่างใหญ่=%s | สูงทดสอบ=%.0f | ไล่จากไกล→ใกล้", tostring(isBigForm()), height), "B")
        if firstHitD then
            local safe = math.floor(firstHitD * 0.85)
            log(string.format("★ ระยะตีโดนสูงสุด (ครั้งแรกจากไกล) ≈ %.0f studs", firstHitD), "G")
            log(string.format("  แนะนำล็อกตีที่ระยะ ≤ %.0f studs", safe), "G")
        else
            log("ไม่โดนเลย — เช็คดาบ/ร่างพระ หรือมอนตายก่อนวัด", "R")
        end
        log("════ จบ Spy ระยะตี v2 — Copy ส่งมาได้ ════", "Y")
    end)
    if not ok then log("Spy error: " .. tostring(err), "R") end
    spyBusy = false
end
